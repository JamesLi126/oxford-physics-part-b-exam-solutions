\section*{Question 1 --- $LS$-coupling, terms, and the anomalous Zeeman effect in cadmium}

\textbf{Problem statement.} (a) Explain $LS$-coupling and the conditions under which it applies, and list the spectroscopic terms arising from $3\mathrm{s}^{2}$, $3\mathrm{s}\,3\mathrm{p}$, and $3\mathrm{s}\,3\mathrm{d}$. (b) Derive the Land\'e $g$-factor in $LS$-coupling. (c) Cadmium ($5\mathrm{s}^{2}$ ground configuration) is placed in a magnetic flux density of $\SI{1.5}{T}$; the Zeeman patterns of three lines $\lambda_{A}=\SI{340.4}{nm}$, $\lambda_{B}=\SI{346.8}{nm}$, $\lambda_{C}=\SI{361.4}{nm}$ are tabulated, with all three sharing a common upper and a common lower term. Deduce $J$ and $g_{J}$ for the levels of lines $A$ and $C$, and propose $L,S$ values for the two terms. (d) Identify line $B$. (e) List all allowed transitions between these two terms and compute the Zeeman pattern of one of them.

\subsection*{(a) The $LS$-coupling scheme and example terms}

Condition: residual electrostatic repulsion dominates the spin--orbit interaction (light atoms, low $Z$, since spin--orbit $\propto Z^{4}$).

Couple orbital angular momenta first:
\begin{equation}
\vb L=\sum_i\vb\ell_i,\qquad \vb S=\sum_i\vb s_i.
\end{equation}
Couple resultants to $\vb J$:
\begin{equation}
\vb J=\vb L+\vb S.
\end{equation}
Good quantum numbers $L,S,J,M_{J}$; terms labelled $^{2S+1}L_{J}$.

Terms from each configuration:
\begin{itemize}
\item $3\mathrm{s}^{2}$: closed sub-shell with two equivalent electrons; Pauli forces $L=0,S=0$. Term: $\boxed{\,^{1}\mathrm{S}_{0}\,}$.
\item $3\mathrm{s}\,3\mathrm{p}$: $\ell_1=0,\ell_2=1\Rightarrow L=1$; $S=0,1$. Terms: $\boxed{\,^{1}\mathrm{P}_{1},\ ^{3}\mathrm{P}_{0,1,2}\,}$.
\item $3\mathrm{s}\,3\mathrm{d}$: $\ell_1=0,\ell_2=2\Rightarrow L=2$; $S=0,1$. Terms: $\boxed{\,^{1}\mathrm{D}_{2},\ ^{3}\mathrm{D}_{1,2,3}\,}$.
\end{itemize}

\subsection*{(b) Derivation of the Land\'e $g$-factor}

Zeeman Hamiltonian with $\vb B=B\hat{\vb z}$ and $g_{s}\simeq 2$:
\begin{equation}
\hat H_{Z}=\frac{\mu_{B}}{\hbar}\,\vb B\cdot(\hat{\vb L}+2\hat{\vb S}).
\label{eq:HZ}
\end{equation}
Write $\hat{\vb L}+2\hat{\vb S}=\hat{\vb J}+\hat{\vb S}$:
\begin{equation}
\hat H_{Z}=\frac{\mu_{B}}{\hbar}\,\vb B\cdot(\hat{\vb J}+\hat{\vb S}).
\end{equation}
Project $\hat{\vb S}$ onto $\hat{\vb J}$ (perpendicular components average to zero):
\begin{equation}
\hat H_{Z}=g_{J}\,\frac{\mu_{B}}{\hbar}\,B\,\hat J_{z},
\end{equation}
with
\begin{equation}
g_{J}=1+\frac{\langle\hat{\vb S}\cdot\hat{\vb J}\rangle}{\langle\hat{\vb J}^{2}\rangle}.
\label{eq:gJ-defn}
\end{equation}
Expand $\hat{\vb L}^{2}=(\hat{\vb J}-\hat{\vb S})^{2}$:
\begin{equation}
\hat{\vb L}^{2}=\hat{\vb J}^{2}+\hat{\vb S}^{2}-2\hat{\vb S}\cdot\hat{\vb J}.
\end{equation}
Solve for $\hat{\vb S}\cdot\hat{\vb J}$:
\begin{equation}
\langle\hat{\vb S}\cdot\hat{\vb J}\rangle=\tfrac{1}{2}\hbar^{2}\bigl[J(J+1)+S(S+1)-L(L+1)\bigr].
\label{eq:SdotJ}
\end{equation}
Insert into \eqref{eq:gJ-defn} with $\langle\hat{\vb J}^{2}\rangle=\hbar^{2}J(J+1)$:
\begin{equation}
g_{J}=1+\frac{J(J+1)+S(S+1)-L(L+1)}{2J(J+1)}.
\end{equation}
Split $1=3/2-J(J+1)/[2J(J+1)]$:
\begin{equation}
\boxed{\;g_{J}=\tfrac{3}{2}+\frac{S(S+1)-L(L+1)}{2J(J+1)}\;}.
\label{eq:gJ-final}
\end{equation}

\subsection*{(c) $J$ and $g_{J}$ for the levels involved in lines $A$ and $C$}

Compute the Zeeman unit at $B=\SI{1.5}{T}$:
\begin{equation}
\frac{\mu_{B}B}{h}=\SI{13.996}{GHz/T}\times\SI{1.5}{T}=\SI{20.99}{GHz}\approx \SI{21}{GHz}.
\label{eq:muBoverh}
\end{equation}
Identify component shifts with $\Delta f/(\mu_{B}B/h)=g_{u}m_{u}-g_{l}m_{l}$, $\Delta m\in\{0,\pm 1\}$.

\paragraph{Line $A$ ($f_{A},\,f_{A}\pm 10.5$).} Three symmetric components; spacing $10.5\,\text{GHz}=\tfrac{1}{2}\mu_{B}B/h$ is the signature of $J=0\leftrightarrow J=1$ (one level does not split).
Set spacing $g\mu_{B}B/h=\SI{10.5}{GHz}$:
\begin{equation}
g=\tfrac{1}{2}.
\end{equation}
\begin{equation}
\boxed{\;\text{line }A:\ J=0\ (g\text{ undefined}),\quad J=1\ (g=\tfrac{1}{2}).\;}
\label{eq:lineA-Jg}
\end{equation}

\paragraph{Line $C$ (9 components, $f_{C},\,\pm 10.5,\,\pm 21,\,\pm 31.5,\,\pm 52.5$).} Largest shift in units of $\mu_{B}B/h$:
\begin{equation}
\pm 52.5/21=\pm\tfrac{5}{2}.
\end{equation}
Try $J_{u}=2,\,J_{l}=1$ with $g_{l}=1/2$ (shared with line $A$). Extremal $(m_{u},m_{l})=(\pm 2,\pm 1)$:
\begin{equation}
|2g_{u}-g_{l}|=\tfrac{5}{2}.
\end{equation}
Insert $g_{l}=1/2$:
\begin{equation}
2g_{u}-\tfrac{1}{2}=\tfrac{5}{2}.
\end{equation}
Solve for $g_{u}$:
\begin{equation}
g_{u}=\tfrac{3}{2}.
\end{equation}
Tabulate $g_{u}m_{u}-g_{l}m_{l}$ for all $\Delta m\in\{0,\pm 1\}$:
\begin{equation}
\begin{array}{c|c}
(m_{u},m_{l}) & g_{u}m_{u}-g_{l}m_{l} \\ \hline
(\pm 2,\pm 1) & \pm 5/2 \\
(\pm 1,\pm 1) & \pm 1 \\
(\pm 1,0) & \pm 3/2 \\
(0,\pm 1) & \mp 1/2 \\
(0,0) & 0
\end{array}
\label{eq:lineC-table}
\end{equation}
The shift set $\{0,\pm 1/2,\pm 1,\pm 3/2,\pm 5/2\}\times\mu_{B}B/h$ matches the observed components.
\begin{equation}
\boxed{\;\text{line }C:\ J=2\ (g=\tfrac{3}{2}),\quad J=1\ (g=\tfrac{1}{2}).\;}
\label{eq:lineC-Jg}
\end{equation}

\paragraph{Identifying $L,S$.} Apply \eqref{eq:gJ-final} to the shared $J=1,\,g=1/2$ level:
\begin{equation}
\tfrac{3}{2}+\frac{S(S+1)-L(L+1)}{2\cdot 1\cdot 2}=\tfrac{1}{2}.
\end{equation}
Solve for the bracket:
\begin{equation}
S(S+1)-L(L+1)=-4.
\end{equation}
Try $L=2,S=1$:
\begin{equation}
2-6=-4.\ \checkmark
\end{equation}
Hence this term is $^{3}\mathrm{D}_{1}$. Apply \eqref{eq:gJ-final} to the $J=2,\,g=3/2$ level:
\begin{equation}
\tfrac{3}{2}+\frac{S(S+1)-L(L+1)}{2\cdot 2\cdot 3}=\tfrac{3}{2}.
\end{equation}
Hence:
\begin{equation}
S(S+1)-L(L+1)=0.
\end{equation}
Try $L=1,S=1$:
\begin{equation}
2-2=0.\ \checkmark
\end{equation}
Hence this term is $^{3}\mathrm{P}_{2}$, with $J=0$ partner $^{3}\mathrm{P}_{0}$ on line $A$.
\begin{equation}
\boxed{\;\text{two terms: }^{3}\mathrm{P}\ (L=1,S=1)\text{ and }^{3}\mathrm{D}\ (L=2,S=1).\;}
\label{eq:terms}
\end{equation}

\subsection*{(d) Identification of line $B$}

Line $B$ has six components but no central ($\pi$) component. Missing $m=0\to m'=0$ is the signature of $J\to J$ with $J=J'\neq 0$. The only available $J=1$ levels are $^{3}\mathrm{P}_{1}$ and $^{3}\mathrm{D}_{1}$.

Compute $g(^{3}\mathrm{P}_{1})$ from \eqref{eq:gJ-final} with $L=S=J=1$:
\begin{equation}
g(^{3}\mathrm{P}_{1})=\tfrac{3}{2}+\frac{2-2}{4}=\tfrac{3}{2}.
\end{equation}
Recall $g(^{3}\mathrm{D}_{1})=1/2$ from part~(c).

Take $\{g_{u},g_{l}\}=\{3/2,1/2\}$ and list $g_{u}m_{u}-g_{l}m_{l}$ with $\Delta m\in\{0,\pm 1\}$, $0\to 0$ forbidden:
\begin{equation}
\{\pm\tfrac{1}{2},\pm 1,\pm\tfrac{3}{2}\}\times\mu_{B}B/h=\{\pm 10.5,\pm 21,\pm 31.5\}\,\mathrm{GHz}.
\end{equation}
Pattern matches the observed six components.
\begin{equation}
\boxed{\;\text{line }B:\ ^{3}\mathrm{P}_{1}\leftrightarrow{}^{3}\mathrm{D}_{1}.\;}
\label{eq:lineB}
\end{equation}

\section*{Question 2 --- Multi-electron atoms, quantum defects, alkali spectra}

\textbf{Problem statement.} (a) State the approximations underlying the radial Schr\"odinger equation for the $i$-th electron of a multi-electron atom and sketch $V(r)$. (b) Identify the rubidium absorption series at $\{794.8, 780.0, 421.6, 420.2, 359.2, 358.7\}\,\mathrm{nm}$, compute the quantum defects $\delta_{l}$, estimate the next doublet wavelength, and explain why the doublets close up. (c) Why is $\delta_{4d}=1.23$ smaller than the $\delta_{l}$ found in (b), and is the 4d level more or less bound than 6s? (d) Identify the spontaneous-emission lines from the $\lambda=421.6\,\mathrm{nm}$ excitation. (e) Krypton's ground configuration; estimate the energy needed to excite a 5d electron.

\subsection*{(a) Approximations and the effective potential}

Approximations underlying the radial equation:
\begin{enumerate}
\item \textbf{Central field}: replace electron--electron repulsion by an angle-averaged spherical mean field $V(r_{i})$.
\item \textbf{Independent electron}: total wavefunction is a Slater determinant of single-electron orbitals.
\item \textbf{Separation of variables}: $\psi=R_{n\ell}(r)Y_{\ell m}(\theta,\phi)$, substitute $P_{n\ell}=rR_{n\ell}$ to obtain the 1D radial equation with centrifugal term $\hbar^{2}\ell(\ell+1)/(2mr^{2})$.
\item \textbf{Self-consistency}: iterate $V(r)$ from the orbitals (Hartree--Fock).
\end{enumerate}

Limits: $V(r)\to-Ze^{2}/4\pi\epsilon_{0}r$ as $r\to 0$ (bare nucleus); $V(r)\to-e^{2}/4\pi\epsilon_{0}r$ as $r\to\infty$ (fully screened):
\begin{center}
\begin{tikzpicture}[xscale=1.0,yscale=0.7]
\draw[->] (0,0)--(7,0) node[right]{$r$};
\draw[->] (0,-3.5)--(0,0.5) node[above]{$V(r)$};
\draw[domain=0.4:6.5,smooth,thick] plot (\x,{-3.0/\x*(0.7+2.0*exp(-1.2*\x))/(0.7+2.0)});
\draw[dashed] (0.5,-3.0)--(7,0) node[pos=0.05]{$\sim -Ze^{2}/4\pi\epsilon_{0}r$};
\draw[dashed,domain=2:6.8] plot (\x,{-1.0/\x}) ;
\node at (5.5,-0.5) {$\sim -e^{2}/4\pi\epsilon_{0}r$};
\end{tikzpicture}
\end{center}
For $\ell>0$ the centrifugal barrier excludes the electron from the core: high-$\ell$ orbitals are hydrogenic.

\subsection*{(b) Identifying the Rb absorption series}

Rubidium: single $5\mathrm{s}_{1/2}$ valence electron above a Kr-like core. E1 selection rule $\Delta\ell=\pm 1$ permits only the principal series $5\mathrm{s}\to n\mathrm{p}_{1/2,3/2}$; six lines form three fine-structure doublets for $n=5,6,7$.

Quantum-defect formula:
\begin{equation}
E_{n\ell}=-\frac{R_{\infty}}{(n-\delta_{\ell})^{2}},\qquad R_{\infty}=\SI{13.606}{eV}.
\label{eq:QDF}
\end{equation}
Apply to the $5\mathrm{s}$ ground state with $|E_{5\mathrm{s}}|=\SI{4.177}{eV}$:
\begin{equation}
n^{*}_{5\mathrm{s}}=\sqrt{\frac{13.606}{4.177}}=1.804.
\end{equation}
Hence:
\begin{equation}
\delta_{5\mathrm{s}}=5-1.804=3.20.
\label{eq:delta-5s}
\end{equation}
Procedure for each line: $|E_{n\mathrm{p}}|=|E_{5\mathrm{s}}|-hc/\lambda$, $n^{*}=\sqrt{R_{\infty}/|E_{n\mathrm{p}}|}$, $\delta=n-n^{*}$. Use $hc=\SI{1239.84}{eV.nm}$:
\begin{equation}
\begin{array}{c|c|c|c|c|c}
\lambda/\mathrm{nm} & hc/\lambda/\mathrm{eV} & |E_{n\mathrm{p}}|/\mathrm{eV} & n^{*} & n & \delta_{n\mathrm{p}} \\ \hline
794.8 & 1.5599 & 2.6171 & 2.280 & 5 & 2.720 \\
780.0 & 1.5895 & 2.5875 & 2.293 & 5 & 2.707 \\
421.6 & 2.9408 & 1.2362 & 3.318 & 6 & 2.682 \\
420.2 & 2.9505 & 1.2265 & 3.331 & 6 & 2.669 \\
359.2 & 3.4517 & 0.7253 & 4.331 & 7 & 2.669 \\
358.7 & 3.4566 & 0.7204 & 4.346 & 7 & 2.654
\end{array}
\label{eq:Rb-table}
\end{equation}
Identify the longer-$\lambda$ (more bound) member as $n\mathrm{p}_{1/2}$, with $\delta\simeq 2.71$; the shorter-$\lambda$ member is $n\mathrm{p}_{3/2}$, with $\delta\simeq 2.66$. The defect is roughly $n$-independent.

\paragraph{Next doublet ($n=8$).} Take a mean $\delta_{8\mathrm{p}}\simeq 2.66$:
\begin{equation}
(8-2.66)^{2}=5.34^{2}=28.52.
\end{equation}
Compute the binding energy:
\begin{equation}
|E_{8\mathrm{p}}|=\frac{13.606}{28.52}=\SI{0.477}{eV}.
\label{eq:E8p}
\end{equation}
Transition energy:
\begin{equation}
\Delta E=4.177-0.477=\SI{3.700}{eV}.
\end{equation}
Mean wavelength:
\begin{equation}
\boxed{\;\bar\lambda_{8\mathrm{p}}=\frac{1239.84}{3.700}=\SI{335.1}{nm}.\;}
\label{eq:lambda8p}
\end{equation}

\paragraph{Why doublets close up.} Spin--orbit splitting scales as $\langle r^{-3}\rangle_{n\mathrm{p}}\propto (n-\delta)^{-3}$:
\begin{equation}
\Delta E_{\mathrm{FS}}(n\mathrm{p})\propto \frac{1}{(n-\delta)^{3}}.
\end{equation}
Observed splittings $\SI{29.6}{meV}$ ($5\mathrm{p}$), $\SI{9.7}{meV}$ ($6\mathrm{p}$), $\SI{4.8}{meV}$ ($7\mathrm{p}$) follow this scaling.

\subsection*{(c) Quantum defect of the 4d level; comparison with 6s binding energy}

Centrifugal barrier $\propto \ell(\ell+1)/r^{2}$ excludes high-$\ell$ orbitals from the core, leaving them in the screened single-charge tail and pushing $\delta_{\ell}\to 0$. Hence $\delta_{4\mathrm{d}}=1.23$ is much smaller than $\delta_{5\mathrm{s}}=3.20$ and $\delta_{n\mathrm{p}}\simeq 2.7$.

Compute $|E_{4\mathrm{d}}|$ from \eqref{eq:QDF}:
\begin{equation}
(4-1.23)^{2}=2.77^{2}=7.673.
\end{equation}
\begin{equation}
|E_{4\mathrm{d}}|=\frac{13.606}{7.673}=\SI{1.773}{eV}.
\label{eq:E4d}
\end{equation}
Compute $|E_{6\mathrm{s}}|$ with $\delta_{6\mathrm{s}}\simeq\delta_{5\mathrm{s}}=3.20$:
\begin{equation}
(6-3.20)^{2}=2.80^{2}=7.840.
\end{equation}
\begin{equation}
|E_{6\mathrm{s}}|=\frac{13.606}{7.840}=\SI{1.736}{eV}.
\label{eq:E6s}
\end{equation}
Compare:
\begin{equation}
\boxed{\;|E_{4\mathrm{d}}|>|E_{6\mathrm{s}}|:\ \text{4d is (slightly) more tightly bound than 6s.}\;}
\end{equation}

\subsection*{(e) Krypton: ground configuration and 5d-excitation energy}

Fill shells up to $Z=36$:
\begin{equation}
\boxed{\;\mathrm{Kr}:\ [\mathrm{Ar}]\,3\mathrm{d}^{10}\,4\mathrm{s}^{2}\,4\mathrm{p}^{6}.\;}
\end{equation}
Promote a $4\mathrm{p}$ electron: lowest $5\mathrm{d}$-bearing configuration is $4\mathrm{p}^{5}\,5\mathrm{d}$. The $5\mathrm{d}$ electron sees a Kr$^{+}$ core (net charge $+e$); $\ell=2$ is non-penetrating, so $\delta_{5\mathrm{d}}\approx 0$ and the binding is hydrogenic.

Apply the hydrogenic formula with $n=5$:
\begin{equation}
|E_{5\mathrm{d}}|\simeq \frac{R_{\infty}}{n^{2}}=\frac{13.606}{25}=\SI{0.544}{eV}.
\label{eq:E5d-Kr}
\end{equation}
Excitation $=$ ionisation cost minus 5d binding:
\begin{equation}
\Delta E\simeq E_{i}-|E_{5\mathrm{d}}|=14.0-0.544\,\mathrm{eV}.
\end{equation}
\begin{equation}
\boxed{\;\Delta E_{4\mathrm{p}^{6}\to 4\mathrm{p}^{5}5\mathrm{d}}\simeq \SI{13.5}{eV}.\;}
\label{eq:Kr-excitation}
\end{equation}

\section*{Question 3 --- Two- and three-level dynamics, Rabi flopping, and two-photon transitions}

\textbf{Problem statement.} A two-level atom with eigenstates $\ket{1},\ket{2}$ and energies $E_{1}<E_{2}$ is driven by $\hat V(t)=ex\mathcal{E}\cos\omega t$, with general time-dependent state $\Psi=c_{1}\psi_{1}\mathrm{e}^{-iE_{1}t/\hbar}+c_{2}\psi_{2}\mathrm{e}^{-iE_{2}t/\hbar}$. (a) Derive $\dot c_{2}$ in the form quoted, defining the Rabi frequency. (b) Solve in the rotating-wave approximation (RWA) for $|c_{2}(t)|^{2}$ when $c_{1}\simeq 1$. (c) For a three-level $\Lambda$-like system driven near a two-photon resonance with $|\Delta|\gg|\delta|$, derive $|c_{3}(t)|^{2}=(W^{2}/\beta^{2})\sin^{2}(\beta t)$ and compare with the one-photon result. (d) Justify the simplification of the more general coupling matrix.

\subsection*{(a) Coupled equations and Rabi frequency}

Substitute $\Psi=c_{1}\psi_{1}\mathrm{e}^{-iE_{1}t/\hbar}+c_{2}\psi_{2}\mathrm{e}^{-iE_{2}t/\hbar}$ into $i\hbar\dot\Psi=(\hat H_{0}+\hat V)\Psi$ and use $\hat H_{0}\psi_{j}=E_{j}\psi_{j}$ to cancel the $E_{j}$ terms:
\begin{equation}
i\hbar\bigl[\dot c_{1}\psi_{1}\mathrm{e}^{-iE_{1}t/\hbar}+\dot c_{2}\psi_{2}\mathrm{e}^{-iE_{2}t/\hbar}\bigr]=\hat V(t)\bigl[c_{1}\psi_{1}\mathrm{e}^{-iE_{1}t/\hbar}+c_{2}\psi_{2}\mathrm{e}^{-iE_{2}t/\hbar}\bigr].
\label{eq:TDSE-2lvl}
\end{equation}
Project onto $\psi_{2}^{*}$ using $\langle\psi_{i}|\psi_{j}\rangle=\delta_{ij}$:
\begin{equation}
i\hbar\dot c_{2}\mathrm{e}^{-iE_{2}t/\hbar}=c_{1}\langle 2|\hat V|1\rangle\mathrm{e}^{-iE_{1}t/\hbar}+c_{2}\langle 2|\hat V|2\rangle\mathrm{e}^{-iE_{2}t/\hbar}.
\label{eq:project2}
\end{equation}
Drop $\langle 2|\hat x|2\rangle=0$ by parity:
\begin{equation}
i\hbar\dot c_{2}\mathrm{e}^{-iE_{2}t/\hbar}=c_{1}\,e\mathcal{E}\cos\omega t\,\langle 2|\hat x|1\rangle\,\mathrm{e}^{-iE_{1}t/\hbar}.
\end{equation}
Define $\omega_{21}=(E_{2}-E_{1})/\hbar$ and the Rabi frequency:
\begin{equation}
\boxed{\;\Omega\equiv\frac{e\mathcal{E}}{\hbar}\langle 2|\hat x|1\rangle.\;}
\label{eq:Rabi-defn}
\end{equation}
Divide by $i\hbar\,\mathrm{e}^{-iE_{2}t/\hbar}$:
\begin{equation}
\boxed{\;\dot c_{2}(t)=-i\Omega\cos(\omega t)\,\mathrm{e}^{i\omega_{21}t}c_{1}(t).\;}
\label{eq:c2dot}
\end{equation}

\subsection*{(b) Weak excitation in the RWA}

Set $c_{1}\simeq 1$, $c_{2}(0)=0$; integrate \eqref{eq:c2dot}:
\begin{equation}
c_{2}(t)=-i\Omega\int_{0}^{t}\cos(\omega t')\mathrm{e}^{i\omega_{21}t'}\de t'.
\end{equation}
Expand $\cos\omega t'=(\mathrm{e}^{i\omega t'}+\mathrm{e}^{-i\omega t'})/2$:
\begin{equation}
c_{2}(t)=-\frac{i\Omega}{2}\int_{0}^{t}\bigl[\mathrm{e}^{i(\omega_{21}+\omega)t'}+\mathrm{e}^{i(\omega_{21}-\omega)t'}\bigr]\de t'.
\label{eq:c2-int}
\end{equation}
Drop the fast $\mathrm{e}^{i(\omega_{21}+\omega)t'}$ term (RWA):
\begin{equation}
c_{2}(t)\simeq -\frac{i\Omega}{2}\int_{0}^{t}\mathrm{e}^{i(\omega_{21}-\omega)t'}\de t'.
\end{equation}
Evaluate the integral:
\begin{equation}
c_{2}(t)\simeq -\frac{i\Omega}{2}\,\frac{\mathrm{e}^{i(\omega_{21}-\omega)t}-1}{i(\omega_{21}-\omega)}.
\end{equation}
Simplify:
\begin{equation}
c_{2}(t)\simeq -\frac{\Omega}{2}\,\frac{\mathrm{e}^{i(\omega_{21}-\omega)t}-1}{\omega_{21}-\omega}.
\end{equation}
Apply $\mathrm{e}^{ix}-1=2i\mathrm{e}^{ix/2}\sin(x/2)$:
\begin{equation}
c_{2}(t)\simeq -i\Omega\,\frac{\mathrm{e}^{i(\omega_{21}-\omega)t/2}\sin\!\bigl[\tfrac{1}{2}(\omega_{21}-\omega)t\bigr]}{\omega_{21}-\omega}.
\label{eq:c2-RWA}
\end{equation}
Take the modulus squared:
\begin{equation}
\boxed{\;|c_{2}(t)|^{2}=\frac{\Omega^{2}}{(\omega-\omega_{21})^{2}}\sin^{2}\!\Bigl[\tfrac{1}{2}(\omega-\omega_{21})t\Bigr].\;}
\label{eq:c2sq}
\end{equation}
On resonance ($\omega=\omega_{21}$) the result $\to\Omega^{2}t^{2}/4$.

\subsection*{(c) Two-photon transition $1\to 3$ via off-resonant level 2}

Apply RWA to the $\dot c_{2}$ equation, keeping the slow phase $\omega_{21}-\omega_{a}=-\Delta$:
\begin{equation}
\dot c_{2}\simeq -\frac{i\Omega_{a}}{2}\mathrm{e}^{-i\Delta t}c_{1}.
\label{eq:c2dot-3lvl}
\end{equation}
For $\dot c_{3}$, identify the slow phase. Use $\omega_{23}=\omega_{21}-\omega_{31}$ and the given $\omega_{a}-\omega_{b}=\omega_{31}+\delta$, i.e.\ $\omega_{b}=\omega_{a}-\omega_{31}-\delta$:
\begin{equation}
-\omega_{23}+\omega_{b}=-(\omega_{21}-\omega_{31})+(\omega_{a}-\omega_{31}-\delta)=\omega_{a}-\omega_{21}-\delta.
\end{equation}
Simplify:
\begin{equation}
-\omega_{23}+\omega_{b}=\Delta-\delta.
\end{equation}
Hence:
\begin{equation}
\dot c_{3}\simeq -\frac{i\Omega_{b}}{2}\mathrm{e}^{i(\Delta-\delta)t}c_{2}.
\label{eq:c3dot-3lvl}
\end{equation}

\paragraph{Adiabatic elimination of level 2.} Ansatz $c_{2}=A(t)\mathrm{e}^{-i\Delta t}$ with $|\dot A|\ll|\Delta A|$ (Rabi rates $\ll |\Delta|$). Substitute into \eqref{eq:c2dot-3lvl} with $c_{1}\simeq 1$:
\begin{equation}
\dot A\,\mathrm{e}^{-i\Delta t}-i\Delta A\,\mathrm{e}^{-i\Delta t}=-\frac{i\Omega_{a}}{2}\mathrm{e}^{-i\Delta t}.
\end{equation}
Drop $\dot A$ and cancel $\mathrm{e}^{-i\Delta t}$:
\begin{equation}
-i\Delta A=-\frac{i\Omega_{a}}{2}.
\end{equation}
Solve for $A$:
\begin{equation}
A=\frac{\Omega_{a}}{2\Delta}.
\end{equation}
\begin{equation}
c_{2}(t)\simeq \frac{\Omega_{a}}{2\Delta}\mathrm{e}^{-i\Delta t}.
\label{eq:c2-adiab}
\end{equation}

\paragraph{Effective two-photon coupling.} Substitute \eqref{eq:c2-adiab} into \eqref{eq:c3dot-3lvl}:
\begin{equation}
\dot c_{3}\simeq -\frac{i\Omega_{b}}{2}\,\mathrm{e}^{i(\Delta-\delta)t}\cdot\frac{\Omega_{a}}{2\Delta}\mathrm{e}^{-i\Delta t}.
\end{equation}
Combine exponentials:
\begin{equation}
\dot c_{3}\simeq -i W\,\mathrm{e}^{-i\delta t},\qquad \boxed{\;W\equiv\frac{\Omega_{a}\Omega_{b}}{4\Delta}.\;}
\label{eq:c3dot-eff}
\end{equation}
Integrate with $c_{3}(0)=0$:
\begin{equation}
c_{3}(t)=-iW\,\frac{\mathrm{e}^{-i\delta t}-1}{-i\delta}.
\end{equation}
Simplify:
\begin{equation}
c_{3}(t)=\frac{W}{\delta}\bigl(\mathrm{e}^{-i\delta t}-1\bigr).
\end{equation}
Apply $\mathrm{e}^{-i\delta t}-1=-2i\mathrm{e}^{-i\delta t/2}\sin(\delta t/2)$:
\begin{equation}
c_{3}(t)=-\frac{2iW}{\delta}\,\mathrm{e}^{-i\delta t/2}\sin(\delta t/2).
\end{equation}
Take $|c_{3}|^{2}$:
\begin{equation}
|c_{3}(t)|^{2}=\frac{4W^{2}}{\delta^{2}}\sin^{2}(\delta t/2).
\end{equation}
Identify $\beta=\delta/2$:
\begin{equation}
\boxed{\;|c_{3}(t)|^{2}=\frac{W^{2}}{\beta^{2}}\sin^{2}(\beta t),\quad \beta=\frac{\delta}{2},\quad W=\frac{\Omega_{a}\Omega_{b}}{4\Delta}.\;}
\label{eq:c3sq}
\end{equation}
Same form as \eqref{eq:c2sq}: an effective two-level Rabi oscillation $1\leftrightarrow 3$ with coupling $W$ and detuning $\delta/2$. The $1/\Delta$ suppression of $W$ is the price of using a far-detuned intermediate level to suppress spontaneous loss.

\subsection*{(d) Why the simplified equations are a good approximation}

Expand each $\cos$ and identify the slow phase of every coupling:
\begin{itemize}
\item $\Omega_{a}\cos(\omega_{a}t)\,\mathrm{e}^{i\omega_{21}t}$: slow phase $\omega_{21}-\omega_{a}=-\Delta$. \textbf{Kept.}
\item $\Phi_{b}\cos(\omega_{b}t)\,\mathrm{e}^{i\omega_{21}t}$: slow phase $\omega_{21}-\omega_{b}\simeq \omega_{31}+\Delta$ (using $\omega_{b}=\omega_{a}-\omega_{31}-\delta$). \textbf{Drop} (fast).
\item $\Phi_{a}\cos(\omega_{a}t)\,\mathrm{e}^{-i\omega_{23}t}$: slow phase $-\omega_{23}+\omega_{a}=\Delta+\omega_{31}$. \textbf{Drop} (fast).
\item $\Omega_{b}\cos(\omega_{b}t)\,\mathrm{e}^{-i\omega_{23}t}$: slow phase $-\omega_{23}+\omega_{b}=\Delta-\delta$. \textbf{Kept.}
\end{itemize}
Dropped terms oscillate at frequencies $\sim\omega_{31}$, much faster than the dynamics they drive ($\sim\Omega$). Their cumulative contribution is suppressed by $\Omega/\omega_{31}\ll 1$. Keeping only the two resonant pairings (laser $a$ on $1\!\leftrightarrow\!2$, laser $b$ on $2\!\leftrightarrow\!3$) is the multi-frequency RWA.

\section*{Question 4 --- Rate equations, gain saturation, and Lamb dip in a He--Ne laser}

\textbf{Problem statement.} A homogeneously broadened transition between levels of degeneracies $g_{1},g_{2}$ has populations $N_{1},N_{2}$, with the upper level pumped at rate $R_{2}$ and decaying only by spontaneous emission $A_{21}$, while the lower level has lifetime $\tau_{1}$. (a) Steady-state rate equations and inversion. (b) Add a narrow-band radiation field; derive the gain-saturation form $\alpha=\alpha_{0}/(1+I/I_{\mathrm{sat}})$ and identify $\alpha_{0}$, $I_{\mathrm{sat}}$. (c) Sketch the gain seen by a weak probe in three different irradiation conditions. (d) HeNe Lamb dip: identify two broadening mechanisms, estimate physical parameters from the curves. (e) Explain the dip and its broadening with discharge current.

\subsection*{(a) Rate equations and steady-state inversion}

Write rate equations:
\begin{align}
\dot N_{2}&= R_{2}-A_{21}N_{2},\label{eq:N2dot-noI}\\
\dot N_{1}&= A_{21}N_{2}-N_{1}/\tau_{1}. \label{eq:N1dot-noI}
\end{align}
Set $\dot N_{2}=0$:
\begin{equation}
N_{2}=\frac{R_{2}}{A_{21}}.
\end{equation}
Set $\dot N_{1}=0$:
\begin{equation}
N_{1}=\tau_{1}A_{21}N_{2}=R_{2}\tau_{1}.
\label{eq:NSS-noI}
\end{equation}
Form the (degeneracy-weighted) inversion:
\begin{equation}
N^{*}\equiv N_{2}-\frac{g_{2}}{g_{1}}N_{1}=\frac{R_{2}}{A_{21}}-\frac{g_{2}}{g_{1}}R_{2}\tau_{1}.
\end{equation}
Factor $R_{2}$:
\begin{equation}
\boxed{\;N^{*}=R_{2}\!\left(\frac{1}{A_{21}}-\frac{g_{2}}{g_{1}}\tau_{1}\right).\;}
\label{eq:NS-noI}
\end{equation}
Positive inversion requires $1/A_{21}>(g_{2}/g_{1})\tau_{1}$ (the lower-level bottleneck condition).

\subsection*{(b) Saturated gain coefficient}

Let $W_{21}$ be the stimulated downward rate; the upward rate is $(g_{2}/g_{1})W_{21}$. Add stimulated terms to \eqref{eq:N2dot-noI}--\eqref{eq:N1dot-noI}:
\begin{align}
\dot N_{2}&=R_{2}-A_{21}N_{2}-W_{21}\Bigl(N_{2}-\tfrac{g_{2}}{g_{1}}N_{1}\Bigr),\label{eq:N2-I}\\
\dot N_{1}&=A_{21}N_{2}-N_{1}/\tau_{1}+W_{21}\Bigl(N_{2}-\tfrac{g_{2}}{g_{1}}N_{1}\Bigr).
\label{eq:N1-I}
\end{align}
Add \eqref{eq:N2-I} and \eqref{eq:N1-I} and set both LHS to zero:
\begin{equation}
0=R_{2}-N_{1}/\tau_{1}.
\end{equation}
Hence $N_{1}=R_{2}\tau_{1}$, unchanged. From \eqref{eq:N2-I} in steady state:
\begin{equation}
R_{2}=A_{21}N_{2}+W_{21}N^{*}.
\end{equation}
Solve for $N_{2}$:
\begin{equation}
N_{2}=\frac{R_{2}-W_{21}N^{*}}{A_{21}}.
\end{equation}
Substitute into $N^{*}=N_{2}-(g_{2}/g_{1})N_{1}$:
\begin{equation}
N^{*}=\frac{R_{2}-W_{21}N^{*}}{A_{21}}-\frac{g_{2}}{g_{1}}R_{2}\tau_{1}.
\end{equation}
Collect $N^{*}$ terms:
\begin{equation}
N^{*}\!\left(1+\frac{W_{21}}{A_{21}}\right)=R_{2}\!\left(\frac{1}{A_{21}}-\frac{g_{2}}{g_{1}}\tau_{1}\right)=N_{0}^{*}.
\end{equation}
Solve for the saturated inversion:
\begin{equation}
N^{*}=\frac{N_{0}^{*}}{1+W_{21}/A_{21}}.
\label{eq:NS-sat}
\end{equation}
Use the cross-section to relate $W_{21}$ and $I$ at the laser frequency:
\begin{equation}
W_{21}=\frac{\sigma(\omega)\,I}{\hbar\omega},\qquad \sigma(\omega)=\frac{\pi^{2}c^{2}}{\omega^{2}}A_{21}\,g_{H}(\omega).
\label{eq:W21-cross}
\end{equation}
Multiply $N^{*}$ by $\sigma(\omega)$ to get the gain coefficient $\alpha=\sigma(\omega)N^{*}$:
\begin{equation}
\alpha(\omega)=\frac{\sigma(\omega)N_{0}^{*}}{1+W_{21}/A_{21}}.
\end{equation}
At line centre ($\omega=\omega_{21}$):
\begin{equation}
\boxed{\;\alpha(\omega_{21})=\frac{\alpha_{0}}{1+I/I_{\mathrm{sat}}},\;}
\label{eq:gain-sat}
\end{equation}
with
\begin{equation}
\alpha_{0}=\sigma(\omega_{21})\,N_{0}^{*}=\frac{\pi^{2}c^{2}}{\omega_{21}^{2}}A_{21}\,g_{H}(\omega_{21})\,R_{2}\!\left(\frac{1}{A_{21}}-\frac{g_{2}}{g_{1}}\tau_{1}\right),
\label{eq:alpha0}
\end{equation}
and, equating $W_{21}/A_{21}=I/I_{\mathrm{sat}}$,
\begin{equation}
I_{\mathrm{sat}}=\frac{\hbar\omega_{21}A_{21}}{\sigma(\omega_{21})}.
\end{equation}
Substitute $\sigma(\omega_{21})$:
\begin{equation}
\boxed{\;I_{\mathrm{sat}}=\frac{\hbar\omega_{21}^{3}}{\pi^{2}c^{2}\,g_{H}(\omega_{21})}.\;}
\label{eq:Isat}
\end{equation}
At $I=I_{\mathrm{sat}}$ the inversion drops to half its small-signal value.

\subsection*{(c) Sketches of gain vs.\ frequency}

\begin{center}
\begin{tikzpicture}[xscale=1.2,yscale=1.6]
\draw[->] (-0.2,0)--(6.5,0) node[right]{$\omega$};
\draw[->] (0,-0.1)--(0,2.6) node[above]{$\alpha(\omega)$};
\draw[domain=0.4:6.0,smooth,thick,blue] plot (\x,{2.0/(1+(\x-3)^2*4)}) node[right,blue]{(i) weak};
\draw[domain=0.4:6.0,smooth,thick,red] plot (\x,{2.0/(1+(\x-3)^2*4)/(1+1.0/(1+(\x-3)^2*4))}) node[right,red]{(ii)/(iii)};
\filldraw (3,0) circle (0.5pt) node[below]{$\omega_{21}$};
\end{tikzpicture}
\end{center}

(i) Weak probe sees the unsaturated Lorentzian, peak $\alpha_{0}$, FWHM $\Delta\omega_{H}$.

(ii) Single saturating beam at $I=I_{\mathrm{sat}}$. Use \eqref{eq:gain-sat} with local Lorentzian $L(\omega)$:
\begin{equation}
\alpha(\omega)=\frac{\alpha_{0}L(\omega)}{1+IL(\omega)/I_{\mathrm{sat}}},\qquad L(\omega)=\frac{1}{1+4(\omega-\omega_{21})^{2}/\Delta\omega_{H}^{2}}.
\end{equation}
Evaluate at line centre ($L=1$, $I=I_{\mathrm{sat}}$):
\begin{equation}
\alpha(\omega_{21})=\alpha_{0}/2.
\end{equation}
Solve $\alpha(\omega)=\alpha_{0}/4$ for FWHM: $L=1/3\Rightarrow$ FWHM $=\sqrt{2}\,\Delta\omega_{H}$ (power broadening).

(iii) Homogeneous medium: every atom contributes at every frequency, so the strong pump at $\omega_{21}$ reduces $N^{*}$ uniformly. The weak probe sees the same curve as (ii) --- no hole is burned. A spectral hole would arise only in an inhomogeneously broadened medium.

\subsection*{(d) Two broadening mechanisms in the He--Ne laser at 633\,nm}

\paragraph{Doppler (inhomogeneous).} Each atom emits at its Doppler-shifted frequency; the resulting Gaussian envelope has FWHM
\begin{equation}
\Delta\nu_{D} = \frac{2}{\lambda}\sqrt{\frac{2k_{B}T\ln 2}{m}}.
\label{eq:Doppler}
\end{equation}
Take the observed gain envelope width as $\Delta\nu_{D}\simeq \SI{1.5}{GHz}$ (the dip-bearing region of the curve is roughly half the full envelope). Solve \eqref{eq:Doppler} for $T$:
\begin{equation}
T=\frac{mc^{2}}{8k_{B}\ln 2}\!\left(\frac{\Delta\nu_{D}}{\nu}\right)^{2}.
\end{equation}
Insert $m=\SI{20}{u}=\SI{3.32e-26}{kg}$, $\nu=c/\lambda=\SI{4.74e14}{Hz}$, $\Delta\nu_{D}=\SI{1.5}{GHz}$:
\begin{equation}
T\simeq \SI{400}{K},
\end{equation}
a reasonable discharge-tube temperature.

\paragraph{Natural + collisional (homogeneous).} The Lorentzian FWHM is
\begin{equation}
\boxed{\;\Delta\nu_{H}=\frac{1}{2\pi\tau_{2}}+\frac{1}{2\pi\tau_{1}}+\frac{n\sigma_{\mathrm{coll}}\bar v}{2\pi}.\;}
\label{eq:Lorentz-FWHM}
\end{equation}
Use $\tau_{2}\sim \SI{30}{ns}$, $\tau_{1}\sim \SI{10}{ns}$:
\begin{equation}
\Delta\nu_{N}=\frac{1}{2\pi\tau_{2}}+\frac{1}{2\pi\tau_{1}}\sim \SI{20}{MHz}.
\end{equation}
At pressure $\sim 1\,\mathrm{mbar}$ ($n\sim \SI{2.5e22}{m^{-3}}$), with $\sigma_{\mathrm{coll}}\sim \SI{e-19}{m^{2}}$, $\bar v\sim \SI{700}{m.s^{-1}}$:
\begin{equation}
\Delta\nu_{\mathrm{coll}}=\frac{n\sigma_{\mathrm{coll}}\bar v}{2\pi}\sim \SI{10}{MHz}.
\end{equation}
Total $\Delta\nu_{H}\sim \SIrange{20}{30}{MHz}$, consistent with the 30~MHz Lamb-dip width of curve~$A$.

\subsection*{(e) Why a Lamb dip appears, and why it broadens with $R_{2}$}

\paragraph{Mechanism.} A cavity standing wave at frequency $\omega$ is the sum of two counter-propagating running waves. Each saturates atoms whose longitudinal velocity satisfies $\omega(1\mp v_{z}/c)=\omega_{21}$:
\begin{equation}
v_{z}=\pm c(\omega-\omega_{21})/\omega_{21}.
\end{equation}
For $\omega\neq\omega_{21}$ the two classes are different atoms (no mutual saturation). For $\omega=\omega_{21}$ both reduce to $v_{z}=0$ (same atoms), doubling the saturation rate and depressing the gain --- the Lamb dip. Its width is the homogeneous width $\Delta\nu_{H}$.

\paragraph{Broadening with $R_{2}$.} Higher discharge current $\to$ larger $R_{2}\to$ larger intracavity intensity $I$. The hole at $v_{z}=0$ power-broadens via the $\sqrt{1+I/I_{\mathrm{sat}}}$ factor of (c):
\begin{equation}
\boxed{\;\Delta\nu_{\mathrm{dip}}=\Delta\nu_{H}\sqrt{1+I/I_{\mathrm{sat}}}.\;}
\label{eq:dip-broadening}
\end{equation}
Apply to curves $A\to C$ ($30\to 40$ MHz):
\begin{equation}
\frac{I}{I_{\mathrm{sat}}}=\left(\frac{40}{30}\right)^{2}-1\simeq 0.78.
\end{equation}
Power broadening is the dominant mechanism; collisional broadening at higher current makes a smaller contribution.
