\section*{Question 1 --- Neutron diffraction from \texorpdfstring{NaTaO$_3$}{NaTaO3}}

\subsection*{(a) Systematic absences from Na, Ta, O$_1$}

Define the species structure factor:
\begin{equation}
F_X(hkl)=b_X\sum_{j=1}^{N_X}\exp\!\bigl[2\pi i(hx_j+ky_j+lz_j)\bigr].
\label{eq:Fdef}
\end{equation}
Na at $(0,\tfrac12,\tfrac12)$ and $(\tfrac12,0,\tfrac12)$:
\begin{equation}
F_{\mathrm{Na}}=b_{\mathrm{Na}}\bigl[e^{2\pi i(k/2+l/2)}+e^{2\pi i(h/2+l/2)}\bigr].
\end{equation}
Pull out $e^{i\pi l}$:
\begin{equation}
F_{\mathrm{Na}}=b_{\mathrm{Na}}\,e^{i\pi l}\bigl[e^{i\pi k}+e^{i\pi h}\bigr].
\label{eq:Fna}
\end{equation}
Ta at $(0,0,0)$ and $(\tfrac12,\tfrac12,0)$:
\begin{equation}
F_{\mathrm{Ta}}=b_{\mathrm{Ta}}\bigl[1+e^{i\pi(h+k)}\bigr].
\label{eq:Fta}
\end{equation}
O$_1$ at $(0,0,\tfrac12)$ and $(\tfrac12,\tfrac12,\tfrac12)$:
\begin{equation}
F_{\mathrm{O1}}=b_{\mathrm{O}}\,e^{i\pi l}\bigl[1+e^{i\pi(h+k)}\bigr].
\label{eq:Fo1}
\end{equation}
Take $h+k=2n+1$ (odd): $e^{i\pi(h+k)}=-1$:
\begin{equation}
F_{\mathrm{Ta}}=0,\qquad F_{\mathrm{O1}}=0.
\end{equation}
Same parity argument: $h,k$ have opposite parity, so $e^{i\pi h}=-e^{i\pi k}$:
\begin{equation}
F_{\mathrm{Na}}=0.
\end{equation}
Hence none of Na, Ta, O$_1$ contributes when $h+k$ is odd. Take $h+k=2n$: $e^{i\pi h}=e^{i\pi k}=(-1)^h$ and $e^{i\pi(h+k)}=+1$:
\begin{equation}
F_{\mathrm{Na}}=2b_{\mathrm{Na}}(-1)^h e^{i\pi l}=2b_{\mathrm{Na}}(-1)^{h+l},\qquad F_{\mathrm{Ta}}=2b_{\mathrm{Ta}},\qquad F_{\mathrm{O1}}=2b_{\mathrm{O}}(-1)^l.
\end{equation}
Sum the three contributions:
\begin{equation}
\boxed{\,F_{\mathrm{Na+Ta+O1}}(hkl)=2\bigl[b_{\mathrm{Ta}}+b_{\mathrm{O}}(-1)^{l}+b_{\mathrm{Na}}(-1)^{h+l}\bigr]\quad(h+k=2n).\,}
\label{eq:Fmain}
\end{equation}

\subsection*{(b) Structure factor of O$_2$ to first order in $\delta$}

All four O$_2$ sites have $z=0$, so $e^{2\pi i l z}=1$ for every term. Sites:
\begin{equation}
(\tfrac14+\delta,-\tfrac14+\delta,0),\;(-\tfrac14-\delta,\tfrac14-\delta,0),\;(\tfrac14-\delta,\tfrac14+\delta,0),\;(-\tfrac14+\delta,-\tfrac14-\delta,0).
\end{equation}
Group sites 1,2 (conjugates) and sites 3,4 (conjugates):
\begin{equation}
F_{\mathrm{O2}}=2b_{\mathrm{O}}\Bigl\{\cos\!\bigl[\tfrac{\pi(h-k)}{2}+2\pi\delta(h+k)\bigr]+\cos\!\bigl[\tfrac{\pi(h+k)}{2}+2\pi\delta(k-h)\bigr]\Bigr\}.
\label{eq:Fo2-exact}
\end{equation}
Linearise with $\cos(\theta+\epsilon)\approx\cos\theta-\epsilon\sin\theta$:
\begin{equation}
\frac{F_{\mathrm{O2}}}{2b_{\mathrm{O}}}\approx\cos\!\tfrac{\pi(h-k)}{2}+\cos\!\tfrac{\pi(h+k)}{2}-2\pi\delta\bigl[(h+k)\sin\!\tfrac{\pi(h-k)}{2}+(k-h)\sin\!\tfrac{\pi(h+k)}{2}\bigr].
\label{eq:Fo2-lin}
\end{equation}

\paragraph{(i) $h,k$ both odd.} Then $(h\pm k)/2$ are integers, so both sines vanish. The cosines are $(-1)^{(h-k)/2}$ and $(-1)^{(h+k)/2}$; since $(h+k)/2 - (h-k)/2 = k$ (odd), they have opposite signs:
\begin{equation}
\boxed{\,F_{\mathrm{O2}}=0\quad(h,k\text{ both odd}).\,}
\label{eq:Fo2-oddodd}
\end{equation}

\paragraph{(ii) $h,k$ both even.} Sines vanish; $(h-k)/2$ and $(h+k)/2$ differ by $k$ (even), so the two cosines are equal:
\begin{equation}
\boxed{\,F_{\mathrm{O2}}=4b_{\mathrm{O}}\,(-1)^{(h+k)/2}\quad(h,k\text{ both even}).\,}
\label{eq:Fo2-eveneven}
\end{equation}

\paragraph{(iii) One of $h,k$ even, the other odd.} Then $(h\pm k)/2$ are half-integers, so both cosines vanish; only the $\delta$-linear term survives. Write $s_1=\sin\tfrac{\pi(h-k)}{2}$ and $s_2=\sin\tfrac{\pi(h+k)}{2}$. If $k$ is even ($h$ odd): $s_2=s_1$, so:
\begin{equation}
F_{\mathrm{O2}}=-4\pi\delta b_{\mathrm{O}}\bigl[(h+k)+(k-h)\bigr]s_1=-8\pi\delta b_{\mathrm{O}}\,k\,s_1.
\label{eq:Fo2-keven}
\end{equation}
If $k$ is odd ($h$ even): $s_2=-s_1$, so:
\begin{equation}
F_{\mathrm{O2}}=-4\pi\delta b_{\mathrm{O}}\bigl[(h+k)-(k-h)\bigr]s_1=-8\pi\delta b_{\mathrm{O}}\,h\,s_1.
\label{eq:Fo2-kodd}
\end{equation}

\subsection*{(c) Powder diffraction at $\lambda=\SI{0.159}{nm}$}

Apply $d^{-2}=(h^2+k^2)/a^2+l^2/c^2$:
\begin{align}
d_{001}&=c=\SI{0.393}{nm},\\
d_{110}&=a/\sqrt{2}=\SI{0.555}{nm}/\sqrt2=\SI{0.39245}{nm},\\
d_{210}&=a/\sqrt{5}=\SI{0.555}{nm}/\sqrt5=\SI{0.24820}{nm}.
\end{align}
Apply Bragg $\sin\theta=\lambda/(2d)$:
\begin{align}
2\theta_{001}&=2\arcsin(0.159/0.786)=2\arcsin(0.20229)=\SI{23.34}{\degree},\\
2\theta_{110}&=2\arcsin(0.159/0.7849)=2\arcsin(0.20257)=\SI{23.38}{\degree},\\
2\theta_{210}&=2\arcsin(0.159/0.4964)=2\arcsin(0.32034)=\SI{37.36}{\degree}.
\end{align}
Compare splitting to resolution:
\begin{equation}
|2\theta_{110}-2\theta_{001}|=\SI{0.04}{\degree}<\SI{0.05}{\degree},
\end{equation}
so $(001)$ and $(110)$ merge into one peak.

\paragraph{$(001)$ structure factor.} Here $h+k=0$ (even), $h,k$ both even. Use \eqref{eq:Fmain} with $h=k=0,\,l=1$:
\begin{equation}
F_{\mathrm{Na+Ta+O1}}=2\bigl[b_{\mathrm{Ta}}-b_{\mathrm{O}}-b_{\mathrm{Na}}\bigr].
\end{equation}
Plug in numbers:
\begin{equation}
F_{\mathrm{Na+Ta+O1}}=2(6.91-5.803-3.63)\,\si{fm}=-5.046\,\si{fm}.
\end{equation}
Use \eqref{eq:Fo2-eveneven} with $(h+k)/2=0$:
\begin{equation}
F_{\mathrm{O2}}=4b_{\mathrm{O}}=4\cdot 5.803\,\si{fm}=23.212\,\si{fm}.
\end{equation}
Sum:
\begin{equation}
F(001)=-5.046+23.212=18.166\,\si{fm},\qquad |F(001)|^{2}\approx 330\,\si{fm^2}.
\end{equation}

\paragraph{$(110)$ structure factor.} Here $h+k=2$ (even), $h,k$ both odd. Use \eqref{eq:Fmain} with $h=k=1,\,l=0$:
\begin{equation}
F_{\mathrm{Na+Ta+O1}}=2\bigl[b_{\mathrm{Ta}}+b_{\mathrm{O}}-b_{\mathrm{Na}}\bigr].
\end{equation}
Plug in numbers:
\begin{equation}
F_{\mathrm{Na+Ta+O1}}=2(6.91+5.803-3.63)\,\si{fm}=18.166\,\si{fm}.
\end{equation}
Apply \eqref{eq:Fo2-oddodd}:
\begin{equation}
F_{\mathrm{O2}}=0,\qquad |F(110)|^{2}=(18.166)^{2}\,\si{fm^2}\approx 330\,\si{fm^2}.
\end{equation}

\paragraph{$(210)$ structure factor.} Here $h+k=3$ (odd), so $F_{\mathrm{Na+Ta+O1}}=0$. With $h=2,k=1$, use \eqref{eq:Fo2-keven} with $s_1=\sin\tfrac{\pi(1)}{2}=+1$:
\begin{equation}
F(210)=-8\pi\delta b_{\mathrm{O}}\cdot k\cdot s_1=-8\pi\delta b_{\mathrm{O}}.
\end{equation}
Square it:
\begin{equation}
|F(210)|^{2}=64\pi^{2}\delta^{2}b_{\mathrm{O}}^{2}=64\pi^{2}\delta^{2}\cdot 33.675\,\si{fm^2}\approx 2.128\times 10^{4}\,\delta^{2}\,\si{fm^2}.
\end{equation}

\paragraph{Tetragonal multiplicities ($4/mmm$).}
\begin{itemize}
\item $(00l)$: only $\pm l$; $m_{001}=2$.
\item $(hh0)$, $h\ne0$: $(\pm h,\pm h,0)$; $m_{110}=4$.
\item $(hk0)$, $h\ne k\ne0$: $(\pm h,\pm k,0)$ and $(\pm k,\pm h,0)$; $m_{210}=8$.
\end{itemize}

\subsection*{(d) Determination of $\delta$}

Write powder intensity as $I_{hkl}=K\,m_{hkl}\,|F_{hkl}|^{2}$. Use $|F(001)|^{2}=|F(110)|^{2}=330\,\si{fm^2}$:
\begin{equation}
I_{001}+I_{110}=K(m_{001}+m_{110})\cdot 330\,\si{fm^2}=K\cdot 6\cdot 330\,\si{fm^2}=1980\,K\,\si{fm^2}.
\label{eq:I-sum}
\end{equation}
Set equal to the measured $1843$:
\begin{equation}
K=\frac{1843}{1980}\,\si{fm^{-2}}\approx 0.9308\,\si{fm^{-2}}.
\label{eq:K-value}
\end{equation}
Use $I_{210}=K\,m_{210}\,|F(210)|^{2}$ with $m_{210}=8$:
\begin{equation}
I_{210}=K\cdot 8\cdot 64\pi^{2}\delta^{2}b_{\mathrm{O}}^{2}=512\pi^{2}\,K\,b_{\mathrm{O}}^{2}\,\delta^{2}.
\label{eq:I210}
\end{equation}
Solve for $\delta^{2}$ with $b_{\mathrm{O}}^{2}=33.675\,\si{fm^2}$, $K=0.9308$, $I_{210}=345$:
\begin{equation}
\delta^{2}=\frac{345}{512\pi^{2}\cdot 0.9308\cdot 33.675}.
\end{equation}
Evaluate denominator:
\begin{equation}
512\pi^{2}\cdot 0.9308\cdot 33.675=1.583\times 10^{5}.
\end{equation}
Hence:
\begin{equation}
\delta^{2}=\frac{345}{1.583\times 10^{5}}=2.18\times 10^{-3}.
\end{equation}
Take the square root:
\begin{equation}
\boxed{\,\delta\approx 0.047.\,}
\label{eq:delta-value}
\end{equation}

\section*{Question 2 --- Molecular orbital theory of H$_2$}

\subsection*{(a) Secular equation retaining overlap}

Write the variational quotient:
\begin{equation}
\lambda=\frac{N}{D},\qquad N=\sum_{ij}c_i^{*}c_j H_{ij},\quad D=\sum_{ij}c_i^{*}c_j S_{ij},
\end{equation}
with $H_{ij}=\bra i\hat H\ket j$ and $S_{ij}=\braket{i|j}$. Apply $(\hat K+\hat V_1)\ket1=\epsilon_0\ket1$:
\begin{equation}
\hat H\ket 1=(\hat K+\hat V_1)\ket 1+\hat V_2\ket1=\epsilon_0\ket1+\hat V_2\ket1.
\end{equation}
Take $\bra 1$:
\begin{equation}
H_{11}=\epsilon_0+\bra1\hat V_2\ket1=\epsilon_0+V_{c2}.
\end{equation}
By analogy ($1\leftrightarrow 2$):
\begin{equation}
H_{22}=\epsilon_0+V_{c1}.
\end{equation}
Apply $(\hat K+\hat V_2)\ket2=\epsilon_0\ket2$:
\begin{equation}
\hat H\ket 2=\epsilon_0\ket 2+\hat V_1\ket 2.
\end{equation}
Take $\bra 1$:
\begin{equation}
H_{12}=\epsilon_0\braket{1|2}+\bra1\hat V_1\ket2=\epsilon_0\sigma-t,
\end{equation}
using $t=-\bra1\hat V_2\ket2$ and the convention $\bra1\hat V_1\ket2=-t$ that follows from the symmetric problem (see (b)). By Hermiticity:
\begin{equation}
H_{21}=\epsilon_0\sigma^{*}-t^{*}.
\end{equation}
Assemble:
\begin{equation}
\mathsf H=\begin{pmatrix}\epsilon_0+V_{c2}&\epsilon_0\sigma-t\\\epsilon_0\sigma^{*}-t^{*}&\epsilon_0+V_{c1}\end{pmatrix},\qquad
\mathsf S=\begin{pmatrix}1&\sigma\\\sigma^{*}&1\end{pmatrix}.
\label{eq:HS-mat}
\end{equation}
Extremise $\lambda=N/D$ via $\partial\lambda/\partial c_i^{*}=0$:
\begin{equation}
\pdone{N}{c_i^{*}}=\lambda\pdone{D}{c_i^{*}}\quad\Longleftrightarrow\quad\sum_j H_{ij}c_j=\lambda\sum_j S_{ij}c_j.
\end{equation}
Cast into matrix form:
\begin{equation}
\boxed{\,(\mathsf H-\lambda\mathsf S)\begin{pmatrix}c_1\\ c_2\end{pmatrix}=0.\,}
\label{eq:secular-mat}
\end{equation}

\subsection*{(b) Symmetric two-atom case}

Apply $x\to-x$ which swaps $1\leftrightarrow2$:
\begin{equation}
V_{c1}=\bra2\hat V_1\ket2\;\xrightarrow{1\leftrightarrow 2}\;\bra1\hat V_2\ket1=V_{c2},
\end{equation}
hence $V_{c1}=V_{c2}\equiv V_c$. Choose real 1s orbitals; the operators $\hat V_i$ are real, so $\sigma,t\in\mathbb R$. Insert into \eqref{eq:HS-mat}:
\begin{equation}
\begin{pmatrix}\epsilon_0+V_c-\lambda&(\epsilon_0-\lambda)\sigma-t\\(\epsilon_0-\lambda)\sigma-t&\epsilon_0+V_c-\lambda\end{pmatrix}\begin{pmatrix}c_1\\c_2\end{pmatrix}=0.
\label{eq:secular-sym}
\end{equation}
Diagonal entries equal and off-diagonal entries equal, so eigenvectors are $c_1=\pm c_2$. Substitute $c_1=+c_2$:
\begin{equation}
(\epsilon_0+V_c-\lambda)+(\epsilon_0-\lambda)\sigma-t=0.
\end{equation}
Collect $\lambda$:
\begin{equation}
\lambda(1+\sigma)=\epsilon_0(1+\sigma)+V_c-t.
\end{equation}
Solve for $\lambda$:
\begin{equation}
\boxed{\,\lambda_+=\epsilon_0+\frac{V_c-t}{1+\sigma}\;\;\text{(bonding)}.\,}
\label{eq:lambda+}
\end{equation}
Substitute $c_1=-c_2$:
\begin{equation}
(\epsilon_0+V_c-\lambda)-[(\epsilon_0-\lambda)\sigma-t]=0.
\end{equation}
Collect $\lambda$:
\begin{equation}
\lambda(1-\sigma)=\epsilon_0(1-\sigma)+V_c+t.
\end{equation}
Solve for $\lambda$:
\begin{equation}
\boxed{\,\lambda_-=\epsilon_0+\frac{V_c+t}{1-\sigma}\;\;\text{(antibonding)}.\,}
\label{eq:lambda-}
\end{equation}
Normalise: $\braket{\psi_\pm|\psi_\pm}=2(1\pm\sigma)$ (using $\braket{1|2}=\sigma$):
\begin{equation}
\ket{\psi_+}=\frac{\ket1+\ket2}{\sqrt{2(1+\sigma)}},\qquad\ket{\psi_-}=\frac{\ket1-\ket2}{\sqrt{2(1-\sigma)}}.
\label{eq:psipm}
\end{equation}

\subsection*{(c) Hydrogen molecule with explicit integrals}

\paragraph{(i) Large-$d$ asymptotics.} Write $t,\sigma$ as bilinear overlaps of $\phi(r)=\pi^{-1/2}e^{-r}$: the product $\phi(\vb r)\phi(\vb r-\vb d)\sim e^{-r}e^{-|\vb r-\vb d|}$ is exponentially suppressed for any fixed point unless $\vb r$ sits in a narrow region between the two centres, so:
\begin{equation}
t,\sigma\propto e^{-d}\quad(d\gg a_0).
\end{equation}
By contrast $V_c=\bra1\hat V_2\ket1$ is the electrostatic energy of $|\phi_1|^2$ (total charge $1$, centred at site 1) in the Coulomb potential of the distant proton at site 2. For $d\gg a_0$ the charge $|\phi_1|^2$ looks like a point charge:
\begin{equation}
V_c\to -2/d\quad(d\gg a_0).
\end{equation}

\paragraph{(ii) Ground-state energy at $d_{\mathrm{exp}}=1.4$.} Compute $t$:
\begin{equation}
t=2e^{-1.4}(1+1.4)=2\cdot 0.2466\cdot 2.4=1.184\,E_R.
\end{equation}
Compute $\sigma$:
\begin{equation}
\sigma=e^{-1.4}(1+1.4+1.96/3)=0.2466\cdot 3.0533=0.7530.
\end{equation}
Compute $V_c$:
\begin{equation}
V_c=\frac{2[e^{-2.8}(1+1.4)-1]}{1.4}=\frac{2(0.06081\cdot 2.4-1)}{1.4}=\frac{2(-0.8541)}{1.4}=-1.220\,E_R.
\end{equation}
Substitute into \eqref{eq:lambda+} with $\epsilon_0=-1$:
\begin{equation}
\lambda_+^{(\sigma)}=-1+\frac{-1.220-1.184}{1+0.7530}=-1+\frac{-2.404}{1.753}=-1-1.371=-2.371\,E_R.
\label{eq:lp-sigma}
\end{equation}
Set $\sigma\to 0$:
\begin{equation}
\lambda_+^{(\sigma=0)}=-1+(-1.220-1.184)=-3.404\,E_R.
\label{eq:lp-nosigma}
\end{equation}
Place two electrons in $\ket{\psi_+}$ with opposite spin and add proton repulsion $2/d$ (in $E_R$):
\begin{equation}
E_{\mathrm{tot}}^{(\sigma)}=2\lambda_+^{(\sigma)}+2/d=2(-2.371)+2/1.4=-4.742+1.4286=-3.314\,E_R,
\end{equation}
\begin{equation}
E_{\mathrm{tot}}^{(\sigma=0)}=2(-3.404)+1.4286=-5.379\,E_R.
\end{equation}
Subtract the dissociation limit $E_{2\mathrm H}=2\epsilon_0=-2\,E_R$:
\begin{equation}
E_d^{(\sigma)}=-2-(-3.314)=1.31\,E_R,\qquad E_d^{(\sigma=0)}=-2-(-5.379)=3.38\,E_R.
\end{equation}
The two values differ by a factor of $\sim 2.6$, so $\sigma$ cannot be neglected.

\paragraph{(iii) Comparison with experiment.} The MO prediction $E_d^{(\sigma)}=1.31\,E_R$ greatly exceeds $E_d^{\mathrm{exp}}\approx 0.33\,E_R$: the MO calculation overbinds by $\sim 1\,E_R\approx \SI{13.6}{eV}$.

\paragraph{(iv) Behaviour at $d=0$.} Take $d\to 0$ in $t$:
\begin{equation}
t(0)=2\cdot 1\cdot(1+0)=2\,E_R.
\end{equation}
Take $d\to 0$ in $\sigma$:
\begin{equation}
\sigma(0)=1\cdot(1+0+0)=1.
\end{equation}
Take $d\to 0$ in $V_c$: expand $e^{-2d}(1+d)=(1-2d+2d^2)(1+d)=1-d+O(d^3)$:
\begin{equation}
V_c(0)=\lim_{d\to0}\frac{2(-d+O(d^3))}{d}=-2\,E_R.
\end{equation}
Insert into \eqref{eq:lambda+}:
\begin{equation}
\lambda_+(0)=-1+\frac{-2-2}{1+1}=-1-2=-3\,E_R.
\label{eq:lp-zero}
\end{equation}
Hence $\lambda_+(0)=-3\,E_R<\lambda_+(d_{\mathrm{exp}})=-2.371\,E_R$: $\lambda_+$ keeps decreasing as $d\to 0$, with absolute minimum at $d=0$.

\subsection*{(d) Missing physics}

The MO calculation overestimates the dissociation energy by a factor $\sim 4$ at $d_{\mathrm{exp}}$ and predicts an unphysical minimum of $\lambda_+$ at $d=0$. Both flaws stem from omitted \emph{electron--electron repulsion}. The trial state places two electrons in the same spatial orbital $\ket{\psi_+}$; squeezing the molecule to small $d$ forces both electrons into the same small region and the Coulomb energy $\sim 1/r_{12}$ grows. Adding the Hartree term $J=\langle\psi_+\psi_+|1/r_{12}|\psi_+\psi_+\rangle>0$ raises the energy strongly at small $d$ (removing the spurious $d=0$ minimum in the total energy) and weakens the binding at $d_{\mathrm{exp}}$. A related deficiency is the absence of \emph{electron correlation}: the closed-shell MO state retains equal amplitudes for ionic ($\mathrm H^-\mathrm H^+$) and covalent configurations as $d\to\infty$, so dissociation is described poorly. Heitler--London or CI with the antibonding orbital cures this and pushes the binding curve toward the experimental $0.33\,E_R$.

\section*{Question 3 --- Massless Dirac fermions in 2D}

\subsection*{(a) Energy surfaces and comparison with $E_g\to 0$ semiconductor}

\begin{center}
\begin{tikzpicture}[scale=1]

\draw[->] (0,-2.4) -- (0,2.4) node[above]{$E$};
\draw[thick,blue] (-2.2,2.2) -- (0,0) -- (2.2,2.2) node[above right]{conduction $E_{0}+c\hbar k$};
\draw[thick,red]  (-2.2,-2.2) -- (0,0) -- (2.2,-2.2) node[below right]{valence $E_{0}-c\hbar k$};
\draw[dashed] (-2.6,0) -- (2.6,0);
\node at (-0.5,0.3){$E_{0}$};
\end{tikzpicture}
\end{center}

\paragraph{(ii) Comparison with a parabolic $E_g\to 0$ semiconductor.}
\begin{itemize}
\item Dispersion: linear (here) vs.\ quadratic. Group velocity here is energy-independent: $|\vb v_g|=c$.
\item Density of states: $g(E)\propto|E-E_0|$ (vanishes at Dirac point) vs.\ constant in 2D parabolic.
\item Effective mass: undefined for linear bands; finite for parabolic.
\item Berry phase: $\pi$ around the Dirac point (suppresses backscattering); absent in parabolic case.
\end{itemize}

\subsection*{(b) Conduction-band electron density}

Count states (factor 2 for spin) in a $k$-shell of area $2\pi k\,dk$:
\begin{equation}
g(E)\,dE=2\cdot\frac{2\pi k\,dk}{(2\pi)^{2}}=\frac{k}{\pi}\,dk.
\end{equation}
Substitute $k=(E-E_0)/(c\hbar)$, $dk=dE/(c\hbar)$:
\begin{equation}
g(E)=\frac{E-E_0}{\pi c^{2}\hbar^{2}}.
\label{eq:dos}
\end{equation}
Apply the Fermi function:
\begin{equation}
n_{\mathrm{el}}=\int_{E_{0}}^{\infty}\frac{(E-E_0)\,dE}{\pi c^{2}\hbar^{2}\bigl[e^{\beta(E-\mu)}+1\bigr]}.
\end{equation}
Change variables $u=\beta(E-E_0)$, $du=\beta\,dE$:
\begin{equation}
n_{\mathrm{el}}=\frac{1}{\pi c^{2}\hbar^{2}\beta^{2}}\int_{0}^{\infty}\frac{u\,du}{e^{u-\alpha}+1}.
\end{equation}
Recognise the Fermi integral with $j=1$:
\begin{equation}
\int_0^\infty\frac{u\,du}{e^{u-\alpha}+1}=1!\,F_1(\alpha)=F_1(\alpha).
\end{equation}
Hence:
\begin{equation}
\boxed{\,n_{\mathrm{el}}=\frac{1}{\pi c^{2}\hbar^{2}}\,\beta^{-2}\,F_{1}(\alpha),\qquad C_{1}=\frac{1}{\pi c^{2}\hbar^{2}},\quad p=-2.\,}
\label{eq:nel-final}
\end{equation}

\subsection*{(c) Conduction-band specific heat}

Energy density above $E_0$:
\begin{equation}
\tilde u_{\mathrm{el}}=\int_{E_{0}}^{\infty}(E-E_{0})\,g(E)\,f(E)\,dE.
\label{eq:utilde}
\end{equation}
Substitute $u=\beta(E-E_0)$ again:
\begin{equation}
\tilde u_{\mathrm{el}}=\frac{1}{\pi c^{2}\hbar^{2}\beta^{3}}\int_{0}^{\infty}\frac{u^{2}\,du}{e^{u-\alpha}+1}.
\end{equation}
Use $\int_0^\infty u^2\,du/(e^{u-\alpha}+1)=2!\,F_2(\alpha)=2F_2(\alpha)$:
\begin{equation}
\tilde u_{\mathrm{el}}=\frac{2}{\pi c^{2}\hbar^{2}\beta^{3}}F_{2}(\alpha)=\frac{2(k_{\mathrm B}T)^{3}}{\pi c^{2}\hbar^{2}}F_{2}(\alpha).
\label{eq:utilde2}
\end{equation}
At fixed $\mu$ (so fixed $\tilde\mu=\mu-E_0$):
\begin{equation}
\frac{d\alpha}{dT}\bigg|_{\mu}=\frac{d}{dT}\!\left(\frac{\tilde\mu}{k_{\mathrm B}T}\right)=-\frac{\tilde\mu}{k_{\mathrm B}T^{2}}=-\frac{\alpha}{T}.
\label{eq:dalpha-dT}
\end{equation}
Differentiate \eqref{eq:utilde2} using $dF_2/d\alpha=F_1$:
\begin{equation}
c_{v}^{\mathrm{el}}=\frac{d\tilde u_{\mathrm{el}}}{dT}\bigg|_{\mu}=\frac{2}{\pi c^{2}\hbar^{2}}\Bigl[3k_{\mathrm B}^{3}T^{2}F_{2}(\alpha)+(k_{\mathrm B}T)^{3}F_{1}(\alpha)\frac{d\alpha}{dT}\Bigr].
\end{equation}
Insert \eqref{eq:dalpha-dT}:
\begin{equation}
c_{v}^{\mathrm{el}}=\frac{2}{\pi c^{2}\hbar^{2}}\bigl[3k_{\mathrm B}^{3}T^{2}F_{2}(\alpha)-k_{\mathrm B}^{3}T^{2}\alpha F_{1}(\alpha)\bigr].
\end{equation}
Factor $k_B^3 T^2$:
\begin{equation}
\boxed{\,c_{v}^{\mathrm{el}}=\frac{2}{\pi c^{2}\hbar^{2}}\,k_{\mathrm B}^{3}T^{2}\bigl[3F_{2}(\alpha)-\alpha F_{1}(\alpha)\bigr],\quad C_{2}=\frac{2}{\pi c^{2}\hbar^{2}},\;q=2,\;a=3,\;b=-1.\,}
\label{eq:cv-final}
\end{equation}

\subsection*{(d) Charge neutrality at $\tilde\mu=0$ and total specific heat}

\paragraph{(i) Equal electron and hole numbers.} The valence band has $g_v(E)=(E_0-E)/(\pi c^2\hbar^2)$ for $E\le E_0$ (reflection of \eqref{eq:dos} about $E_0$). Hole occupation:
\begin{equation}
1-f(E)=\frac{1}{e^{\beta(\mu-E)}+1}.
\end{equation}
Hole density:
\begin{equation}
n_{\mathrm{h}}=\int_{-\infty}^{E_{0}}g_{v}(E)\bigl[1-f(E)\bigr]\,dE.
\end{equation}
Substitute $u=\beta(E_0-E)$:
\begin{equation}
n_{\mathrm{h}}=\frac{1}{\pi c^{2}\hbar^{2}\beta^{2}}\int_{0}^{\infty}\frac{u\,du}{e^{u+\alpha}+1}=\frac{F_{1}(-\alpha)}{\pi c^{2}\hbar^{2}\beta^{2}}.
\label{eq:nh}
\end{equation}
Set $\alpha=0$:
\begin{equation}
n_{\mathrm h}=\frac{F_1(0)}{\pi c^2\hbar^2\beta^2}=n_{\mathrm{el}}.
\end{equation}
By the same $\alpha\to-\alpha$ symmetry, the hole specific heat equals the electron specific heat at $\alpha=0$.

\paragraph{(ii) Total specific heat at $\tilde\mu=0$.} Sum and set $\alpha=0$ (so $\alpha F_1=0$):
\begin{equation}
c_{v}^{\mathrm{tot}}\bigr|_{\tilde\mu=0}=2\cdot\frac{2k_{\mathrm B}^{3}T^{2}}{\pi c^{2}\hbar^{2}}\cdot 3F_{2}(0)=\frac{12k_{\mathrm B}^{3}T^{2}F_2(0)}{\pi c^{2}\hbar^{2}}.
\label{eq:cvtot}
\end{equation}
Read $F_2(0)=3\zeta/4$ from the small-$\alpha$ expansion:
\begin{equation}
\boxed{\,c_{v}^{\mathrm{tot}}\bigr|_{\tilde\mu=0}=\frac{12\cdot 3\zeta/4}{\pi c^{2}\hbar^{2}}\,k_{\mathrm B}^{3}T^{2}=\frac{9\zeta\,k_{\mathrm B}^{3}T^{2}}{\pi c^{2}\hbar^{2}}.\,}
\label{eq:cvtot-final}
\end{equation}
Plug in $k_{\mathrm B}=\SI{1.381e-23}{J/K}$, $\hbar=\SI{1.055e-34}{J.s}$, $c=10^6\,\si{m/s}$, $T=100\,\si K$:
\begin{equation}
\pi c^{2}\hbar^{2}=\pi\cdot 10^{12}\cdot(1.055\times 10^{-34})^{2}=3.497\times 10^{-56}\,\si{J^2.m^2}.
\end{equation}
\begin{equation}
k_{\mathrm B}^{3}T^{2}=(1.381\times 10^{-23})^{3}\cdot 10^{4}=2.634\times 10^{-65}\,\si{J^3/K}.
\end{equation}
\begin{equation}
c_{v}^{\mathrm{tot}}=\frac{9\cdot 1.2021\cdot 2.634\times 10^{-65}}{3.497\times 10^{-56}}\approx 8.15\times 10^{-9}\,\si{J\,K^{-1}\,m^{-2}}.
\end{equation}

\subsection*{(e) Degenerate limit $\tilde\mu\gg k_{\mathrm B}T$}

Differentiate the given $F_2(\alpha)\approx\alpha^3/6+\pi^2\alpha/6$ to get $F_1$:
\begin{equation}
F_1(\alpha)=\frac{dF_2}{d\alpha}\approx\frac{\alpha^{2}}{2}+\frac{\pi^{2}}{6}.
\end{equation}

\paragraph{(i) Temperature dependence of $n_{\mathrm{el}}$.} Insert into \eqref{eq:nel-final}:
\begin{equation}
n_{\mathrm{el}}=\frac{(k_{\mathrm B}T)^{2}}{\pi c^{2}\hbar^{2}}\Bigl(\tfrac{\alpha^{2}}{2}+\tfrac{\pi^{2}}{6}\Bigr).
\end{equation}
Use $\alpha k_{\mathrm B}T=\tilde\mu$:
\begin{equation}
\boxed{\,n_{\mathrm{el}}=\frac{1}{\pi c^{2}\hbar^{2}}\Bigl[\tfrac{\tilde\mu^{2}}{2}+\tfrac{\pi^{2}}{6}(k_{\mathrm B}T)^{2}\Bigr].\,}
\label{eq:nel-degen}
\end{equation}

\paragraph{(ii) Chemical potential at $T\to 0$.} Drop the $T^2$ term:
\begin{equation}
n_{\mathrm{el}}=\frac{\tilde\mu^{2}}{2\pi c^{2}\hbar^{2}}.
\end{equation}
Solve for $\tilde\mu$:
\begin{equation}
\tilde\mu=\hbar c\sqrt{2\pi n_{\mathrm{el}}}.
\end{equation}
Restore $\mu=\tilde\mu+E_0$:
\begin{equation}
\mu\to E_{0}+\hbar c\sqrt{2\pi}\,n_{\mathrm{el}}^{1/2}.
\label{eq:mu-final}
\end{equation}
Read off:
\begin{equation}
\boxed{\,C_{3}=\hbar c\sqrt{2\pi},\qquad r=\tfrac{1}{2}.\,}
\label{eq:C3r}
\end{equation}

\section*{Question 4 --- Easy-axis anisotropy with transverse field, $j=1$}

\subsection*{(a) Matrix representation in the $j=1$ basis}

Work in $\{\ket{1,1},\ket{1,0},\ket{1,-1}\}$. List $\hat J^2$:
\begin{equation}
\hat J^{2}=j(j+1)\hbar^{2}\,\mathbb 1=2\hbar^{2}\,\mathbb 1,
\end{equation}
setting $\hbar=1$ for the numerical coefficients. List $\hat J_z$ and its square:
\begin{equation}
\hat J_{z}=\mathrm{diag}(1,0,-1),\qquad \hat J_{z}^{2}=\mathrm{diag}(1,0,1).
\end{equation}
Compute $\hat J^2-\hat J_z^2$:
\begin{equation}
\hat J^{2}-\hat J_{z}^{2}=\mathrm{diag}(2-1,2-0,2-1)=\mathrm{diag}(1,2,1).
\end{equation}
Halve it:
\begin{equation}
\tfrac12(\hat J^{2}-\hat J_{z}^{2})=\mathrm{diag}(\tfrac12,1,\tfrac12).
\end{equation}
Compute $\hat J_+^2\ket{1,-1}$ using $\hat J_+|1,m\rangle=\sqrt{(1-m)(2+m)}|1,m+1\rangle$:
\begin{equation}
\hat J_+\ket{1,-1}=\sqrt 2\ket{1,0},\quad\hat J_+^{2}\ket{1,-1}=\sqrt 2\,\hat J_+\ket{1,0}=2\ket{1,1}.
\end{equation}
All other actions of $\hat J_+^2$ vanish. By Hermitian conjugation $\hat J_-^2\ket{1,1}=2\ket{1,-1}$, so:
\begin{equation}
\tfrac{1}{4}(\hat J_+^{2}+\hat J_-^{2})=\begin{pmatrix}0&0&1/2\\0&0&0\\1/2&0&0\end{pmatrix}.
\end{equation}
Add the two pieces:
\begin{equation}
\hat J_{x}^{2}=\begin{pmatrix}1/2&0&1/2\\0&1&0\\1/2&0&1/2\end{pmatrix}.
\label{eq:Jx2}
\end{equation}
Match to the given form:
\begin{equation}
-D\hat J_x^2=-D\begin{pmatrix}a&0&a\\0&2a&0\\a&0&a\end{pmatrix}\;\Rightarrow\;\boxed{\,a=\tfrac12.\,}
\end{equation}
Match the Zeeman term:
\begin{equation}
-g\mu_{\mathrm B}B\hat J_z=-g\mu_{\mathrm B}B\,\mathrm{diag}(b,0,-b)=-g\mu_{\mathrm B}B\,\mathrm{diag}(1,0,-1)\;\Rightarrow\;\boxed{\,b=1.\,}
\end{equation}

\subsection*{(b) Eigenvalues to lowest order in $B$}

The middle row/column decouples; eigenvalue is
\begin{equation}
E_2=-D\cdot 2a=-D.
\end{equation}
The remaining $2\times 2$ block in $\{\ket{1,1},\ket{1,-1}\}$:
\begin{equation}
\mathsf H'=\begin{pmatrix}-D/2-g\mu_{\mathrm B}B&-D/2\\-D/2&-D/2+g\mu_{\mathrm B}B\end{pmatrix}.
\label{eq:Hprime}
\end{equation}
Compute the trace:
\begin{equation}
\mathrm{tr}\,\mathsf H'=-D.
\end{equation}
Compute the determinant:
\begin{equation}
\det\mathsf H'=(D/2)^{2}-(g\mu_{\mathrm B}B)^{2}-(D/2)^{2}=-(g\mu_{\mathrm B}B)^{2}.
\end{equation}
Solve the characteristic equation:
\begin{equation}
E_{\pm}=\frac{\mathrm{tr}\pm\sqrt{\mathrm{tr}^{2}-4\det}}{2}=-\frac{D}{2}\pm\frac{1}{2}\sqrt{D^{2}+4(g\mu_{\mathrm B}B)^{2}}.
\label{eq:Epm-exact}
\end{equation}
Expand the square root for $g\mu_{\mathrm B}B\ll D$:
\begin{equation}
\sqrt{D^{2}+4(g\mu_{\mathrm B}B)^{2}}=D\sqrt{1+4(g\mu_{\mathrm B}B)^{2}/D^{2}}\approx D+\frac{2(g\mu_{\mathrm B}B)^{2}}{D}.
\label{eq:sqrt-expand}
\end{equation}
Substitute into \eqref{eq:Epm-exact}:
\begin{equation}
\boxed{\,E_{1}=+\frac{(g\mu_{\mathrm B}B)^{2}}{D},\quad E_{2}=-D,\quad E_{3}=-D-\frac{(g\mu_{\mathrm B}B)^{2}}{D}.\,}
\label{eq:E123}
\end{equation}

\paragraph{Sketch.}
\begin{center}
\begin{tikzpicture}[scale=1]
\draw[->] (0,0) -- (5.2,0) node[right]{$B$};
\draw[->] (0,-2.6) -- (0,2) node[above]{$E$};
\draw[domain=0:5,smooth,thick,blue] plot(\x, {0.07*\x*\x});
\draw[domain=0:5,smooth,thick,red,dashed] plot(\x,-1.5);
\draw[domain=0:5,smooth,thick,red] plot(\x, {-1.5-0.07*\x*\x});
\node[blue] at (5.3,1.5){$E_{1}\propto+B^{2}$};
\node[red]  at (5.3,-1.4){$E_{2}=-D$};
\node[red]  at (5.3,-2.4){$E_{3}\propto-D-B^{2}$};
\node at (-0.3,0){$0$};
\node at (-0.4,-1.5){$-D$};
\end{tikzpicture}
\end{center}

\subsection*{(c) Partition function and transverse susceptibility}

\paragraph{(i) Partition function.} For $N$ independent ions:
\begin{equation}
Z_{N}(B)=[Z_{1}(B)]^{N},\qquad Z_{1}=e^{-\beta E_{1}}+e^{-\beta E_{2}}+e^{-\beta E_{3}}.
\label{eq:ZN}
\end{equation}
Introduce $w\equiv(g\mu_{\mathrm B})^{2}/D$ and insert \eqref{eq:E123}:
\begin{equation}
Z_{1}(B)=e^{-\beta w B^{2}}+e^{\beta D}+e^{\beta D+\beta w B^{2}}.
\label{eq:Z1}
\end{equation}

\paragraph{(ii) Magnetic moment in the low-field limit.} Expand each exponential to $O(B^{2})$:
\begin{equation}
e^{-\beta w B^{2}}\approx 1-\beta w B^{2},\qquad e^{\beta D+\beta w B^{2}}\approx e^{\beta D}(1+\beta w B^{2}).
\end{equation}
Add the three terms:
\begin{equation}
Z_{1}(B)\approx 1+2e^{\beta D}+\beta w B^{2}(e^{\beta D}-1).
\label{eq:Z1-expand}
\end{equation}
Take $\ln$ to leading order in $B^2$:
\begin{equation}
\ln Z_1\approx\ln(1+2e^{\beta D})+\frac{\beta w B^{2}(e^{\beta D}-1)}{1+2e^{\beta D}}.
\end{equation}
Compute the moment per ion:
\begin{equation}
m_{1}=-\pdone{F_{1}}{B}=\frac{1}{\beta}\pdone{\ln Z_{1}}{B}=\frac{2 w B(e^{\beta D}-1)}{1+2e^{\beta D}}+O(B^{3}).
\label{eq:m-ion}
\end{equation}
Multiply by $N$ for the ensemble: $m=Nm_1$.

\paragraph{(iii) Transverse susceptibility.} Differentiate at $B\to 0$:
\begin{equation}
\chi_{\perp}=\mu_{0}\frac{dm}{dB}\bigg|_{B\to 0}=\frac{2N\mu_{0}(g\mu_{\mathrm B})^{2}}{D}\cdot\frac{e^{\beta D}-1}{1+2e^{\beta D}}.
\label{eq:chiperp-raw}
\end{equation}
Multiply and divide by $\beta D/3$ to factor $\chi_C=2N\mu_0(g\mu_{\mathrm B})^2\beta/3$:
\begin{equation}
\chi_{\perp}=\chi_C\cdot\frac{3}{D\beta}\cdot\frac{e^{D\beta}-1}{1+2e^{D\beta}}.
\label{eq:chiperp-form}
\end{equation}
Read off $f_\perp$:
\begin{equation}
\boxed{\,\chi_{\perp}=\chi_{C}\,f_{\perp}(D\beta),\qquad f_{\perp}(x)=\frac{3}{x}\cdot\frac{e^{x}-1}{1+2e^{x}}.\,}
\label{eq:fperp}
\end{equation}

\subsection*{(d) Limits and comparison with parallel case}

\paragraph{(i) Asymptotic $T\to\infty$ ($x\to 0$).} Expand $e^x-1\approx x$, $1+2e^x\approx 3$:
\begin{equation}
f_{\parallel}(0)=\frac{3\cdot 1}{1+2}=1.
\end{equation}
\begin{equation}
f_{\perp}(0)=\frac{3}{x}\cdot\frac{x}{3}=1.
\end{equation}
Hence $\chi_\parallel,\chi_\perp\to\chi_C$.

\paragraph{(ii) $\chi_{\parallel}>\chi_{C}$ at all $T$.} Require $f_\parallel(x)>1$:
\begin{equation}
\frac{3e^{x}}{1+2e^{x}}>1\;\Longleftrightarrow\;3e^{x}>1+2e^{x}\;\Longleftrightarrow\;e^{x}>1,
\label{eq:fpar-ineq}
\end{equation}
which holds for all $x>0$.

\paragraph{(iii) $\chi_{\perp}$ finite at $T=0$.} Take $x\to\infty$: $e^x-1\sim e^x$, $1+2e^x\sim 2e^x$:
\begin{equation}
f_\perp(x)\to\frac{3}{x}\cdot\frac{e^x}{2e^x}=\frac{3}{2x}.
\end{equation}
Multiply by $\chi_C=2N\mu_0(g\mu_B)^2\beta/3$ and use $\beta=x/D$:
\begin{equation}
\chi_\perp\to\frac{2N\mu_0(g\mu_B)^2\beta}{3}\cdot\frac{3}{2D\beta}=\frac{N\mu_{0}(g\mu_{\mathrm B})^{2}}{D}.
\label{eq:chiperp-zeroT}
\end{equation}
Finite (Van Vleck).

\subsection*{(e) Interpretation of the figure and determination of $D$}

\paragraph{Physical picture.} At high $T$, $k_{\mathrm B}T\gg D$, so all three curves coincide with the Curie tail. At low $T$, $\chi_\parallel\to(3/2)\chi_C\to\infty$ while $\chi_\perp$ saturates at the temperature-independent Van Vleck value $N\mu_0(g\mu_B)^2/D$. The ground states $\ket{m_x=\pm1}$ have $\langle\hat J_z\rangle=0$; the transverse field admixes them with $\ket{m_x=0}$ at energy $D$ above, giving a $B$-linear induced moment with $T$-independent coefficient.

\paragraph{Extracting $D$.} Read off the saturation value:
\begin{equation}
\chi_{\perp}^{\mathrm{sat}}\approx 3\times 10^{-6}\,\si{m^{3}/mol}.
\end{equation}
Apply the Van Vleck expression (with $N\to N_A$ for molar susceptibility):
\begin{equation}
D=\frac{N_{A}\mu_{0}(g\mu_{\mathrm B})^{2}}{\chi_{\perp}^{\mathrm{sat}}}.
\label{eq:D-extract}
\end{equation}
Compute $(g\mu_B)^2$ with $g=2$, $\mu_B=9.274\times 10^{-24}\,\si{J/T}$:
\begin{equation}
(g\mu_B)^2=(1.8548\times 10^{-23})^2=3.440\times 10^{-46}\,\si{J^2/T^2}.
\end{equation}
Compute $N_A\mu_0(g\mu_B)^2$:
\begin{equation}
N_{A}\mu_{0}(g\mu_{\mathrm B})^{2}=6.022\times 10^{23}\cdot 1.2566\times 10^{-6}\cdot 3.440\times 10^{-46}=2.603\times 10^{-28}\,\si{J\,m^{3}/mol}.
\end{equation}
Divide by $\chi_\perp^{\mathrm{sat}}$:
\begin{equation}
D=\frac{2.603\times 10^{-28}}{3\times 10^{-6}}\,\si{J}=8.7\times 10^{-23}\,\si{J}.
\end{equation}
Convert to a temperature:
\begin{equation}
D/k_B=\frac{8.7\times 10^{-23}}{1.381\times 10^{-23}}\,\si K\approx 6.3\,\si K.
\end{equation}
Hence:
\begin{equation}
\boxed{\,D\approx 8.7\times 10^{-23}\,\si J\;\;(D/k_{\mathrm B}\approx 6\,\si K).\,}
\label{eq:D-final}
\end{equation}
Consistency: the curves visibly merge for $T\gtrsim 4D/k_B\approx 25\,\si K$, matching $f_\perp\approx f_\parallel\approx 1$ from part (d)(i).
