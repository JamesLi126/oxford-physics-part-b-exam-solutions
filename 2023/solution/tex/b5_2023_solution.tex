\section*{Question 1 --- Weak-field gravity, gravitomagnetism, Nordstr\"om's theory}

\textbf{Problem.}  Linearised Einstein gravity, $g_{\mu\nu}=\eta_{\mu\nu}+h_{\mu\nu}$, with weak-field Ricci tensor as quoted.  Show: (a) static non-relativistic source recovers $\Lap\Phi=4\pi G\rho$ with $h_{00}=-2\Phi/c^{2}$; (b) the test-particle equation of motion is $\ddot{\vb x}=-\nabla\Phi+\vb v\times\vb B_{G}$ with $\vb B_{G}=\nabla\times c\vb h$; (c) compare $|\vb B_{G}|$-acceleration to Newtonian gravity at geosynchronous orbit; (d) for Nordstr\"om's theory $g_{\mu\nu}=e^{2\Phi/c^{2}}\eta_{\mu\nu}$ with $R=-\kappa_{N}g_{\mu\nu}T^{\mu\nu}$, recover $\Lap\Phi=4\pi G\rho$ and find $\kappa_{N}/\kappa_{E}$; (e) explain why light is undeflected and $\vb B_{G}=0$.

\subsection*{(a) Newtonian limit}

Adopt signature $(-,+,+,+)$ and write the weak-field Ricci:
\[
R_{\mu\nu}=\tfrac{1}{2}\eta^{\alpha\gamma}\bigl(-\partial_{\mu}\partial_{\alpha}h_{\nu\gamma}+\partial_{\gamma}\partial_{\alpha}h_{\nu\mu}+\partial_{\mu}\partial_{\nu}h_{\alpha\gamma}-\partial_{\nu}\partial_{\gamma}h_{\alpha\mu}\bigr).
\]
Specialise to $\mu=\nu=0$ with $\partial_{0}h_{\mu\nu}=0$:
\[
R_{00}=\tfrac{1}{2}\eta^{\alpha\gamma}\bigl(-\partial_{0}\partial_{\alpha}h_{0\gamma}+\partial_{\gamma}\partial_{\alpha}h_{00}+\partial_{0}\partial_{0}h_{\alpha\gamma}-\partial_{0}\partial_{\gamma}h_{\alpha 0}\bigr).
\]
Drop $\partial_{0}$-terms:
\[
R_{00}=\tfrac{1}{2}\eta^{\alpha\gamma}\partial_{\gamma}\partial_{\alpha}h_{00}.
\]
Restrict to spatial sum $\alpha=\gamma=i$ (only spatial derivatives survive):
\[
R_{00}=\tfrac{1}{2}\Lap h_{00}.
\]
Trace-reverse the field equation by contracting with $g^{\mu\nu}$:
\[
R-\tfrac{1}{2}\cdot 4\,R=-\kappa_{E}T.
\]
Solve for $R$:
\[
R=\kappa_{E}T.
\]
Substitute back into the field equation:
\[
R_{\mu\nu}=-\kappa_{E}\bigl(T_{\mu\nu}-\tfrac{1}{2}T g_{\mu\nu}\bigr).
\]
Compute the trace of $T_{\mu\nu}$ for the non-relativistic source:
\[
T=g^{\mu\nu}T_{\mu\nu}=\eta^{00}T_{00}=(-1)(\rho c^{2})=-\rho c^{2}.
\]
Form the source combination:
\[
T_{00}-\tfrac{1}{2}T\eta_{00}=\rho c^{2}-\tfrac{1}{2}(-\rho c^{2})(-1).
\]
Simplify:
\[
T_{00}-\tfrac{1}{2}T\eta_{00}=\rho c^{2}-\tfrac{1}{2}\rho c^{2}=\tfrac{1}{2}\rho c^{2}.
\]
Insert into the $00$-equation $R_{00}=-\kappa_{E}(T_{00}-\tfrac{1}{2}T g_{00})$:
\[
\tfrac{1}{2}\Lap h_{00}=-\tfrac{1}{2}\kappa_{E}\rho c^{2}.
\]
Substitute $h_{00}=-2\Phi/c^{2}$:
\[
\tfrac{1}{2}\Lap\!\Bigl(-\tfrac{2\Phi}{c^{2}}\Bigr)=-\tfrac{1}{2}\kappa_{E}\rho c^{2}.
\]
Substitute $\kappa_{E}=8\pi G/c^{4}$:
\[
-\frac{1}{c^{2}}\Lap\Phi=-\frac{4\pi G}{c^{4}}\rho c^{2}.
\]
Cancel the $c^{2}$:
\[
-\frac{1}{c^{2}}\Lap\Phi=-\frac{4\pi G\rho}{c^{2}}.
\]
Multiply by $-c^{2}$:
\begin{equation}
\boxed{\;\Lap\Phi=4\pi G\rho.\;}
\label{eq:NewtonPoisson}
\end{equation}

\subsection*{(b) Gravitomagnetic equation of motion}

Write the geodesic equation with $u^{\alpha}=\de x^{\alpha}/\de\tau$:
\[
\frac{\de^{2}x^{\mu}}{\de\tau^{2}}+\Gamma^{\mu}_{\alpha\beta}u^{\alpha}u^{\beta}=0.
\]
Slow motion: $u^{0}\approx c$, $u^{i}\approx v^{i}$, $\de\tau\approx\de t$:
\[
\frac{\de^{2}x^{i}}{\de t^{2}}=-\Gamma^{i}_{\alpha\beta}\frac{u^{\alpha}u^{\beta}}{(u^{0})^{2}}c^{2}.
\]
Keep only terms up to first order in $v^{i}$:
\[
\frac{\de^{2}x^{i}}{\de t^{2}}=-c^{2}\Gamma^{i}_{00}-2c\,\Gamma^{i}_{0j}v^{j}.
\]
Compute the linearised Christoffel $\Gamma^{i}_{00}$ with $\partial_{0}h_{\mu\nu}=0$:
\[
\Gamma^{i}_{00}=\tfrac{1}{2}\eta^{ii}\bigl(\partial_{0}h_{i0}+\partial_{0}h_{0i}-\partial_{i}h_{00}\bigr)=-\tfrac{1}{2}\partial_{i}h_{00}.
\]
Compute $\Gamma^{i}_{0j}$ similarly:
\[
\Gamma^{i}_{0j}=\tfrac{1}{2}\eta^{ii}\bigl(\partial_{0}h_{ij}+\partial_{j}h_{0i}-\partial_{i}h_{0j}\bigr)=\tfrac{1}{2}\bigl(\partial_{j}h_{0i}-\partial_{i}h_{0j}\bigr).
\]
Evaluate the static piece using $h_{00}=-2\Phi/c^{2}$:
\[
-c^{2}\Gamma^{i}_{00}=\tfrac{1}{2}c^{2}\partial_{i}h_{00}.
\]
Substitute $h_{00}=-2\Phi/c^{2}$:
\[
-c^{2}\Gamma^{i}_{00}=\tfrac{1}{2}c^{2}\partial_{i}\!\bigl(-\tfrac{2\Phi}{c^{2}}\bigr)=-\partial_{i}\Phi.
\]
Evaluate the velocity-dependent piece:
\[
-2c\,\Gamma^{i}_{0j}v^{j}=-c\bigl(\partial_{j}h_{0i}-\partial_{i}h_{0j}\bigr)v^{j}.
\]
Define $\vb h=(h_{01},h_{02},h_{03})$ and $\vb B_{G}=\nabla\times c\vb h$, so $(\vb B_{G})_{k}=c\,\epsilon_{klm}\partial_{l}h_{0m}$:
\[
\bigl(\vb v\times\vb B_{G}\bigr)_{i}=\epsilon_{ijk}v^{j}(\vb B_{G})_{k}=c\,\epsilon_{ijk}\epsilon_{klm}v^{j}\partial_{l}h_{0m}.
\]
Apply $\epsilon_{ijk}\epsilon_{klm}=\delta_{il}\delta_{jm}-\delta_{im}\delta_{jl}$:
\[
\bigl(\vb v\times\vb B_{G}\bigr)_{i}=c\,v^{j}\bigl(\partial_{i}h_{0j}-\partial_{j}h_{0i}\bigr).
\]
Compare with the velocity-dependent piece:
\[
-2c\,\Gamma^{i}_{0j}v^{j}=\bigl(\vb v\times\vb B_{G}\bigr)_{i}.
\]
Combine the two pieces:
\begin{equation}
\boxed{\;\frac{\de^{2}\vb x}{\de t^{2}}=-\nabla\Phi+\vb v\times\vb B_{G},\qquad \vb B_{G}=\nabla\times c\vb h.\;}
\label{eq:gravito-eom}
\end{equation}

\subsection*{(c) Geosynchronous comparison}

Geosynchronous radius from Kepler with $T=\SI{86164}{s}$:
\[
r_{\rm geo}=\Bigl(\frac{GM_{\oplus}T^{2}}{4\pi^{2}}\Bigr)^{1/3}\approx \SI{4.22e7}{m}.
\]
Compute the Newtonian centripetal acceleration:
\[
a_{N}=\frac{GM_{\oplus}}{r_{\rm geo}^{2}}\approx \SI{0.224}{m\,s^{-2}}.
\]
Compute the orbital speed:
\[
v_{\rm geo}=\sqrt{\frac{GM_{\oplus}}{r_{\rm geo}}}\approx \SI{3.07e3}{m\,s^{-1}}.
\]
Compute Earth's angular momentum with $M_{\oplus}=\SI{5.97e24}{kg}$, $R_{\oplus}=\SI{6.37e6}{m}$, $\omega_{\oplus}=\SI{7.27e-5}{rad\,s^{-1}}$:
\[
J=\tfrac{2}{5}M_{\oplus}R_{\oplus}^{2}\omega_{\oplus}\approx \SI{7.05e33}{kg\,m^{2}\,s^{-1}}.
\]
Plug into the gravitomagnetic field amplitude:
\[
B_{G}=\frac{2GJ}{c^{2}r_{\rm geo}^{3}}=\frac{2\cdot \SI{6.67e-11}{}\cdot \SI{7.05e33}{}}{(\SI{3e8}{})^{2}(\SI{4.22e7}{})^{3}}.
\]
Evaluate:
\[
B_{G}\approx \SI{1.4e-16}{s^{-1}}.
\]
Compute the gravitomagnetic acceleration $a_{G}=v_{\rm geo}B_{G}$:
\[
a_{G}\approx \SI{3.07e3}{}\cdot \SI{1.4e-16}{}\approx \SI{4e-13}{m\,s^{-2}}.
\]
Form the ratio:
\[
\boxed{\;\frac{a_{G}}{a_{N}}\sim \frac{4\times 10^{-13}}{0.22}\approx 2\times 10^{-12}.\;}
\]
The gravitomagnetic acceleration is some twelve orders of magnitude smaller than Newtonian gravity at geosynchronous altitude.

\subsection*{(d) Nordstr\"om's scalar theory}

Expand the conformal metric to first order in $\Phi/c^{2}$:
\[
g_{\mu\nu}=e^{2\Phi/c^{2}}\eta_{\mu\nu}\approx \eta_{\mu\nu}+\frac{2\Phi}{c^{2}}\eta_{\mu\nu}.
\]
Read off $h_{\mu\nu}$:
\[
h_{\mu\nu}=\frac{2\Phi}{c^{2}}\eta_{\mu\nu}.
\]
Substitute into the weak-field Ricci:
\[
R_{\mu\nu}=\tfrac{1}{2}\eta^{\alpha\gamma}\bigl(-\partial_{\mu}\partial_{\alpha}h_{\nu\gamma}+\partial_{\gamma}\partial_{\alpha}h_{\nu\mu}+\partial_{\mu}\partial_{\nu}h_{\alpha\gamma}-\partial_{\nu}\partial_{\gamma}h_{\alpha\mu}\bigr).
\]
Use $\eta^{\alpha\gamma}\eta_{\nu\gamma}=\delta^{\alpha}_{\nu}$ in the first term:
\[
\tfrac{1}{2}\eta^{\alpha\gamma}(-\partial_{\mu}\partial_{\alpha}h_{\nu\gamma})=-\frac{1}{c^{2}}\partial_{\mu}\partial_{\nu}\Phi.
\]
Contract $\eta^{\alpha\gamma}\partial_{\gamma}\partial_{\alpha}=\Box$ in the second term:
\[
\tfrac{1}{2}\eta^{\alpha\gamma}\partial_{\gamma}\partial_{\alpha}h_{\nu\mu}=\frac{1}{c^{2}}\eta_{\mu\nu}\Box\Phi.
\]
Use $\eta^{\alpha\gamma}\eta_{\alpha\gamma}=4$ in the third term:
\[
\tfrac{1}{2}\eta^{\alpha\gamma}\partial_{\mu}\partial_{\nu}h_{\alpha\gamma}=\frac{4}{c^{2}}\partial_{\mu}\partial_{\nu}\Phi.
\]
Use $\eta^{\alpha\gamma}\eta_{\alpha\mu}=\delta^{\gamma}_{\mu}$ in the fourth term:
\[
-\tfrac{1}{2}\eta^{\alpha\gamma}\partial_{\nu}\partial_{\gamma}h_{\alpha\mu}=-\frac{1}{c^{2}}\partial_{\nu}\partial_{\mu}\Phi.
\]
Sum the four pieces:
\[
R_{\mu\nu}=\frac{1}{c^{2}}\bigl(-\partial_{\mu}\partial_{\nu}\Phi+\eta_{\mu\nu}\Box\Phi+4\partial_{\mu}\partial_{\nu}\Phi-\partial_{\mu}\partial_{\nu}\Phi\bigr).
\]
Collect like terms:
\[
R_{\mu\nu}=\frac{1}{c^{2}}\bigl(2\partial_{\mu}\partial_{\nu}\Phi+\eta_{\mu\nu}\Box\Phi\bigr).
\]
Take the trace with $\eta^{\mu\nu}$:
\[
R=\frac{1}{c^{2}}\bigl(2\Box\Phi+4\Box\Phi\bigr)=\frac{6}{c^{2}}\Box\Phi.
\]
Specialise to static fields with $\Box\to\Lap$:
\[
R=\frac{6}{c^{2}}\Lap\Phi.
\]
Compute the trace of the energy-momentum tensor in the non-relativistic limit:
\[
g_{\mu\nu}T^{\mu\nu}=T=-\rho c^{2}.
\]
Insert into $R=-\kappa_{N}g_{\mu\nu}T^{\mu\nu}=-\kappa_{N}T$:
\[
\frac{6}{c^{2}}\Lap\Phi=\kappa_{N}\rho c^{2}.
\]
Solve for $\Lap\Phi$:
\[
\Lap\Phi=\frac{\kappa_{N}c^{4}}{6}\rho.
\]
Match to $\Lap\Phi=4\pi G\rho$:
\[
\frac{\kappa_{N}c^{4}}{6}=4\pi G.
\]
Solve for $\kappa_{N}$:
\[
\kappa_{N}=\frac{24\pi G}{c^{4}}.
\]
Compare with $\kappa_{E}=8\pi G/c^{4}$:
\begin{equation}
\boxed{\;\kappa_{N}=3\kappa_{E}.\;}
\label{eq:kappaN}
\end{equation}

\subsection*{(e) No light deflection, $\vb B_{G}=0$}

Conformal metric: under $g_{\mu\nu}=e^{2\Phi/c^{2}}\eta_{\mu\nu}$, null geodesics of $g$ coincide with null geodesics of $\eta$ up to reparametrisation, since the null condition $g_{\mu\nu}\de x^{\mu}\de x^{\nu}=0$ is preserved by the conformal factor and the geodesic equation for null curves is conformally invariant.  Light therefore travels on Minkowski straight lines: \emph{no deflection}.

Off-diagonal components: in this theory $h_{\mu\nu}\propto\eta_{\mu\nu}$ is purely diagonal, so $h_{0i}=0$ identically.  Hence
\[
\boxed{\;\vb B_{G}=\nabla\times c\vb h=\vb 0.\;}
\]
There is no gravitomagnetic / frame-dragging effect: the theory has no rotational ``magnetic'' analogue.

\section*{Question 2 --- Geodesic deviation; Schwarzschild orbits}

\textbf{Problem.}  (a) Derive the geodesic-deviation equation in a LIF and justify covariance.  (b) For the metric $\de s^{2}=-(1-F)c^{2}\de t^{2}+\de r^{2}/(1-F)+r^{2}(\de\theta^{2}+\sin^{2}\theta\de\phi^{2})$, $F=F(r)$, show that an equatorial orbit obeys $(1-F)\dot t=k$ and $r^{2}\dot\phi=h$.  (c) For $F=r_{S}/r$, derive the radial equation, define $V_{\rm eff}$, find ISCO and show $k=2\sqrt{2}/3$.  (d) Radial fall from rest at infinity into the black hole: find $r(\tau)$ and $\de r/\de t$ at $r=r_{S}$.

\subsection*{(a) Geodesic deviation in a LIF}

Write the geodesic equation for both curves $x^{\alpha}$ and $x^{\alpha}+\Delta x^{\alpha}$:
\[
\ddot x^{\lambda}+\Gamma^{\lambda}_{\alpha\beta}(x)\dot x^{\alpha}\dot x^{\beta}=0,
\]
\[
\ddot x^{\lambda}+\ddot{\Delta x}^{\lambda}+\Gamma^{\lambda}_{\alpha\beta}(x+\Delta x)(\dot x^{\alpha}+\dot{\Delta x}^{\alpha})(\dot x^{\beta}+\dot{\Delta x}^{\beta})=0.
\]
Choose a Local Inertial Frame at $P$: $\Gamma^{\lambda}_{\alpha\beta}(P)=0$, $g_{\mu\nu}(P)=\eta_{\mu\nu}$, $\partial_{\sigma}g_{\mu\nu}(P)=0$.

Taylor-expand $\Gamma^{\lambda}_{\alpha\beta}(x+\Delta x)$ about $P$:
\[
\Gamma^{\lambda}_{\alpha\beta}(x+\Delta x)=\Gamma^{\lambda}_{\alpha\beta}(P)+\bigl(\partial_{\nu}\Gamma^{\lambda}_{\alpha\beta}\bigr)_{P}\Delta x^{\nu}+\dots
\]
Subtract the two geodesic equations and keep terms linear in $\Delta x$:
\[
\ddot{\Delta x}^{\lambda}+\bigl(\partial_{\nu}\Gamma^{\lambda}_{\alpha\beta}\bigr)_{P}\Delta x^{\nu}\dot x^{\alpha}\dot x^{\beta}=0.
\]
At $P$ in the LIF the Riemann tensor reads
\[
R^{\lambda}{}_{\alpha\nu\beta}\bigl|_{P}=\partial_{\nu}\Gamma^{\lambda}_{\alpha\beta}-\partial_{\beta}\Gamma^{\lambda}_{\alpha\nu}\bigl|_{P}.
\]
Compute $D^{2}\Delta x^{\lambda}/\de\tau^{2}$ at $P$ in the LIF:
\[
\frac{D^{2}\Delta x^{\lambda}}{\de\tau^{2}}\Big|_{P}=\ddot{\Delta x}^{\lambda}+\bigl(\partial_{\beta}\Gamma^{\lambda}_{\alpha\nu}\bigr)_{P}\dot x^{\alpha}\dot x^{\beta}\Delta x^{\nu}.
\]
Subtract this from the linearised geodesic difference:
\[
\frac{D^{2}\Delta x^{\lambda}}{\de\tau^{2}}-\bigl(\partial_{\nu}\Gamma^{\lambda}_{\alpha\beta}-\partial_{\beta}\Gamma^{\lambda}_{\alpha\nu}\bigr)\dot x^{\alpha}\dot x^{\beta}\Delta x^{\nu}=0.
\]
Identify the bracket as $R^{\lambda}{}_{\alpha\nu\beta}$:
\begin{equation}
\boxed{\;\frac{D^{2}\Delta x^{\lambda}}{\de\tau^{2}}-R^{\lambda}{}_{\alpha\nu\beta}\dot x^{\alpha}\dot x^{\beta}\Delta x^{\nu}=0.\;}
\label{eq:GD}
\end{equation}
Both sides are tensors so by the principle of general covariance the equation, valid at $P$ in the LIF, holds in any frame.  The relative acceleration of neighbouring free-falling test particles is sourced by the Riemann curvature: spacetime curvature manifests as tidal gravity — gravitation is geometry.

\subsection*{(b) Conserved quantities}

Write the Lagrangian for an equatorial orbit ($\theta=\pi/2$, $\dot\theta=0$):
\[
2\mathcal L=-(1-F)c^{2}\dot t^{2}+\frac{\dot r^{2}}{1-F}+r^{2}\dot\phi^{2}.
\]
$\mathcal L$ has no explicit $t$ or $\phi$ dependence:
\[
\frac{\de}{\de\tau}\frac{\partial\mathcal L}{\partial\dot t}=0,\qquad \frac{\de}{\de\tau}\frac{\partial\mathcal L}{\partial\dot\phi}=0.
\]
Compute the conjugate momenta:
\[
\frac{\partial\mathcal L}{\partial\dot t}=-(1-F)c^{2}\dot t,\qquad \frac{\partial\mathcal L}{\partial\dot\phi}=r^{2}\dot\phi.
\]
Absorb constants:
\begin{equation}
\boxed{\;(1-F)\dot t=k,\qquad r^{2}\dot\phi=h.\;}
\label{eq:cons}
\end{equation}

\subsection*{(c) Radial equation, ISCO}

Write the normalisation for a massive particle:
\[
g_{\mu\nu}\dot x^{\mu}\dot x^{\nu}=-c^{2}.
\]
Expand using the equatorial metric:
\[
-(1-F)c^{2}\dot t^{2}+\frac{\dot r^{2}}{1-F}+r^{2}\dot\phi^{2}=-c^{2}.
\]
Substitute $\dot t=k/(1-F)$ and $\dot\phi=h/r^{2}$:
\[
-\frac{c^{2}k^{2}}{1-F}+\frac{\dot r^{2}}{1-F}+\frac{h^{2}}{r^{2}}=-c^{2}.
\]
Multiply through by $(1-F)$:
\[
\dot r^{2}-c^{2}k^{2}+(1-F)\Bigl(\frac{h^{2}}{r^{2}}+c^{2}\Bigr)=0.
\]
Insert $F=r_{S}/r$ and expand the bracket:
\[
\dot r^{2}-c^{2}k^{2}+\frac{h^{2}}{r^{2}}-\frac{r_{S}h^{2}}{r^{3}}+c^{2}-\frac{c^{2}r_{S}}{r}=0.
\]
Rearrange to the question form:
\begin{equation}
\boxed{\;\dot r^{2}-\frac{c^{2}r_{S}}{r}+\frac{h^{2}}{r^{2}}-\frac{r_{S}h^{2}}{r^{3}}=c^{2}(k^{2}-1).\;}
\label{eq:rad}
\end{equation}

Define the effective potential by $\dot r^{2}+2V_{\rm eff}(r)=c^{2}(k^{2}-1)$:
\[
V_{\rm eff}(r)=-\frac{c^{2}r_{S}}{2r}+\frac{h^{2}}{2r^{2}}-\frac{r_{S}h^{2}}{2r^{3}}.
\]
Set $V'_{\rm eff}=0$ for circular orbits:
\[
V'_{\rm eff}=\frac{c^{2}r_{S}}{2r^{2}}-\frac{h^{2}}{r^{3}}+\frac{3r_{S}h^{2}}{2r^{4}}=0.
\]
Multiply by $2r^{4}/r_{S}$:
\[
c^{2}r^{2}-\frac{2h^{2}r}{r_{S}}+3h^{2}=0.
\]
Solve as a quadratic in $r$:
\[
r_{\pm}=\frac{h^{2}}{c^{2}r_{S}}\Bigl(1\pm\sqrt{1-\frac{3c^{2}r_{S}^{2}}{h^{2}}}\Bigr).
\]
$r_{+}$ is the stable circular orbit (minimum of $V_{\rm eff}$); $r_{-}$ is unstable (maximum).  The two coalesce at the innermost stable circular orbit (ISCO) when the discriminant vanishes:
\[
1-\frac{3c^{2}r_{S}^{2}}{h^{2}}=0.
\]
Solve for $h_{\rm ISCO}$:
\[
h_{\rm ISCO}^{2}=3c^{2}r_{S}^{2}.
\]
Substitute back into $r_{\pm}$:
\[
r_{\rm ISCO}=\frac{h_{\rm ISCO}^{2}}{c^{2}r_{S}}=\frac{3c^{2}r_{S}^{2}}{c^{2}r_{S}}=3r_{S}.
\]
Box the ISCO parameters:
\[
\boxed{\;r_{\rm ISCO}=3r_{S},\qquad h_{\rm ISCO}=\sqrt{3}\,c\,r_{S}.\;}
\]
Substitute $\dot r=0$, $r=3r_{S}$, $h^{2}=3c^{2}r_{S}^{2}$ into \eqref{eq:rad}:
\[
-\frac{c^{2}r_{S}}{3r_{S}}+\frac{3c^{2}r_{S}^{2}}{(3r_{S})^{2}}-\frac{r_{S}\cdot 3c^{2}r_{S}^{2}}{(3r_{S})^{3}}=c^{2}(k^{2}-1).
\]
Reduce each fraction:
\[
-\tfrac{1}{3}c^{2}+\tfrac{1}{3}c^{2}-\tfrac{1}{9}c^{2}=c^{2}(k^{2}-1).
\]
Combine the left side:
\[
-\tfrac{1}{9}c^{2}=c^{2}(k^{2}-1).
\]
Cancel $c^{2}$ and solve:
\[
k^{2}=1-\tfrac{1}{9}=\tfrac{8}{9}.
\]
Take the positive root:
\[
\boxed{\;k=\frac{2\sqrt{2}}{3}.\;}
\]

\subsection*{(d) Radial fall from rest at infinity}

Set $h=0$ in \eqref{eq:rad}:
\[
\dot r^{2}-\frac{c^{2}r_{S}}{r}=c^{2}(k^{2}-1).
\]
``No energy at infinity'' means $\dot r\to 0$ as $r\to\infty$, fixing $k=1$:
\[
\dot r^{2}=\frac{c^{2}r_{S}}{r}.
\]
Take the in-falling root $\dot r<0$:
\[
\frac{\de r}{\de\tau}=-c\sqrt{\frac{r_{S}}{r}}.
\]
Separate variables:
\[
\sqrt{r}\,\de r=-c\sqrt{r_{S}}\,\de\tau.
\]
Integrate from $r_{0}$ at $\tau=0$:
\[
\tfrac{2}{3}\bigl(r^{3/2}-r_{0}^{3/2}\bigr)=-c\sqrt{r_{S}}\,\tau.
\]
Solve for $r$:
\[
r^{3/2}=r_{0}^{3/2}-\tfrac{3}{2}c\sqrt{r_{S}}\,\tau.
\]
Box the trajectory:
\begin{equation}
\boxed{\;r(\tau)=\Bigl[r_{0}^{3/2}-\tfrac{3}{2}c\sqrt{r_{S}}\,\tau\Bigr]^{2/3}.\;}
\label{eq:radialfall}
\end{equation}
The particle reaches $r=0$ in finite proper time $\tau_{\rm fall}=\tfrac{2}{3}r_{0}^{3/2}/(c\sqrt{r_{S}})$.

Use $\dot t=k/(1-F)=1/(1-r_{S}/r)$:
\[
\frac{\de r}{\de t}=\frac{\dot r}{\dot t}=\dot r\,(1-F).
\]
Substitute $\dot r=-c\sqrt{r_{S}/r}$ and $F=r_{S}/r$:
\[
\frac{\de r}{\de t}=-c\sqrt{\frac{r_{S}}{r}}\Bigl(1-\frac{r_{S}}{r}\Bigr).
\]
Take the limit $r\to r_{S}$:
\[
\boxed{\;\frac{\de r}{\de t}\to 0\;\text{ as }\;r\to r_{S}.\;}
\]
A distant Schwarzschild observer sees the particle freeze at the horizon, while in proper time the crossing is unremarkable.

\section*{Question 3 --- Plane gravitational-wave metric}

\textbf{Problem.}  Vacuum metric $\de s^{2}=-c^{2}\de t^{2}+\alpha^{2}(e^{2\beta}\de x^{2}+e^{-2\beta}\de y^{2})+\de z^{2}$, with $\alpha,\beta$ functions of $ct+z$.  (a) Change to $u=ct+z$, $v=ct-z$; obtain $\de s^{2}=-\de u\,\de v+\alpha^{2}(e^{2\beta}\de x^{2}+e^{-2\beta}\de y^{2})$ and compute the Christoffels.  (b) Show $\dot t=-1$, $\dot x=\dot y=\dot z=0$ is a timelike geodesic.  (c) Show the only non-zero Ricci component is $R_{uu}=-2(\alpha''/\alpha+\beta'^{2})$ and explain how Einstein's equations reduce.  (d) For $\beta=0$, $\alpha=1$ on $u<0$ and $\beta=u$ for $u\ge 0$, find $\alpha(u)$ and describe a ring of test particles.  (e) Comment on its interpretation as an exact transverse gravitational wave.

\subsection*{(a) Light-cone form and Christoffels}

Apply the coordinate change $u=ct+z$, $v=ct-z$:
\[
ct=\tfrac{1}{2}(u+v),\qquad z=\tfrac{1}{2}(u-v).
\]
Differentiate:
\[
c\de t=\tfrac{1}{2}(\de u+\de v),\qquad \de z=\tfrac{1}{2}(\de u-\de v).
\]
Compute $-c^{2}\de t^{2}+\de z^{2}$:
\[
-c^{2}\de t^{2}+\de z^{2}=-\tfrac{1}{4}(\de u+\de v)^{2}+\tfrac{1}{4}(\de u-\de v)^{2}.
\]
Expand both squares:
\[
=-\tfrac{1}{4}(\de u^{2}+2\de u\de v+\de v^{2})+\tfrac{1}{4}(\de u^{2}-2\de u\de v+\de v^{2}).
\]
Cancel:
\[
-c^{2}\de t^{2}+\de z^{2}=-\de u\,\de v.
\]
Hence the metric becomes
\begin{equation}
\boxed{\;\de s^{2}=-\de u\,\de v+\alpha^{2}\bigl(e^{2\beta}\de x^{2}+e^{-2\beta}\de y^{2}\bigr).\;}
\label{eq:ppmetric}
\end{equation}
Read off non-zero metric components:
\[
g_{uv}=g_{vu}=-\tfrac{1}{2},\quad g_{xx}=\alpha^{2}e^{2\beta},\quad g_{yy}=\alpha^{2}e^{-2\beta}.
\]
Invert (the $(u,v)$ block has determinant $-1/4$):
\[
g^{uv}=g^{vu}=-2,\quad g^{xx}=\alpha^{-2}e^{-2\beta},\quad g^{yy}=\alpha^{-2}e^{2\beta}.
\]
Differentiate $g_{xx}$ with respect to $u$:
\[
\partial_{u}g_{xx}=2\alpha\alpha'e^{2\beta}+2\alpha^{2}\beta'e^{2\beta}=2\alpha^{2}e^{2\beta}\bigl(\alpha'/\alpha+\beta'\bigr).
\]
Similarly differentiate $g_{yy}$:
\[
\partial_{u}g_{yy}=2\alpha^{2}e^{-2\beta}\bigl(\alpha'/\alpha-\beta'\bigr).
\]
Apply $\Gamma^{\lambda}_{\mu\nu}=\tfrac{1}{2}g^{\lambda\sigma}(\partial_{\mu}g_{\sigma\nu}+\partial_{\nu}g_{\sigma\mu}-\partial_{\sigma}g_{\mu\nu})$ to obtain $\Gamma^{v}_{xx}$ (only $g^{vu}$ contributes, with $\partial_x$-terms zero):
\[
\Gamma^{v}_{xx}=-\tfrac{1}{2}g^{vu}\partial_{u}g_{xx}=-\tfrac{1}{2}(-2)\partial_{u}g_{xx}.
\]
Substitute $\partial_{u}g_{xx}$:
\[
\Gamma^{v}_{xx}=2\alpha^{2}e^{2\beta}\bigl(\alpha'/\alpha+\beta'\bigr).
\]
Analogously for $\Gamma^{v}_{yy}$:
\[
\Gamma^{v}_{yy}=2\alpha^{2}e^{-2\beta}\bigl(\alpha'/\alpha-\beta'\bigr).
\]
Compute $\Gamma^{x}_{ux}=\tfrac{1}{2}g^{xx}\partial_{u}g_{xx}$:
\[
\Gamma^{x}_{ux}=\tfrac{1}{2}\alpha^{-2}e^{-2\beta}\cdot 2\alpha^{2}e^{2\beta}(\alpha'/\alpha+\beta')=\frac{\alpha'}{\alpha}+\beta'.
\]
Likewise:
\[
\Gamma^{y}_{uy}=\frac{\alpha'}{\alpha}-\beta'.
\]
Collect:
\[
\boxed{\;\Gamma^{v}_{xx}=2\alpha^{2}e^{2\beta}\!\bigl(\tfrac{\alpha'}{\alpha}+\beta'\bigr),\;\Gamma^{v}_{yy}=2\alpha^{2}e^{-2\beta}\!\bigl(\tfrac{\alpha'}{\alpha}-\beta'\bigr),\;\Gamma^{x}_{ux}=\tfrac{\alpha'}{\alpha}+\beta',\;\Gamma^{y}_{uy}=\tfrac{\alpha'}{\alpha}-\beta'.\;}
\]
All other Christoffels vanish.

\subsection*{(b) The curve $\dot t=-1$, $\dot{\vb x}=\vb 0$ is a timelike geodesic}

Compute $\dot u$ and $\dot v$ from $u=ct+z$, $v=ct-z$ with $\dot t=-1$, $\dot z=0$:
\[
\dot u=c\dot t+\dot z=-c,\qquad \dot v=c\dot t-\dot z=-c.
\]
Both are constants, so:
\[
\ddot u=\ddot v=\ddot x=\ddot y=0.
\]
Compute the connection contraction. Only $\dot u,\dot v$ are nonzero:
\[
\Gamma^{\lambda}_{\mu\nu}\dot x^{\mu}\dot x^{\nu}=\Gamma^{\lambda}_{uu}\dot u^{2}+2\Gamma^{\lambda}_{uv}\dot u\dot v+\Gamma^{\lambda}_{vv}\dot v^{2}.
\]
From the Christoffel list, $\Gamma^{\lambda}_{uu}=\Gamma^{\lambda}_{uv}=\Gamma^{\lambda}_{vv}=0$ for all $\lambda$:
\[
\Gamma^{\lambda}_{\mu\nu}\dot x^{\mu}\dot x^{\nu}=0.
\]
The geodesic equation $\ddot x^{\lambda}+\Gamma^{\lambda}_{\mu\nu}\dot x^{\mu}\dot x^{\nu}=0$ is satisfied identically.

Compute the norm:
\[
g_{\mu\nu}\dot x^{\mu}\dot x^{\nu}=2g_{uv}\dot u\dot v=2\bigl(-\tfrac{1}{2}\bigr)(-c)(-c)=-c^{2}<0.
\]
Hence the curve is timelike. $\Box$

\subsection*{(c) Ricci tensor and Einstein equations}

The metric depends only on $u$, so $\partial_{v}h=\partial_{x}h=\partial_{y}h=0$ on every metric quantity.  Compute the Ricci tensor directly from the Christoffels using $R_{\alpha\beta}=-R^{\gamma}{}_{\alpha\beta\gamma}$.

Compute $R^{\gamma}{}_{uu\gamma}$ from the Riemann formula with $\alpha=\beta=u$, $\delta=\gamma$:
\[
R^{\gamma}{}_{uu\gamma}=\partial_{\gamma}\Gamma^{\gamma}_{uu}-\partial_{u}\Gamma^{\gamma}_{u\gamma}+\Gamma^{\epsilon}_{uu}\Gamma^{\gamma}_{\gamma\epsilon}-\Gamma^{\epsilon}_{u\gamma}\Gamma^{\gamma}_{u\epsilon}.
\]
Drop the $\Gamma^{\gamma}_{uu}=0$ and $\Gamma^{\epsilon}_{uu}=0$ terms:
\[
R^{\gamma}{}_{uu\gamma}=-\partial_{u}\Gamma^{\gamma}_{u\gamma}-\Gamma^{\epsilon}_{u\gamma}\Gamma^{\gamma}_{u\epsilon}.
\]
Compute the trace $\Gamma^{\gamma}_{u\gamma}=\Gamma^{x}_{ux}+\Gamma^{y}_{uy}$:
\[
\Gamma^{\gamma}_{u\gamma}=\bigl(\tfrac{\alpha'}{\alpha}+\beta'\bigr)+\bigl(\tfrac{\alpha'}{\alpha}-\beta'\bigr)=\frac{2\alpha'}{\alpha}.
\]
Differentiate:
\[
\partial_{u}\Gamma^{\gamma}_{u\gamma}=\frac{2\alpha''}{\alpha}-\frac{2\alpha'^{2}}{\alpha^{2}}.
\]
Compute the quadratic term, with only $(\epsilon,\gamma)=(x,x)$ and $(y,y)$ surviving:
\[
\Gamma^{\epsilon}_{u\gamma}\Gamma^{\gamma}_{u\epsilon}=(\Gamma^{x}_{ux})^{2}+(\Gamma^{y}_{uy})^{2}=\bigl(\tfrac{\alpha'}{\alpha}+\beta'\bigr)^{2}+\bigl(\tfrac{\alpha'}{\alpha}-\beta'\bigr)^{2}.
\]
Expand and collect:
\[
\Gamma^{\epsilon}_{u\gamma}\Gamma^{\gamma}_{u\epsilon}=2\frac{\alpha'^{2}}{\alpha^{2}}+2\beta'^{2}.
\]
Substitute both pieces:
\[
R^{\gamma}{}_{uu\gamma}=-\Bigl(\frac{2\alpha''}{\alpha}-\frac{2\alpha'^{2}}{\alpha^{2}}\Bigr)-\Bigl(\frac{2\alpha'^{2}}{\alpha^{2}}+2\beta'^{2}\Bigr).
\]
Cancel the $\alpha'^{2}/\alpha^{2}$ pieces:
\[
R^{\gamma}{}_{uu\gamma}=-\frac{2\alpha''}{\alpha}-2\beta'^{2}=-2\Bigl(\frac{\alpha''}{\alpha}+\beta'^{2}\Bigr).
\]
Identify with the question's $R_{uu}$ (taking the contraction $R^{\gamma}{}_{uu\gamma}$ directly to match the stated form):
\begin{equation}
\boxed{\;R_{uu}=-2\Bigl(\frac{\alpha''}{\alpha}+\beta'^{2}\Bigr).\;}
\label{eq:Ruu}
\end{equation}

Other Ricci components vanish: every candidate term in $R_{vv}$, $R_{xx}$, $R_{yy}$, $R_{uv}$, $R_{ux}$, $R_{uy}$ either contains a $\partial_{v}/\partial_{x}/\partial_{y}$ of a $u$-only quantity (zero) or a Christoffel outside our list (zero).  Hence $R_{uu}$ is the only non-zero component.

Vacuum equations give $T_{\mu\nu}=0\Rightarrow R_{\mu\nu}=0$; taking the trace of $R_{\mu\nu}-\tfrac{1}{2}g_{\mu\nu}R=0$ yields $R=0$, hence $R_{\mu\nu}=0$.  Therefore
\begin{equation}
\boxed{\;\frac{\alpha''}{\alpha}+\beta'^{2}=0.\;}
\label{eq:alphaeq}
\end{equation}
Given $\beta(u)$ and initial data $\alpha(u_{0}),\alpha'(u_{0})$, this is a linear second-order ODE for $\alpha(u)$ that integrates uniquely.  $\beta$ encodes the wave's profile; $\alpha$ is then determined.

\subsection*{(d) Pulse $\beta=u\,\Theta(u)$, $\alpha(u<0)=1$}

For $u<0$ the data give $\beta=0$, $\beta'=0$, so $\alpha''=0$.  With $\alpha=1$ specified:
\[
\alpha(u)=1\quad(u<0).
\]
For $u\ge 0$ use $\beta=u$, $\beta'=1$ in \eqref{eq:alphaeq}:
\[
\alpha''+\alpha=0.
\]
Write the general solution:
\[
\alpha(u)=A\cos u+B\sin u.
\]
Match $\alpha$ and $\alpha'$ continuously at $u=0$ to the past values $\alpha=1$, $\alpha'=0$:
\[
A=1,\qquad B=0.
\]
Box the result:
\begin{equation}
\boxed{\;\alpha(u)=\cos u\;\;(0\le u<\pi/2),\qquad \beta(u)=u.\;}
\label{eq:alpha-pulse}
\end{equation}
The metric in the wave region is
\[
\de s^{2}=-\de u\,\de v+\cos^{2}\!u\bigl(e^{2u}\de x^{2}+e^{-2u}\de y^{2}\bigr).
\]
A ring of test particles initially at rest stays at fixed $(x,y)$ (part (b) geodesic), but their proper distances oscillate as the spatial metric coefficients vary with $u$:
\[
\ell_{x}(u)=\alpha e^{\beta}=e^{u}\cos u,\qquad \ell_{y}(u)=\alpha e^{-\beta}=e^{-u}\cos u.
\]
The ring becomes an ellipse with $\ell_{x}$ stretching, $\ell_{y}$ contracting; both shrink as $\cos u\to 0$ at $u=\pi/2$.  At $u=\pi/2$ both vanish: $\alpha\to 0$ is a coordinate caustic — the wave focusses geodesics to a point.

\subsection*{(e) Interpretation as an exact gravitational wave}

The non-trivial metric coefficients depend only on the null coordinate $u=ct+z$: surfaces of constant phase $u={\rm const}$ are null hyperplanes propagating at $c$ in the $+z$ direction.  The wave is \emph{transverse}: the ``$+$''-polarisation strain $e^{\pm\beta(u)}$ acts purely in the $(x,y)$ plane orthogonal to propagation, and the metric is non-trivial only in those directions.  $R_{uu}$ contains the wave amplitude ($\beta'^{2}$) plus the back-reaction of the wave on the transverse area (the $\alpha''/\alpha$ term); vacuum forces these to balance.

Differences from linearised theory:
\begin{itemize}
\item The metric is an \emph{exact} solution of $R_{\mu\nu}=0$ — no expansion in $h_{\mu\nu}$.
\item There is non-linear back-reaction of the wave on its own background: $\alpha(u)$ is driven by $\beta'^{2}$ and can focus geodesics, producing a caustic (here at $u=\pi/2$).  Linearised waves on Minkowski space have no such focussing.
\item Polarisation amplitudes are not bounded — finite-amplitude pulses can produce singularities.
\end{itemize}

\section*{Question 4 --- Closed Universe with matter and dark energy}

\textbf{Problem.}  Closed FRW with non-relativistic matter plus dark energy of density $\rho_{E}\propto R^{-n}$.  (a) Equation of state for the dark fluid.  (b) Friedmann equation in form $\dot R^{2}/R^{2}=(8\pi G\rho_{0}/3R^{3})[1-R/R_{T}+(R/R_{E})^{3-n}]$ and the corresponding $\ddot R/R$.  (c) $n=0$: find Einstein static $R_{S}$, $R_{E}(R_{T})$, derive effective potential, show instability.  (d) $n=2-\epsilon$, $0<\epsilon\ll 1$: loitering Universe.  (e) Quadratic $V_{\rm eff}$ around $R_{S}$, show expansion time $R_{S}\to 2R_{S}$ diverges as $\epsilon\to 0$.

\subsection*{(a) Equation of state of the dark energy}

Energy-momentum conservation for a perfect fluid in FRW:
\[
\dot\rho+3\frac{\dot R}{R}\bigl(\rho+p/c^{2}\bigr)=0.
\]
Apply to the dark energy with $\rho_{E}=\rho_{0E}R^{-n}$ (constants absorbed):
\[
\dot\rho_{E}=-n\rho_{0E}R^{-n-1}\dot R=-n\frac{\dot R}{R}\rho_{E}.
\]
Substitute:
\[
-n\frac{\dot R}{R}\rho_{E}+3\frac{\dot R}{R}\bigl(\rho_{E}+p_{E}/c^{2}\bigr)=0.
\]
Divide by $\dot R/R$:
\[
-n\rho_{E}+3\rho_{E}+3p_{E}/c^{2}=0.
\]
Solve:
\begin{equation}
\boxed{\;p_{E}=\frac{n-3}{3}\rho_{E}c^{2},\qquad w_{E}=\frac{n-3}{3}.\;}
\label{eq:eosE}
\end{equation}
$n=0\Rightarrow w=-1$ (cosmological constant); $n=3\Rightarrow w=0$ (dust); $n=4\Rightarrow w=1/3$ (radiation).

\subsection*{(b) Friedmann equations}

Sum the two component densities:
\[
\rho=\rho_{m}+\rho_{E}=\frac{\rho_{m,0}}{R^{3}}+\frac{\rho_{E,0}}{R^{n}}.
\]
Write the rubric's Friedmann equation:
\[
\dot R^{2}=\frac{8\pi G}{3}\rho R^{2}+\tfrac{1}{3}\Lambda c^{2}R^{2}-c^{2}k.
\]
Set $\Lambda=0$ and $k=+1$ for the closed Universe:
\[
\dot R^{2}=\frac{8\pi G}{3}\Bigl(\frac{\rho_{m,0}}{R}+\rho_{E,0}R^{2-n}\Bigr)-c^{2}.
\]
Divide by $R^{2}$:
\[
\Bigl(\frac{\dot R}{R}\Bigr)^{2}=\frac{8\pi G}{3}\Bigl(\frac{\rho_{m,0}}{R^{3}}+\frac{\rho_{E,0}}{R^{n}}\Bigr)-\frac{c^{2}}{R^{2}}.
\]
Factor $8\pi G\rho_{m,0}/(3R^{3})$ from the density terms:
\[
\Bigl(\frac{\dot R}{R}\Bigr)^{2}=\frac{8\pi G\rho_{m,0}}{3R^{3}}\Bigl[1+\frac{\rho_{E,0}}{\rho_{m,0}}R^{3-n}\Bigr]-\frac{c^{2}}{R^{2}}.
\]
Rewrite the curvature term using $-c^{2}/R^{2}=-(8\pi G\rho_{m,0}/(3R^{3}))\cdot R\cdot(3c^{2}/(8\pi G\rho_{m,0}))$:
\[
-\frac{c^{2}}{R^{2}}=-\frac{8\pi G\rho_{m,0}}{3R^{3}}\cdot\frac{R}{R_{T}},\qquad R_{T}\equiv\frac{8\pi G\rho_{m,0}}{3c^{2}}.
\]
Introduce $R_{E}$:
\[
R_{E}^{3-n}\equiv\frac{\rho_{m,0}}{\rho_{E,0}}.
\]
Combine into the bracket, writing $\rho_{0}\equiv\rho_{m,0}$:
\begin{equation}
\boxed{\;\Bigl(\frac{\dot R}{R}\Bigr)^{2}=\frac{8\pi G\rho_{0}}{3R^{3}}\Bigl[1-\frac{R}{R_{T}}+\Bigl(\frac{R}{R_{E}}\Bigr)^{3-n}\Bigr].\;}
\label{eq:Friedmann}
\end{equation}

For the acceleration equation, use the rubric's $\ddot R/R$ with $\Lambda=0$:
\[
\frac{\ddot R}{R}=-\frac{4\pi G}{3}\bigl(\rho+3p/c^{2}\bigr).
\]
Sum $\rho+3p/c^{2}$ with $p_{m}=0$ and $p_{E}=(n-3)\rho_{E}c^{2}/3$:
\[
\rho+3p/c^{2}=\frac{\rho_{m,0}}{R^{3}}+\rho_{E,0}R^{-n}+(n-3)\rho_{E,0}R^{-n}.
\]
Collect the dark-energy terms:
\[
\rho+3p/c^{2}=\frac{\rho_{m,0}}{R^{3}}+(n-2)\rho_{E,0}R^{-n}.
\]
Factor $\rho_{m,0}/R^{3}$:
\[
\rho+3p/c^{2}=\frac{\rho_{m,0}}{R^{3}}\Bigl[1+(n-2)\frac{\rho_{E,0}}{\rho_{m,0}}R^{3-n}\Bigr]=\frac{\rho_{0}}{R^{3}}\Bigl[1+(n-2)\Bigl(\frac{R}{R_{E}}\Bigr)^{3-n}\Bigr].
\]
Hence
\begin{equation}
\boxed{\;\frac{\ddot R}{R}=-\frac{4\pi G\rho_{0}}{3R^{3}}\Bigl[1+(n-2)\Bigl(\frac{R}{R_{E}}\Bigr)^{3-n}\Bigr].\;}
\label{eq:Raccel}
\end{equation}

\subsection*{(c) $n=0$: Einstein's static Universe}

With $n=0$ the dark energy is a cosmological constant.  Impose the static condition $\dot R=\ddot R=0$.

Set $\ddot R=0$ in \eqref{eq:Raccel} with $n=0$ and $n-2=-2$:
\[
1-2\Bigl(\frac{R_{S}}{R_{E}}\Bigr)^{3}=0.
\]
Solve:
\[
R_{S}^{3}=\tfrac{1}{2}R_{E}^{3}.
\]
Set $\dot R=0$ in \eqref{eq:Friedmann} with $n=0$:
\[
1-\frac{R_{S}}{R_{T}}+\Bigl(\frac{R_{S}}{R_{E}}\Bigr)^{3}=0.
\]
Substitute $(R_{S}/R_{E})^{3}=\tfrac{1}{2}$:
\[
1-\frac{R_{S}}{R_{T}}+\tfrac{1}{2}=0.
\]
Solve for $R_{S}$:
\[
R_{S}=\tfrac{3}{2}R_{T}.
\]
Box the static parameters:
\[
\boxed{\;R_{S}=\tfrac{3}{2}R_{T},\qquad R_{E}=2^{1/3}R_{S}=\tfrac{3\cdot 2^{1/3}}{2}R_{T}.\;}
\]

Multiply \eqref{eq:Friedmann} by $R^{2}/2$:
\[
\tfrac{1}{2}\dot R^{2}=\frac{4\pi G\rho_{0}}{3R}\Bigl[1-\frac{R}{R_{T}}+\Bigl(\frac{R}{R_{E}}\Bigr)^{3}\Bigr].
\]
Expand and rearrange to $\tfrac{1}{2}\dot R^{2}+V_{\rm eff}(R)=E$ form, with $E$ absorbing the constant piece (an additive constant in $V_{\rm eff}$ is physically irrelevant):
\[
V_{\rm eff}(R)=-\frac{4\pi G\rho_{0}}{3R}-\frac{4\pi G\rho_{0}}{3R_{E}^{3}}R^{2}.
\]
Differentiate once:
\[
V'_{\rm eff}=\frac{4\pi G\rho_{0}}{3R^{2}}-\frac{8\pi G\rho_{0}}{3R_{E}^{3}}R.
\]
Set $V'_{\rm eff}=0$:
\[
\frac{1}{R^{2}}=\frac{2R}{R_{E}^{3}}\;\Rightarrow\; R^{3}=\tfrac{1}{2}R_{E}^{3},
\]
confirming the static $R_{S}^{3}=R_{E}^{3}/2$.

Differentiate again:
\[
V''_{\rm eff}=-\frac{8\pi G\rho_{0}}{3R^{3}}-\frac{8\pi G\rho_{0}}{3R_{E}^{3}}.
\]
Both terms are negative:
\[
V''_{\rm eff}(R_{S})<0.
\]
Hence $R_{S}$ is a \emph{maximum} of $V_{\rm eff}$ — Einstein's static Universe is unstable.  Evolution with $\dot R=0$ at $R\neq R_{S}$:
\begin{itemize}
\item $R<R_{S}$: matter wins ($\rho_{m}\propto R^{-3}$ dominant), $\ddot R<0$, the Universe recollapses to a Big Crunch.
\item $R>R_{S}$: cosmological constant wins, $\ddot R>0$, the Universe accelerates to de Sitter expansion.
\end{itemize}

\subsection*{(d) Loitering Universe ($n=2-\epsilon$, $0<\epsilon\ll 1$)}

Set $\ddot R=0$ in \eqref{eq:Raccel} with $n-2=-\epsilon$ and $3-n=1+\epsilon$:
\[
1-\epsilon\Bigl(\frac{R_{S}}{R_{E}}\Bigr)^{1+\epsilon}=0.
\]
Solve for the ratio:
\[
\Bigl(\frac{R_{S}}{R_{E}}\Bigr)^{1+\epsilon}=\frac{1}{\epsilon}.
\]
Take the $1/(1+\epsilon)$ root:
\begin{equation}
\boxed{\;R_{S}=R_{E}\,\epsilon^{-1/(1+\epsilon)}.\;}
\end{equation}
Set $\dot R=0$ in \eqref{eq:Friedmann}:
\[
1-\frac{R_{S}}{R_{T}}+\Bigl(\frac{R_{S}}{R_{E}}\Bigr)^{1+\epsilon}=0.
\]
Substitute $(R_{S}/R_{E})^{1+\epsilon}=1/\epsilon$:
\[
\frac{R_{S}}{R_{T}}=1+\frac{1}{\epsilon}=\frac{1+\epsilon}{\epsilon}.
\]
Invert:
\[
R_{T}=\frac{\epsilon\,R_{S}}{1+\epsilon}.
\]

Write the effective potential from \eqref{eq:Friedmann} (same form as part (c) with $3-n=1+\epsilon$):
\[
V_{\rm eff}(R)=-\frac{4\pi G\rho_{0}}{3R}\Bigl[1+\Bigl(\frac{R}{R_{E}}\Bigr)^{1+\epsilon}\Bigr].
\]
$V_{\rm eff}$ has a maximum at $R_{S}$ as in part (c).  A universe with kinetic energy that just suffices to reach $R_{S}$ (``just overshoot''):
\begin{itemize}
\item Approaches $R_{S}$ from below with $\dot R\to 0^{+}$.
\item Spends an arbitrarily long time near $R_{S}$ — the \emph{loitering} phase.
\item Eventually rolls past $R_{S}$ and accelerates to $R\to\infty$.
\end{itemize}

\subsection*{(e) Time spent in the loitering phase}

Taylor-expand $V_{\rm eff}$ to second order around $R_{S}$ using $V'_{\rm eff}(R_{S})=0$:
\[
V_{\rm eff}(R)\approx V_{\rm eff}(R_{S})+\tfrac{1}{2}V''_{\rm eff}(R_{S})\,\delta R^{2},\qquad \delta R\equiv R-R_{S}.
\]
Write the effective potential for $n=2-\epsilon$:
\[
V_{\rm eff}=-\frac{4\pi G\rho_{0}}{3R}-\frac{4\pi G\rho_{0}}{3R_{E}^{1+\epsilon}}R^{\epsilon}.
\]
Differentiate once:
\[
V'_{\rm eff}=\frac{4\pi G\rho_{0}}{3R^{2}}-\frac{4\pi G\rho_{0}\,\epsilon}{3R_{E}^{1+\epsilon}}R^{\epsilon-1}.
\]
Differentiate again:
\[
V''_{\rm eff}=-\frac{8\pi G\rho_{0}}{3R^{3}}-\frac{4\pi G\rho_{0}\,\epsilon(\epsilon-1)}{3R_{E}^{1+\epsilon}}R^{\epsilon-2}.
\]
Use $R_{E}^{1+\epsilon}=\epsilon R_{S}^{1+\epsilon}$ to substitute at $R=R_{S}$:
\[
V''_{\rm eff}(R_{S})=-\frac{8\pi G\rho_{0}}{3R_{S}^{3}}-\frac{4\pi G\rho_{0}\,\epsilon(\epsilon-1)}{3\epsilon R_{S}^{1+\epsilon}}\cdot R_{S}^{\epsilon-2}.
\]
Cancel $\epsilon$ and combine powers $R_{S}^{\epsilon-2-1-\epsilon}=R_{S}^{-3}$:
\[
V''_{\rm eff}(R_{S})=-\frac{8\pi G\rho_{0}}{3R_{S}^{3}}-\frac{4\pi G\rho_{0}(\epsilon-1)}{3R_{S}^{3}}.
\]
Factor $-4\pi G\rho_{0}/(3R_{S}^{3})$:
\[
V''_{\rm eff}(R_{S})=-\frac{4\pi G\rho_{0}}{3R_{S}^{3}}\bigl[2+(\epsilon-1)\bigr]=-\frac{4\pi G\rho_{0}(1+\epsilon)}{3R_{S}^{3}}.
\]
Write the inverted-oscillator frequency:
\[
\omega^{2}\equiv -V''_{\rm eff}(R_{S})=\frac{4\pi G\rho_{0}(1+\epsilon)}{3R_{S}^{3}}.
\]
Use energy conservation with $E=V_{\rm eff}(R_{S})+\Delta E$, $\Delta E>0$ small (``just overshoot''):
\[
\tfrac{1}{2}\dot R^{2}=\Delta E-\tfrac{1}{2}V''_{\rm eff}(R_{S})\delta R^{2}=\Delta E+\tfrac{1}{2}\omega^{2}\delta R^{2}.
\]
Solve for $\dot R=\de(\delta R)/\de t$:
\[
\dot R=\sqrt{2\Delta E+\omega^{2}\delta R^{2}}.
\]
Integrate from $\delta R=0$ ($R=R_{S}$) to $\delta R=R_{S}$ ($R=2R_{S}$):
\[
t=\int_{0}^{R_{S}}\frac{\de(\delta R)}{\sqrt{2\Delta E+\omega^{2}\delta R^{2}}}=\frac{1}{\omega}\sinh^{-1}\!\Bigl(\frac{\omega R_{S}}{\sqrt{2\Delta E}}\Bigr).
\]
For $\omega R_{S}\gg\sqrt{2\Delta E}$ the inverse sinh behaves logarithmically:
\[
t\sim \frac{1}{\omega}\ln\!\Bigl(\frac{2\omega R_{S}}{\sqrt{2\Delta E}}\Bigr).
\]
Compute $R_{S}$-scaling using $R_{S}=R_{E}\epsilon^{-1/(1+\epsilon)}$:
\[
R_{S}^{3}\propto \epsilon^{-3/(1+\epsilon)}\;\Rightarrow\; \omega^{2}\propto R_{S}^{-3}\propto \epsilon^{3/(1+\epsilon)}.
\]
Therefore $\omega\propto\epsilon^{3/(2(1+\epsilon))}\to 0$ as $\epsilon\to 0$.  Hence the expansion time diverges:
\begin{equation}
\boxed{\;t(R_{S}\to 2R_{S})\sim \frac{1}{\omega}\propto R_{S}^{3/2}\propto \epsilon^{-3/(2(1+\epsilon))}\to\infty\;\text{as }\epsilon\to 0.\;}
\label{eq:tdiv}
\end{equation}
The Universe loiters near the inverted-oscillator inflection for a time that diverges as $\epsilon\to 0$.
