\section*{Question 1 --- Spin--orbit interaction and fine structure}
%==================================================================

\textbf{Problem.} One-electron atom: derive $H_{\rm SO}$ (Thomas factor via $g_s\to g_s-1$), show the $l\neq 0$ splitting $\Delta E=Z^4\alpha^2 hcR_\infty/[n^3 l(l+1)]$, then apply to H ($2\to 3$) vs.\ He$^+$ ($4\to 6$): fine structure, mass shift, E1 selection rules, Doppler width, and the $Z$ giving fine structure equal to the $2\to 3$ energy of H.

\subsection*{(a) Origin and derivation of $H_{\rm SO}$}

In the electron rest frame the orbiting nucleus produces a magnetic field that couples to the electron spin moment $\bm\mu_s=-g_s\mu_{\rm B}\bm S/\hbar$.

Lorentz boost of the nuclear $\bm E=-\nabla\phi$ to the electron frame, $\order{v^2/c^2}$:
\[
\bm B'=-\frac{1}{c^{2}}\,\bm v\times\bm E.
\]
Central potential, $\bm E=-\hat{\bm r}\,\partial_r\phi$:
\[
\bm v\times\bm E=-\partial_r\phi\,(\bm v\times\hat{\bm r}).
\]
Use $\hat{\bm r}=\bm r/r$:
\[
\bm v\times\bm E=-\frac{1}{r}\,\partial_r\phi\,(\bm v\times\bm r).
\]
Identify $\bm L=m_e\bm r\times\bm v=-m_e\bm v\times\bm r$:
\[
\bm v\times\bm r=-\frac{\bm L}{m_e}.
\]
Insert in $\bm B'$:
\[
\bm B'=-\frac{1}{m_e c^{2}}\,\frac{1}{r}\,\partial_r\phi\,\bm L.
\]
Zeeman energy of spin moment in $\bm B'$:
\[
H'=-\bm\mu_s\!\cdot\!\bm B'=\frac{g_s\mu_{\rm B}}{\hbar}\,\bm S\!\cdot\!\bm B'.
\]
Substitute $\mu_{\rm B}=e\hbar/(2m_e)$:
\[
H'=\frac{g_s\,e}{2 m_e}\,\bm S\!\cdot\!\bm B'.
\]
Substitute $\bm B'$:
\[
H'=-\frac{g_s\,e}{2 m_e^{2}c^{2}}\,\frac{1}{r}\,\partial_r\phi\,\bm L\!\cdot\!\bm S.
\]
Apply Thomas correction $g_s\to g_s-1\approx 1$ and write $U=-e\phi$ so $-e\,\partial_r\phi=\partial_r U$:
\begin{equation}
\boxed{\;H_{\rm SO}=\frac{1}{2 m_e^{2}c^{2}}\,\frac{1}{r}\,\frac{\partial U}{\partial r}\,\bm L\!\cdot\!\bm S.\;}
\label{eq:HSO-general}
\end{equation}

Hydrogenic $U=-Ze^{2}/(4\pi\epsilon_0 r)$, so $\partial_r U=Ze^{2}/(4\pi\epsilon_0 r^{2})$:
\begin{equation}
H_{\rm SO}=\frac{Ze^{2}}{8\pi\epsilon_0\,m_e^{2}c^{2}\,r^{3}}\,\bm L\!\cdot\!\bm S\equiv\xi(r)\,\bm L\!\cdot\!\bm S.
\label{eq:HSO-H}
\end{equation}

\subsection*{(b) Splitting of a configuration into two levels}

Coupled basis $\bm J=\bm L+\bm S$:
\[
\bm L\!\cdot\!\bm S=\tfrac12(\bm J^{2}-\bm L^{2}-\bm S^{2}).
\]
Take eigenvalues with $s=1/2$, $l\neq 0$, $j=l\pm 1/2$:
\[
\Braket{\bm L\!\cdot\!\bm S}_{j}=\frac{\hbar^{2}}{2}\,[j(j+1)-l(l+1)-s(s+1)].
\]
Evaluate $j=l+1/2$, $j(j+1)=l^{2}+2l+3/4$:
\[
\Braket{\bm L\!\cdot\!\bm S}_{j=l+1/2}=\frac{\hbar^{2}}{2}\,l.
\]
Evaluate $j=l-1/2$, $j(j+1)=l^{2}-1/4$:
\[
\Braket{\bm L\!\cdot\!\bm S}_{j=l-1/2}=-\frac{\hbar^{2}}{2}(l+1).
\]
Take expectation value $\Delta E_j=\Braket{\xi(r)}\Braket{\bm L\!\cdot\!\bm S}_j$ and subtract:
\[
\Delta E=\Braket{\xi(r)}\left[\frac{\hbar^{2}}{2}\,l+\frac{\hbar^{2}}{2}(l+1)\right].
\]
Collect:
\[
\Delta E=\Braket{\xi(r)}\,\tfrac{\hbar^{2}}{2}(2l+1).
\]
Insert $\Braket{\xi(r)}=\dfrac{Ze^{2}}{8\pi\epsilon_0 m_e^{2}c^{2}}\Braket{1/r^{3}}$ with hydrogenic $\Braket{1/r^3}=2(Z/na_0)^3/[l(l+1)(2l+1)]$:
\[
\Delta E=\frac{Ze^{2}}{8\pi\epsilon_0 m_e^{2}c^{2}}\cdot\frac{2 Z^{3}}{n^{3}a_0^{3}\,l(l+1)(2l+1)}\cdot\frac{\hbar^{2}(2l+1)}{2}.
\]
Cancel the $(2l+1)$ factors:
\[
\Delta E=\frac{Z^{4} e^{2}\hbar^{2}}{8\pi\epsilon_0\,m_e^{2}c^{2}\,n^{3}a_0^{3}\,l(l+1)}.
\]
Apply $e^2/(4\pi\epsilon_0)=\alpha\hbar c$ so $e^2/(8\pi\epsilon_0)=\alpha\hbar c/2$:
\[
\Delta E=\frac{Z^{4}\alpha\hbar^{3}c}{2 m_e^{2}c^{2}\,n^{3}a_0^{3}\,l(l+1)}.
\]
Apply $a_0=\hbar/(\alpha m_e c)$, so $1/a_0^{3}=\alpha^{3}m_e^{3}c^{3}/\hbar^{3}$:
\[
\Delta E=\frac{Z^{4}\alpha^{4} m_e c^{2}}{2 n^{3}\,l(l+1)}.
\]
Apply $hcR_\infty=\tfrac12 m_e c^{2}\alpha^{2}$, so $\alpha^{4}m_e c^{2}=2\alpha^{2}hcR_\infty$:
\begin{equation}
\boxed{\;\Delta E=\frac{Z^{4}\alpha^{2}hcR_{\infty}}{n^{3}\,l(l+1)}\quad(l\neq 0).\;}
\label{eq:DeltaE-final}
\end{equation}

\subsection*{(c) H ($2\to 3$) vs.\ He$^+$ ($4\to 6$)}

Bohr energy $E_{n\to n'}=hcR_\infty Z^2(1/n^2-1/n'^2)$.

H ($Z=1$, $2\to 3$):
\[
\Delta E^{\rm H}=hcR_\infty\!\left(\frac{1}{4}-\frac{1}{9}\right)=\frac{9-4}{36}hcR_\infty=\frac{5}{36}hcR_\infty.
\]
He$^+$ ($Z=2$, $4\to 6$):
\[
\Delta E^{\rm He^+}=4 hcR_\infty\!\left(\frac{1}{16}-\frac{1}{36}\right)=4hcR_\infty\frac{36-16}{16\cdot 36}=\frac{5}{36}hcR_\infty.
\]
Same Bohr wavelength $\lambda\simeq\SI{656}{nm}$ (H$_\alpha$), $\nu_0\simeq\SI{4.57e14}{Hz}$.

Largest fine-structure splitting is at smallest $l\neq 0$, i.e.\ $l=1$. Apply \eqref{eq:DeltaE-final} for H, $Z=1, n=2, l=1$:
\[
\Delta E^{\rm H}_{n=2,l=1}=\frac{1^{4}\cdot\alpha^{2}}{2^{3}\cdot 1\cdot 2}\,hcR_\infty.
\]
Reduce:
\[
\Delta E^{\rm H}_{n=2,l=1}=\frac{\alpha^{2}}{16}\,hcR_\infty.
\]
Apply \eqref{eq:DeltaE-final} for He$^+$, $Z=2, n=4, l=1$:
\[
\Delta E^{\rm He^+}_{n=4,l=1}=\frac{2^{4}\,\alpha^{2}}{4^{3}\cdot 1\cdot 2}\,hcR_\infty=\frac{16}{128}\alpha^{2}hcR_\infty.
\]
Reduce:
\[
\Delta E^{\rm He^+}_{n=4,l=1}=\frac{\alpha^{2}}{8}\,hcR_\infty.
\]
Divide H splitting by $h\nu_0=\tfrac{5}{36}hcR_\infty$:
\[
\frac{\Delta\nu^{\rm H}}{\nu_0}=\frac{\alpha^{2}/16}{5/36}=\frac{36\alpha^{2}}{16\cdot 5}=\frac{36\alpha^{2}}{80}.
\]
Reduce:
\[
\frac{\Delta\nu^{\rm H}}{\nu_0}=\frac{9\alpha^{2}}{20}.
\]
Insert $\alpha^{2}\simeq 5.33\times 10^{-5}$:
\[
\frac{\Delta\nu^{\rm H}}{\nu_0}\simeq\frac{9\cdot 5.33\times 10^{-5}}{20}\simeq 2.4\times 10^{-5}.
\]
Divide He$^+$ splitting similarly:
\[
\frac{\Delta\nu^{\rm He^+}}{\nu_0}=\frac{\alpha^{2}/8}{5/36}=\frac{36\alpha^{2}}{40}=\frac{9\alpha^{2}}{10}.
\]
Insert $\alpha^{2}$:
\[
\frac{\Delta\nu^{\rm He^+}}{\nu_0}\simeq\frac{9\cdot 5.33\times 10^{-5}}{10}\simeq 4.8\times 10^{-5}.
\]
Collect results:
\begin{equation}
\boxed{\;\frac{\Delta\nu^{\rm H}}{\nu_0}=\frac{9\alpha^{2}}{20}\simeq 2.4\times 10^{-5},\qquad
\frac{\Delta\nu^{\rm He^+}}{\nu_0}=\frac{9\alpha^{2}}{10}\simeq 4.8\times 10^{-5}.\;}
\label{eq:fs-fractions}
\end{equation}

\textit{Mass shift.} Reduced-mass Rydberg $R_M=R_\infty/(1+m_e/M)$ shifts the transition frequency:
\[
\frac{\nu(M)}{\nu_0}=\frac{R_M}{R_\infty}=\frac{1}{1+m_e/M}.
\]
Expand to leading order in $m_e/M$:
\[
\frac{\nu(M)}{\nu_0}\simeq 1-\frac{m_e}{M}.
\]
Take the difference between H ($M=m_p$) and He$^+$ ($M=M_\alpha\simeq 4m_p$):
\[
\frac{\Delta\nu_{\rm mass}}{\nu_0}=\frac{\nu(m_p)-\nu(4m_p)}{\nu_0}\simeq\frac{m_e}{m_p}-\frac{m_e}{4m_p}.
\]
Collect:
\[
\frac{\Delta\nu_{\rm mass}}{\nu_0}\simeq\frac{3}{4}\cdot\frac{m_e}{m_p}.
\]
Insert $m_e/m_p\simeq 1/1836$:
\[
\frac{\Delta\nu_{\rm mass}}{\nu_0}\simeq\frac{3}{4}\cdot\frac{1}{1836}\simeq 4\times 10^{-4},
\]
an order of magnitude larger than the fine-structure splittings; mass shift dominates the wavelength difference.

E1 selection rules:
\begin{equation}
\Delta l=\pm 1,\quad\Delta s=0,\quad\Delta j=0,\pm 1\;(0\!\not\to\!0),\quad\Delta m_j=0,\pm 1.
\label{eq:E1-rules}
\end{equation}

\textit{Justification of $\Delta l=\pm 1$.} $\hat{\bm d}=-e\hat{\bm r}$ is a rank-1 odd-parity tensor. Parity selection forces $l-l'$ odd; Wigner--Eckart forces $|l-l'|\le 1$. Together: $\Delta l=\pm 1$.

\subsection*{(e) Doppler broadening at $T=\SI{300}{K}$}

Thermal Doppler FWHM:
\[
\frac{\Delta\nu_{\rm D}}{\nu_0}=\sqrt{\frac{8 k_{\rm B} T \ln 2}{M c^{2}}}.
\]
Insert $k_{\rm B}T\simeq\SI{4.14e-21}{J}$, $M=m_p$, $c^2\simeq\SI{9e16}{m^2 s^{-2}}$:
\[
\frac{\Delta\nu_{\rm D}^{\rm H}}{\nu_0}=\sqrt{\frac{8\cdot 4.14\!\times\!10^{-21}\!\cdot\!0.693}{1.67\!\times\!10^{-27}\cdot 9\!\times\!10^{16}}}.
\]
Evaluate the radicand:
\[
\frac{\Delta\nu_{\rm D}^{\rm H}}{\nu_0}=\sqrt{\frac{2.296\times 10^{-20}}{1.503\times 10^{-10}}}=\sqrt{1.527\times 10^{-10}}.
\]
Take the square root:
\[
\frac{\Delta\nu_{\rm D}^{\rm H}}{\nu_0}\simeq 1.2\times 10^{-5}.
\]
He$^+$ has $M\simeq 4 m_p$, so width scales by $1/\sqrt{M/m_p}=1/\sqrt{4}=1/2$:
\[
\frac{\Delta\nu_{\rm D}^{\rm He^+}}{\nu_0}=\frac{1}{2}\cdot 1.2\times 10^{-5}.
\]
Collect both results:
\begin{equation}
\boxed{\;\frac{\Delta\nu_{\rm D}^{\rm H}}{\nu_0}\simeq 1.2\times 10^{-5},\qquad
\frac{\Delta\nu_{\rm D}^{\rm He^+}}{\nu_0}\simeq 6\times 10^{-6}.\;}
\label{eq:doppRes}
\end{equation}

\textit{Further mechanism in a discharge lamp.} Pressure (collisional) broadening: random collisional dephasing of the radiating dipole, Lorentzian width $\sim 1/\tau_{\rm coll}$. (Stark broadening from plasma microfields is also strong for H because of its linear Stark effect.)

\subsection*{(f) Fine structure equal to the H $2\to 3$ energy}

Equate $n=2,l=1$ splitting to $\tfrac{5}{36}hcR_\infty$:
\[
\frac{Z^{4}\alpha^{2}}{16}\,hcR_\infty=\frac{5}{36}\,hcR_\infty.
\]
Cancel $hcR_\infty$ and rearrange for $Z^4$:
\[
Z^{4}=\frac{5\cdot 16}{36\,\alpha^{2}}=\frac{80}{36\,\alpha^{2}}=\frac{20}{9\alpha^{2}}.
\]
Insert $\alpha^{2}\simeq 5.33\times 10^{-5}$:
\[
Z^{4}\simeq\frac{20}{9\cdot 5.33\times 10^{-5}}=\frac{20}{4.80\times 10^{-4}}\simeq 4.17\times 10^{4}.
\]
Take square root:
\[
Z^{2}\simeq\sqrt{4.17\times 10^{4}}\simeq 204.
\]
Take square root again:
\[
Z\simeq\sqrt{204}\simeq 14.3.
\]
\begin{equation}
\boxed{\;Z\simeq 14\;\;(\text{e.g.\ Si}^{13+}).\;}
\label{eq:Z-eq}
\end{equation}

\textit{Alkali $Z$-scaling.} Valence $np$ electron penetrates the unscreened core; SO coupling samples the bare nuclear field. Empirical $\Delta E_{\rm FS}$: Na \SI{0.42}{meV}, K \SI{2.0}{meV}, Rb \SI{6.96}{meV}, Cs \SI{28}{meV} — super-quartic in $Z$ because both the inner field and the penetrating-wavefunction amplitude grow with $Z$.

\section*{Question 2 --- Hyperfine structure and the hydrogen MASER}
%==================================================================

\textbf{Problem.} Hamiltonian $\mathcal H=A\,\bm I\!\cdot\!\bm J+g_J\mu_{\rm B}\bm J\!\cdot\!\bm B-g_I\mu_N\bm I\!\cdot\!\bm B$. Identify each term, find the weak/strong-field crossover for ground-state H ($A/h=\SI{1420}{MHz}$, $I=1/2$), derive the eigenenergies in both regimes, and analyse a Stern--Gerlach beam with $B_z=\SI{1}{T}$, $\partial_z B_z=\SI{20}{T/m}$ as the basis of an H maser.

\subsection*{(a) Origin of each term}

$A\bm I\!\cdot\!\bm J$: hyperfine coupling of nuclear moment $\bm\mu_I=g_I\mu_N\bm I/\hbar$ to the electron-generated magnetic field at the nucleus (dominated by the Fermi-contact term for $s$ states).

$g_J\mu_{\rm B}\bm J\!\cdot\!\bm B$: electronic Zeeman, energy of $\bm\mu_J=-g_J\mu_{\rm B}\bm J/\hbar$ in $\bm B$.

$-g_I\mu_N\bm I\!\cdot\!\bm B$: nuclear Zeeman, smaller by $m_e/m_p\sim 1/1836$.

\textit{Regimes.} Weak field $g_J\mu_{\rm B}B\ll A$: $\bm I,\bm J$ couple to $\bm F=\bm I+\bm J$; good QNs $|F,m_F\rangle$. Strong field $g_J\mu_{\rm B}B\gg A$: $\bm I,\bm J$ decouple; good QNs $|m_J,m_I\rangle$ (Paschen--Back of hyperfine).

\subsection*{(b) Cross-over field}

Set the Zeeman shift equal to the hyperfine splitting $g_J\mu_{\rm B}B_c\simeq A$.

Compute $A$ from $\nu_{\rm HF}=\SI{1420}{MHz}$:
\[
A=h\,\nu_{\rm HF}=\SI{6.63e-34}{J\,s}\cdot\SI{1.42e9}{Hz}\simeq\SI{9.41e-25}{J}.
\]
With $g_J\simeq 2$ for the $1\,^2S_{1/2}$ ground state, solve for $B_c$:
\[
B_c=\frac{A}{g_J\mu_{\rm B}}=\frac{\SI{9.41e-25}{J}}{2\cdot\SI{9.27e-24}{J/T}}.
\]
Evaluate:
\begin{equation}
\boxed{\;B_c\simeq\SI{50}{mT}\;(\sim\!500\,\mathrm{G}).\;}
\label{eq:Bc-est}
\end{equation}

\subsection*{(c) Weak-field eigenenergies}

Take $\bm I,\bm J$ dimensionless (in units of $\hbar$). Define $\bm F=\bm I+\bm J$ and square:
\[
\bm F^{2}=\bm I^{2}+\bm J^{2}+2\bm I\!\cdot\!\bm J.
\]
Solve for $\bm I\cdot\bm J$:
\[
\bm I\!\cdot\!\bm J=\tfrac{1}{2}[F(F+1)-I(I+1)-J(J+1)].
\]
Insert $I=J=1/2$, $F=1$:
\[
\Braket{\bm I\!\cdot\!\bm J}_{F=1}=\tfrac{1}{2}[2-\tfrac34-\tfrac34]=\tfrac14.
\]
Insert $F=0$:
\[
\Braket{\bm I\!\cdot\!\bm J}_{F=0}=\tfrac{1}{2}[0-\tfrac34-\tfrac34]=-\tfrac34.
\]
Multiply by $A$:
\[
E_{F=1}^{(0)}=+\tfrac14 A,\qquad E_{F=0}^{(0)}=-\tfrac34 A.
\]
Hyperfine splitting $E_{F=1}-E_{F=0}=A=h\nu_{\rm HF}$ identifies $A$.

Apply projection theorem within fixed $F$ to the Zeeman terms:
\[
\Braket{F m_F|J_z|F m_F}=\frac{F(F+1)+J(J+1)-I(I+1)}{2F(F+1)}\,m_F.
\]
Read off the Land\'e $g_F$:
\[
g_F=g_J\,\frac{F(F+1)+J(J+1)-I(I+1)}{2F(F+1)}-\frac{m_e}{m_p}\,g_I\,\frac{F(F+1)+I(I+1)-J(J+1)}{2F(F+1)}.
\]
Combine zero-field and Zeeman:
\begin{equation}
\boxed{\;E(F,m_F)=A\,\tfrac{F(F+1)-I(I+1)-J(J+1)}{2}+g_F\mu_{\rm B}B\,m_F.\;}
\label{eq:weak-field-E}
\end{equation}
For ground-state H: $g_{F=1}\simeq +1$, $g_{F=0}=0$ to leading order in $m_e/m_p$. Triplet splits linearly with $B$; singlet unshifted at first order.

\subsection*{(d) Strong-field eigenenergies}

In $|m_J,m_I\rangle$ Zeeman is diagonal. Decompose:
\[
\bm I\!\cdot\!\bm J=I_z J_z+\tfrac12(I_+J_-+I_-J_+).
\]
Drop ladder pieces (off-diagonal in the strong-field basis):
\[
\Braket{m_J,m_I|\bm I\!\cdot\!\bm J|m_J,m_I}=m_J m_I.
\]
Add the Zeeman terms:
\[
E(m_J,m_I)=A\,m_J m_I+g_J\mu_{\rm B}B\,m_J-g_I\mu_N B\,m_I.
\]
Insert the four sign combinations:
\begin{align}
E_1=E(+\tfrac12,+\tfrac12)&=+\tfrac14 A+\tfrac12 g_J\mu_{\rm B}B-\tfrac12 g_I\mu_N B,\label{eq:strong1}\\
E_2=E(+\tfrac12,-\tfrac12)&=-\tfrac14 A+\tfrac12 g_J\mu_{\rm B}B+\tfrac12 g_I\mu_N B,\label{eq:strong2}\\
E_3=E(-\tfrac12,+\tfrac12)&=-\tfrac14 A-\tfrac12 g_J\mu_{\rm B}B-\tfrac12 g_I\mu_N B,\label{eq:strong3}\\
E_4=E(-\tfrac12,-\tfrac12)&=+\tfrac14 A-\tfrac12 g_J\mu_{\rm B}B+\tfrac12 g_I\mu_N B.\label{eq:strong4}
\end{align}
Stretched states $|\!\uparrow\Uparrow\rangle,|\!\downarrow\Downarrow\rangle$ ($|m_F|=1$) are exact eigenstates at all $B$; the $m_F=0$ pair is given exactly by Breit--Rabi.

\subsection*{(e) Forces and the H maser}

High-field upper states $E_1,E_2$ have $m_J=+1/2$; magnetic moment $\mu_z\simeq -\mu_{\rm B}$ (electron dominates). Differentiate $E(m_J,m_I)$ w.r.t.\ $B$:
\[
\frac{\partial E}{\partial B}=g_J\mu_{\rm B}m_J-g_I\mu_N m_I.
\]
Apply $F_z=-(\partial E/\partial z)=-(\partial E/\partial B)\,\partial_z B_z$:
\[
F_z=-g_J\mu_{\rm B}m_J\,\frac{\partial B_z}{\partial z}+g_I\mu_N m_I\,\frac{\partial B_z}{\partial z}.
\]
Take leading electronic piece, $g_J m_J=+1$ at $m_J=+1/2$:
\[
F_z\simeq-\mu_{\rm B}\,\frac{\partial B_z}{\partial z}.
\]
Divide by $m_p$ to get acceleration:
\[
a\simeq\frac{\mu_{\rm B}}{m_p}\,\frac{\partial B_z}{\partial z}=\frac{\SI{9.27e-24}{J/T}}{\SI{1.67e-27}{kg}}\cdot\SI{20}{T/m}.
\]
Compute the ratio:
\[
\frac{\mu_{\rm B}}{m_p}\simeq 5.55\times 10^{3}\;\si{m\,s^{-2}/(T/m)}.
\]
Multiply by the gradient:
\[
a\simeq 5.55\times 10^{3}\cdot 20\;\si{m\,s^{-2}}.
\]
Collect:
\begin{equation}
\boxed{\;|a|\simeq\SI{1.1e5}{m\,s^{-2}}\;\text{for both upper states.}\;}
\label{eq:accel}
\end{equation}

Force difference between $E_1$ ($m_I=+1/2$) and $E_2$ ($m_I=-1/2$):
\[
\Delta F=F_z(E_1)-F_z(E_2)=g_I\mu_N\,\frac{\partial B_z}{\partial z}.
\]
Divide by the mean $\langle F\rangle\simeq g_J\mu_{\rm B}m_J\,\partial_z B_z$:
\[
\frac{\Delta F}{\langle F\rangle}\simeq\frac{g_I\mu_N}{g_J\mu_{\rm B}m_J}\sim\frac{m_e}{m_p}\sim\frac{1}{1836}\sim 5\times 10^{-4}.
\]

\textit{Breit--Rabi connection.} Stretched states $|\!\uparrow\Uparrow\rangle,|\!\downarrow\Downarrow\rangle$ correspond to $|F=1,m_F=\pm 1\rangle$. The $m_F=0$ pair: $E_2$ at high field connects to $|F=1,m_F=0\rangle$; $E_3$ connects to $|F=0,m_F=0\rangle$.

\section*{Question 3 --- Three- vs.\ four-level lasers and the Yb fibre}
%==================================================================

\textbf{Problem.} (a) Threshold contrast between three- and four-level systems. (b) Derive the gain coefficient from the Einstein relations. (c)--(g) Yb-doped fibre: $r=\SI{2}{\micro m}$, $L=\SI{0.5}{m}$, $\sigma_{\rm abs}=\SI{6e-25}{m^2}$, $\sigma_{\rm em}=\SI{2.5e-25}{m^2}$, $86\%$ pump absorption at \SI{910}{nm}, lower-band sublevels at $E_{1A}/hc=\SI{60000}{m^{-1}}$, $E_{1B}/hc=\SI{106000}{m^{-1}}$, upper level at $\SI{1026000}{m^{-1}}$.

\subsection*{(a) Threshold for three- vs.\ four-level lasers}

Inversion $N^*=N_2-(g_2/g_1)N_1$ must overcome cavity loss.

Four-level: $|1\rangle$ above the ground state, thermally empty ($N_1\sim e^{-\Delta E/k_{\rm B}T}\to 0$); inversion needs an infinitesimal fraction of ground population pumped.

Three-level: $|1\rangle$ is the ground state; need $N_2>N_1$, i.e.\ $>\!\tfrac12$ of all ions pumped. Threshold pump rate $\sim$ spontaneous-decay rate $\times\,N/2$, orders of magnitude larger.

\subsection*{(b) Gain coefficient}

A monochromatic beam of intensity $I$ has energy density $I/c$ concentrated near $\omega$. Stimulated rate per atom on a homogeneously broadened line:
\[
W_{\rm stim}=B_{21}\,\rho_\omega\,g(\omega-\omega_0)=B_{21}\,\frac{I}{c}\,g(\omega-\omega_0),
\]
with $\int g\,\de\omega=1$.

Net rate of photons per volume into the beam:
\[
\frac{1}{\hbar\omega_0}\,\frac{\de I}{\de z}=(B_{21}N_2-B_{12}N_1)\,\frac{I}{c}\,g(\omega-\omega_0).
\]
Multiply by $\hbar\omega_0$:
\[
\frac{\de I}{\de z}=\hbar\omega_0\,(B_{21}N_2-B_{12}N_1)\,\frac{I}{c}\,g(\omega-\omega_0).
\]
Assumptions: (i) plane wave; (ii) homogeneous broadening; (iii) populations not depleted; (iv) thermal photons negligible at optical frequency.

Define $\gamma(\omega)$ by $\de I/\de z=\gamma I$:
\[
\gamma(\omega)=\frac{\hbar\omega_0\,g(\omega-\omega_0)}{c}\,(B_{21}N_2-B_{12}N_1).
\]
Apply $g_1 B_{12}=g_2 B_{21}$:
\[
\gamma(\omega)=\frac{\hbar\omega_0\,g(\omega-\omega_0)}{c}\,B_{21}\!\left[N_2-\frac{g_2}{g_1}N_1\right].
\]
Eliminate $B_{21}$ via $A_{21}=\hbar\omega_0^{3}B_{21}/(\pi^{2}c^{3})$, i.e.\ $B_{21}=\pi^{2}c^{3}A_{21}/(\hbar\omega_0^{3})$:
\[
\gamma(\omega)=\frac{\hbar\omega_0\,g(\omega-\omega_0)}{c}\,\frac{\pi^{2}c^{3}A_{21}}{\hbar\omega_0^{3}}\!\left[N_2-\frac{g_2}{g_1}N_1\right].
\]
Cancel $\hbar\omega_0$ and one factor of $c$:
\begin{equation}
\boxed{\;\gamma(\omega)=\frac{\pi^{2}c^{2}A_{21}}{\omega_0^{2}}\,g(\omega-\omega_0)\!\left[N_2-\frac{g_2}{g_1}N_1\right]=\sigma_{\rm em}(\omega)\,N^{*}.\;}
\label{eq:gain-final}
\end{equation}

\subsection*{(c) Yb ion density}

Beer absorption with $86\%$ absorbed gives $I(L)/I(0)=0.14$:
\[
e^{-\sigma_{\rm abs}NL}=0.14.
\]
Take the log:
\[
\sigma_{\rm abs}\,N\,L=-\ln 0.14=1.97.
\]
Solve for $N$:
\[
N=\frac{1.97}{\sigma_{\rm abs} L}=\frac{1.97}{\SI{6e-25}{m^2}\cdot\SI{0.5}{m}}.
\]
Evaluate the denominator:
\[
N=\frac{1.97}{3\times 10^{-25}\,\si{m^{3}}}.
\]
Compute the quotient:
\begin{equation}
\boxed{\;N\simeq\SI{6.6e24}{m^{-3}}.\;}
\label{eq:N-yb}
\end{equation}

\subsection*{(d) Lower-band thermal populations at \SI{300}{K}}

Three lower-band sublevels $\{0,1A,1B\}$ in thermal equilibrium, all degeneracies $1$.

Convert $k_{\rm B}T$ to wavenumber with $k_{\rm B}T=\SI{4.14e-21}{J}$ and $hc=\SI{1.99e-25}{J\,m}$:
\[
\frac{k_{\rm B}T}{hc}=\frac{4.14\times 10^{-21}}{1.99\times 10^{-25}}\simeq\SI{2.084e4}{m^{-1}}.
\]
Form dimensionless exponents:
\[
\frac{E_{1A}}{k_{\rm B}T}=\frac{60\,000}{2.084\times 10^{4}}\simeq 2.88.
\]
\[
\frac{E_{1B}}{k_{\rm B}T}=\frac{106\,000}{2.084\times 10^{4}}\simeq 5.09.
\]
Form Boltzmann factors:
\[
e^{-2.88}\simeq 0.0561,\qquad e^{-5.09}\simeq 0.00614.
\]
Partition sum over $\{0,1A,1B\}$:
\[
Z_{\rm low}=1+0.0561+0.00614\simeq 1.062.
\]
Normalise to get fractional populations $f_\alpha=e^{-E_\alpha/k_{\rm B}T}/Z_{\rm low}$:
\[
f_{1A}=\frac{0.0561}{1.062}\simeq 5.3\times 10^{-2}.
\]
\[
f_{1B}=\frac{0.00614}{1.062}\simeq 5.8\times 10^{-3}.
\]
\begin{equation}
\boxed{\;f_{1A}\simeq 5.3\times 10^{-2},\qquad f_{1B}\simeq 5.8\times 10^{-3}.\;}
\label{eq:frac-low}
\end{equation}

\subsection*{(e) Threshold pump energies}

Threshold condition $N_{2,\rm th}=N_{1\alpha,\rm th}$ with $g_{2}/g_{1\alpha}=1$. The lower-band ions thermalise, so $N_{1\alpha}=f_{1\alpha}(N-N_2)\simeq f_{1\alpha}N$ for $f_{1\alpha}\ll 1$:
\[
N_{2,\rm th}\simeq f_{1\alpha}N.
\]
Compute pump-photon energy at \SI{910}{nm}:
\[
E_{\rm pump}=\frac{hc}{\lambda_{\rm pump}}=\frac{6.626\times 10^{-34}\cdot 3\times 10^{8}}{910\times 10^{-9}}.
\]
Evaluate:
\[
E_{\rm pump}\simeq\SI{2.18e-19}{J}\simeq\SI{1.36}{eV}.
\]
Core volume:
\[
V=\pi r^{2}L=\pi(\SI{2e-6}{m})^{2}\cdot\SI{0.5}{m}.
\]
Evaluate:
\[
V\simeq\SI{6.28e-12}{m^{3}}.
\]
Total ions in the core:
\[
NV\simeq\SI{6.6e24}{m^{-3}}\cdot\SI{6.28e-12}{m^{3}}\simeq 4.14\times 10^{13}.
\]
Threshold pump energy $E_{\alpha,\rm th}=N_{2,\rm th}V E_{\rm pump}=f_{1\alpha}NV E_{\rm pump}$:
\[
E_{A,\rm th}=5.3\times 10^{-2}\cdot 4.14\times 10^{13}\cdot\SI{2.18e-19}{J}.
\]
Multiply factors for $A$:
\[
E_{A,\rm th}\simeq\SI{4.78e-7}{J}.
\]
Compute $B$:
\[
E_{B,\rm th}=5.8\times 10^{-3}\cdot 4.14\times 10^{13}\cdot\SI{2.18e-19}{J}.
\]
Multiply factors for $B$:
\[
E_{B,\rm th}\simeq\SI{5.23e-8}{J}.
\]
\begin{equation}
\boxed{\;E_{A,\rm th}\simeq\SI{0.48}{\micro J},\qquad E_{B,\rm th}\simeq\SI{52}{nJ}.\;}
\label{eq:Eth-results}
\end{equation}

\subsection*{(f) Round-trip gain at $E=5E_{B,\rm th}$}

Pump $5\times$ threshold for $B$ excites $N_2=5 f_{1B}N$:
\[
N_2=5\cdot 5.8\times 10^{-3}\cdot\SI{6.6e24}{m^{-3}}.
\]
Multiply factors:
\[
N_2\simeq\SI{1.9e23}{m^{-3}}.
\]
Remaining ions in lower band:
\[
N-N_2=6.6\times 10^{24}-0.19\times 10^{24}\simeq\SI{6.41e24}{m^{-3}}.
\]
Thermal lower-band populations:
\[
N_{1A}=f_{1A}(N-N_2)\simeq 5.3\times 10^{-2}\cdot\SI{6.41e24}{m^{-3}}\simeq\SI{3.4e23}{m^{-3}}.
\]
\[
N_{1B}=f_{1B}(N-N_2)\simeq 5.8\times 10^{-3}\cdot\SI{6.41e24}{m^{-3}}\simeq\SI{3.7e22}{m^{-3}}.
\]
Inversions ($g_2/g_{1\alpha}=1$):
\[
N^{*}_{A}=N_2-N_{1A}=\SI{1.9e23}{m^{-3}}-\SI{3.4e23}{m^{-3}}\simeq -\SI{1.5e23}{m^{-3}}<0.
\]
\[
N^{*}_{B}=N_2-N_{1B}=\SI{1.9e23}{m^{-3}}-\SI{3.7e22}{m^{-3}}\simeq +\SI{1.5e23}{m^{-3}}.
\]
Round-trip gain on $2\to 1B$:
\[
G_{\rm rt}=\exp(2\sigma_{\rm em}N^{*}_{B}L)=\exp(2\cdot\SI{2.5e-25}{m^2}\cdot\SI{1.5e23}{m^{-3}}\cdot\SI{0.5}{m}).
\]
Multiply out the exponent:
\[
2\sigma_{\rm em}N^{*}_{B}L=2\cdot 0.5\cdot 2.5\cdot 1.5\times 10^{-2}=3.75\times 10^{-2}.
\]
Exponentiate:
\begin{equation}
\boxed{\;G_{\rm rt}\simeq e^{0.0375}\simeq 1.04\;(\sim 4\%);\;\text{no gain on }2\!\to\!1A.\;}
\label{eq:Grt-num}
\end{equation}

\subsection*{(g) Maximum efficiency and losses}

Quantum-defect efficiency is the ratio of laser to pump photon energy:
\[
\eta_{\rm QD}=\frac{E_{\rm laser}}{E_{\rm pump}}=\frac{\lambda_{\rm pump}}{\lambda_{\rm laser}}.
\]
Compute laser wavenumber $2\to 1B$:
\[
\frac{E_{\rm laser}^{B}}{hc}=(1\,026\,000-106\,000)\,\si{m^{-1}}=920\,000\,\si{m^{-1}}.
\]
Compute pump wavenumber at $\lambda_{\rm pump}=\SI{910}{nm}$:
\[
\frac{E_{\rm pump}}{hc}=\frac{1}{\lambda_{\rm pump}}=\frac{1}{910\times 10^{-9}\,\si{m}}\simeq 1\,099\,000\,\si{m^{-1}}.
\]
Form the ratio:
\[
\eta_{\rm QD}^{B}=\frac{920\,000}{1\,099\,000}\simeq 0.837.
\]
Compute laser wavenumber $2\to 1A$:
\[
\frac{E_{\rm laser}^{A}}{hc}=(1\,026\,000-60\,000)\,\si{m^{-1}}=966\,000\,\si{m^{-1}}.
\]
Form the ratio:
\[
\eta_{\rm QD}^{A}=\frac{966\,000}{1\,099\,000}\simeq 0.879.
\]
Multiply by $\eta_{\rm diode}=0.60$ and $\eta_{\rm abs}=0.86$ for $B$:
\[
\eta_{\rm max}^{B}=0.60\cdot 0.86\cdot 0.837\simeq 0.43.
\]
Repeat for $A$:
\[
\eta_{\rm max}^{A}=0.60\cdot 0.86\cdot 0.879\simeq 0.45.
\]
\begin{equation}
\boxed{\;\eta_{\rm max}\simeq 43\text{--}45\%.\;}
\label{eq:eta-max}
\end{equation}

\textit{Practical losses.} Ground-state absorption on the laser line (especially $2\to 1B$, quasi-three-level); excited-state absorption from level $2$; non-radiative quenching by impurities or Yb clusters; cavity scattering; sub-optimal output coupling; pump-heating broadening of the emission line.

\section*{Question 4 --- Radiative transfer and lithium absorption}
%==================================================================

\textbf{Problem.} Recast $\de I_\nu/\de z=-\alpha_\nu I_\nu+j_\nu$ in optical-depth form, solve for constant $S_\nu$, and use measured ratios $I_A:I_B:I_C=0.85:1:0.15$ to extract $\tau_\nu$ and $S_\nu/I_\nu(0)$. Show $\sigma_{\rm abs}^{\rm res}=\lambda_0^2/(2\pi)$ for a lifetime-broadened line, evaluate for Li $2s\to 2p$ at \SI{671}{nm}, give the column density for $\tau\sim 1$ in a \SI{1}{cm} cell, and discuss when natural-width lines are observed; list the source requirements for the Li $2s\to np$ Rydberg series.

\subsection*{(a) Optical-depth form and formal solution}

Define:
\[
\de\tau_\nu=\alpha_\nu\,\de z,\qquad S_\nu=\frac{j_\nu}{\alpha_\nu}.
\]
Divide transfer equation by $\alpha_\nu$:
\[
\frac{\de I_\nu}{\de\tau_\nu}=-I_\nu+S_\nu.
\]
Multiply by integrating factor $e^{\tau_\nu}$:
\[
\frac{\de}{\de\tau_\nu}\!\left(I_\nu e^{\tau_\nu}\right)=S_\nu e^{\tau_\nu}.
\]
Integrate from $0$ to $\tau_\nu$ at constant $S_\nu$:
\[
I_\nu(\tau_\nu)e^{\tau_\nu}-I_\nu(0)=S_\nu(e^{\tau_\nu}-1).
\]
Solve for $I_\nu$:
\begin{equation}
\boxed{\;I_\nu(\tau_\nu)=I_\nu(0)e^{-\tau_\nu}+S_\nu(1-e^{-\tau_\nu}).\;}
\label{eq:RTsol}
\end{equation}
Beer's law in the limit $\tau_\nu\to 0$; black-body $S_\nu$ in the limit $\tau_\nu\to\infty$.

\subsection*{(b) Inversion of the flame measurement}

Identify the three measurements with \eqref{eq:RTsol}:
\[
I_A=I_\nu(0)e^{-\tau_\nu}+S_\nu(1-e^{-\tau_\nu}),\quad I_B=I_\nu(0),\quad I_C=S_\nu(1-e^{-\tau_\nu}).
\]
Subtract $I_C$ from $I_A$:
\[
I_A-I_C=I_B\,e^{-\tau_\nu}.
\]
Divide by $I_B$, set $X=I_A/I_B$, $Y=I_C/I_B$:
\[
e^{-\tau_\nu}=X-Y.
\]
Insert numerical values:
\[
e^{-\tau_\nu}=0.85-0.15=0.70.
\]
Take log:
\begin{equation}
\boxed{\;\tau_\nu=-\ln 0.70\simeq 0.357.\;}
\label{eq:tau-num}
\end{equation}
Read $S_\nu/I_\nu(0)$ from $I_C/I_B$:
\[
\frac{S_\nu}{I_\nu(0)}=\frac{Y}{1-e^{-\tau_\nu}}=\frac{Y}{1-(X-Y)}.
\]
Insert numerical values:
\[
\frac{S_\nu}{I_\nu(0)}=\frac{0.15}{1-0.70}=\frac{0.15}{0.30}.
\]
Evaluate:
\begin{equation}
\boxed{\;\frac{S_\nu}{I_\nu(0)}=0.50.\;}
\label{eq:S-num}
\end{equation}
Optically thin flame; carbon emits with half the incoming black-body brightness at this frequency.

\subsection*{(c) Resonant cross section and the Li $2s\to 2p$ line}

Natural-width Lorentzian normalised in $\omega$, $\int g_L\,\de\omega=1$, with $\Gamma=A_{21}$:
\[
g_L(\omega-\omega_0)=\frac{1}{2\pi}\,\frac{\Gamma}{(\omega-\omega_0)^{2}+(\Gamma/2)^{2}}.
\]
Set $\omega=\omega_0$:
\[
g_L(0)=\frac{\Gamma/(2\pi)}{(\Gamma/2)^{2}}=\frac{\Gamma/(2\pi)}{\Gamma^{2}/4}.
\]
Simplify:
\[
g_L(0)=\frac{2}{\pi\Gamma}=\frac{2}{\pi A_{21}}.
\]
Insert in $\sigma_{\rm abs}=(g_2/g_1)(\pi^2 c^2 A_{21}/\omega_0^2)\,g_L$, take $g_2/g_1=1$:
\[
\sigma_{\rm abs}^{\rm res}=\frac{\pi^{2}c^{2}A_{21}}{\omega_0^{2}}\cdot\frac{2}{\pi A_{21}}.
\]
Cancel $A_{21}$ and one $\pi$:
\[
\sigma_{\rm abs}^{\rm res}=\frac{2\pi c^{2}}{\omega_0^{2}}.
\]
Use $\omega_0=2\pi c/\lambda_0$, so $\omega_0^{2}=4\pi^{2}c^{2}/\lambda_0^{2}$:
\[
\sigma_{\rm abs}^{\rm res}=\frac{2\pi c^{2}\lambda_0^{2}}{4\pi^{2}c^{2}}.
\]
Cancel:
\begin{equation}
\boxed{\;\sigma_{\rm abs}^{\rm res}=\frac{\lambda_0^{2}}{2\pi}\;\;\text{(independent of }A_{21}).\;}
\label{eq:sigma-res}
\end{equation}

For Li, $\lambda_0=\SI{671}{nm}$:
\[
\sigma_{\rm abs}^{\rm res}=\frac{(\SI{671e-9}{m})^{2}}{2\pi}=\frac{4.50\times 10^{-13}\,\si{m^{2}}}{2\pi}.
\]
Divide:
\begin{equation}
\boxed{\;\sigma_{\rm abs}^{\rm res}\simeq\SI{7.2e-14}{m^{2}}.\;}
\label{eq:sigma-Li}
\end{equation}

Column density for $\tau\sim 1$ at $L=\SI{1}{cm}$:
\[
N\sim\frac{1}{\sigma_{\rm abs}^{\rm res}L}=\frac{1}{\SI{7.2e-14}{m^2}\cdot\SI{e-2}{m}}.
\]
Compute the denominator:
\[
N\sim\frac{1}{7.2\times 10^{-16}\,\si{m^{3}}}.
\]
Evaluate:
\begin{equation}
\boxed{\;N_{\rm Li}\sim\SI{1.4e15}{m^{-3}}\;(\sim 10^{9}\,\mathrm{cm^{-3}}).\;}
\label{eq:N-thick}
\end{equation}

\textit{When are pure lifetime-broadened lines seen?} Natural width must dominate Doppler ($\propto\sqrt{T/M}$) and pressure ($\propto N\sigma_{\rm coll}\bar v$) widths: very low temperature (laser cooling, transverse atomic beams with $\bm k\!\cdot\!\bm v\to 0$), very low pressure ($\tau_{\rm coll}\gg\tau_{\rm rad}$), and absence of inhomogeneous mechanisms (Stark, transit-time). Examples: forbidden lines via saturated absorption, Ramsey spectroscopy, single-ion traps.

\subsection*{(d) Source for the $2s\to np$ Li series}

Li ground state $1s^{2}2s$, ionisation $\SI{5.39}{eV}\equiv\SI{43484}{cm^{-1}}$, series limit $\lambda_{\rm lim}\simeq\SI{230}{nm}$.

Required source properties:

(i) \textit{Continuum coverage} from $\sim\SI{671}{nm}$ down to $\sim\SI{230}{nm}$: D$_2$ arc, Xe arc, tungsten + UV continuum, or synchrotron radiation, so absorption appears as dips on a smooth background.

(ii) \textit{High UV brightness}: peak cross section $\propto 1/n^{*5}$ for high Rydberg states; need adequate UV signal-to-noise.

(iii) \textit{Featureless/stable spectrum}: any source structure mimics absorption; alternative is a swept tunable laser of sufficient range.

(iv) \textit{Resolving power} $\Delta\nu/\nu\lesssim 10^{-6}$ to resolve Doppler width at $T\sim\SI{700}{K}$: Echelle, Fourier-transform, or two-photon Doppler-free laser spectroscopy.

(v) \textit{Long-term intensity stability}: faint high-$n$ lines need steady $I/I_0$ ratios over long acquisition.

In short: continuum or broadly tunable laser covering $230\text{--}\SI{700}{nm}$, high UV brightness, excellent stability, recorded at high resolving power against a wavelength reference.
