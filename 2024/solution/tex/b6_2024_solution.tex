\section*{Question 1 --- Sodium under pressure: bcc--fcc transition and X-ray diffraction}

\textbf{Problem.} Na is bcc at $p=0$ ($a_0=\SI{4.27}{\angstrom}$) and undergoes a first-order bcc$\to$fcc transition at $p_c=\SI{65}{GPa}$, with $a=\SI{2.90}{\angstrom}$ just below and $a'=\SI{3.65}{\angstrom}$ just above. Powder XRD uses $\lambda=\SI{0.43}{\angstrom}$; molar mass $M=\SI{23}{g\,mol^{-1}}$.

Count bcc corners (each shared $\times 1/8$) plus one body-centre:
\[
n_{\mathrm{bcc}} = 8\times\tfrac{1}{8} + 1.
\]
Add the two contributions:
\[
n_{\mathrm{bcc}} = 1 + 1 = 2.
\]
Count fcc corners plus six face-centres (each shared $\times 1/2$):
\[
n_{\mathrm{fcc}} = 8\times\tfrac{1}{8} + 6\times\tfrac{1}{2}.
\]
Add the two contributions:
\[
n_{\mathrm{fcc}} = 1 + 3 = 4.
\]
\[
\boxed{\ n_{\mathrm{bcc}}=2,\quad n_{\mathrm{fcc}}=4.\ }
\]

\subsection*{(a)(iii)--(iv) The $(111)$ plane and nearest-neighbour distance in it}

bcc $(111)$ contains only the three corners; the body-centre lies off-plane. Triangle side equals the face diagonal:
\[
d^{\,(111)}_{\mathrm{bcc}} = \sqrt{2}\,a.
\]
Plug in $a=\SI{2.90}{\angstrom}$:
\[
d^{\,(111)}_{\mathrm{bcc}} = \sqrt{2}\times 2.90\,\si{\angstrom} = \SI{4.10}{\angstrom}.
\]
fcc $(111)$ is close-packed; nearest-neighbour separation equals half the face diagonal:
\[
d^{\,(111)}_{\mathrm{fcc}} = \frac{a'}{\sqrt 2}.
\]
Plug in $a'=\SI{3.65}{\angstrom}$:
\[
d^{\,(111)}_{\mathrm{fcc}} = 3.65/\sqrt{2}\,\si{\angstrom} = \SI{2.58}{\angstrom}.
\]

\subsection*{(b)(i) Diffraction selection rules and reflection positions}

Bragg + cubic spacing $d_{hk\ell}=a/\sqrt{h^2+k^2+\ell^2}$:
\begin{equation}
\sin\theta = \frac{\lambda}{2a}\sqrt{N},\qquad N\equiv h^2+k^2+\ell^2.
\label{eq:braggcubic}
\end{equation}
Selection rules:
\begin{equation}
\boxed{\ \text{bcc: }h+k+\ell\text{ even};\quad \text{fcc: }h,k,\ell\text{ all even or all odd.}\ }
\label{eq:selrules}
\end{equation}
Allowed $N$ sequences:
\[
N_{\mathrm{bcc}}: 2,4,6,8,\dots\ (hk\ell)=110,200,211,220,\dots
\]
\[
N_{\mathrm{fcc}}: 3,4,8,11,12,\dots\ (hk\ell)=111,200,220,311,222,\dots
\]
Window $2\theta<\SI{25}{\degree}$ requires $\sin\theta<0.2164$.

bcc, $\lambda/(2a)=0.07414$, $\sqrt N<2.919$, $N\le 8$:
\begin{equation}
\begin{aligned}
(110): N=2,&\quad 2\theta=\SI{12.04}{\degree};\\
(200): N=4,&\quad 2\theta=\SI{17.04}{\degree};\\
(211): N=6,&\quad 2\theta=\SI{20.90}{\degree};\\
(220): N=8,&\quad 2\theta=\SI{24.18}{\degree}.
\end{aligned}
\label{eq:bccpeaks}
\end{equation}
fcc, $\lambda/(2a')=0.05890$, $\sqrt N<3.674$, $N\le 12$:
\begin{equation}
\begin{aligned}
(111): N=3,&\quad 2\theta=\SI{11.71}{\degree};\\
(200): N=4,&\quad 2\theta=\SI{13.53}{\degree};\\
(220): N=8,&\quad 2\theta=\SI{19.18}{\degree};\\
(311): N=11,&\quad 2\theta=\SI{22.55}{\degree};\\
(222): N=12,&\quad 2\theta=\SI{23.55}{\degree}.
\end{aligned}
\label{eq:fccpeaks}
\end{equation}

\subsection*{(b)(iii) Lattice parameter at $p_1=\SI{100}{GPa}$}

Third allowed fcc reflection is $(220)$ with $N=8$. Shift up by $\SI{1.20}{\degree}$:
\[
2\theta^{(1)}_{220} = 19.18 + 1.20 = \SI{20.38}{\degree}.
\]
Halve to get $\theta$:
\[
\theta^{(1)}_{220} = \SI{10.19}{\degree}.
\]
Evaluate the sine:
\[
\sin\theta^{(1)}_{220} = \sin(\SI{10.19}{\degree}) = 0.17688.
\]
Apply \eqref{eq:braggcubic} with $\sqrt N=2\sqrt 2$:
\[
a'_1 = \frac{\lambda\sqrt N}{2\sin\theta^{(1)}_{220}}.
\]
Substitute the numbers:
\[
a'_1 = \frac{0.43\times 2\sqrt{2}}{2\times 0.17688}\,\si{\angstrom}.
\]
Reduce the numerator and denominator:
\[
a'_1 = \frac{1.2162}{0.35376}\,\si{\angstrom} = 3.438\,\si{\angstrom}.
\]
\begin{equation}
\boxed{\ a'_1 = \SI{3.44}{\angstrom}.\ }
\label{eq:a1box}
\end{equation}

\subsection*{(c) Hard-sphere atomic radius and density change at $p_c$}

bcc spheres touch along the body diagonal $\sqrt 3 a = 4r$:
\[
r_{\mathrm{bcc}} = \frac{\sqrt 3}{4}a.
\]
Plug in $a=\SI{2.90}{\angstrom}$:
\[
r_{\mathrm{bcc}} = \frac{\sqrt 3}{4}\times 2.90\,\si{\angstrom} = \SI{1.256}{\angstrom}.
\]
fcc spheres touch along the face diagonal $\sqrt 2 a' = 4r$:
\[
r_{\mathrm{fcc}} = \frac{\sqrt 2}{4}a'.
\]
Plug in $a'=\SI{3.65}{\angstrom}$:
\[
r_{\mathrm{fcc}} = \frac{\sqrt 2}{4}\times 3.65\,\si{\angstrom} = \SI{1.291}{\angstrom}.
\]
Number density $n=n_{\mathrm{cell}}/a^3$, bcc:
\[
n_b = \frac{2}{(2.90\times 10^{-10})^3}\,\si{m^{-3}}.
\]
Evaluate:
\[
n_b = \frac{2}{24.389\times 10^{-30}}\,\si{m^{-3}} = 8.20\times 10^{28}\,\si{m^{-3}}.
\]
Same for fcc:
\[
n_f = \frac{4}{(3.65\times 10^{-10})^3}\,\si{m^{-3}}.
\]
Evaluate:
\[
n_f = \frac{4}{48.627\times 10^{-30}}\,\si{m^{-3}} = 8.226\times 10^{28}\,\si{m^{-3}}.
\]
Mass density $\rho=nM/\NA$, bcc:
\[
\rho_{\mathrm{bcc}} = \frac{8.20\times 10^{28}\times 23\times 10^{-3}}{6.022\times 10^{23}}\,\si{kg\,m^{-3}}.
\]
Reduce:
\[
\rho_{\mathrm{bcc}} = \SI{3131}{kg\,m^{-3}}.
\]
fcc:
\[
\rho_{\mathrm{fcc}} = \frac{8.226\times 10^{28}\times 23\times 10^{-3}}{6.022\times 10^{23}}\,\si{kg\,m^{-3}}.
\]
Reduce:
\[
\rho_{\mathrm{fcc}} = \SI{3141}{kg\,m^{-3}}.
\]
Take the difference:
\[
\Delta\rho = \rho_{\mathrm{fcc}} - \rho_{\mathrm{bcc}} = +\SI{10}{kg\,m^{-3}}.
\]
Form the fractional change:
\[
\Delta\rho/\rho_{\mathrm{bcc}} = 10/3131 = 0.32\%.
\]
\begin{equation}
\boxed{\ \Delta\rho/\rho_{\mathrm{bcc}} = +0.32\%,\quad \Delta\rho \approx +\SI{10}{kg\,m^{-3}}.\ }
\label{eq:drhobox}
\end{equation}

\subsection*{(d) Compressibility of the bcc phase between ambient and $p_c$}

Treat $\beta=-(1/V)\partial V/\partial p$ as constant and integrate:
\[
V(p) = V_0\,e^{-\beta p}.
\]
Solve for $\beta$ at $p=p_c$:
\[
\beta = -\frac{1}{p_c}\ln\frac{V}{V_0}.
\]
Use $V\propto a^3$:
\[
\beta = -\frac{3}{p_c}\ln\frac{a}{a_0}.
\]
Evaluate the log:
\[
\ln(2.90/4.27) = -0.38670.
\]
Plug in numbers with $p_c=\SI{65e9}{Pa}$:
\[
\beta_{\mathrm{bcc}} = \frac{3\times 0.38670}{6.5\times 10^{10}}\,\si{Pa^{-1}}.
\]
Reduce:
\[
\beta_{\mathrm{bcc}} = \frac{1.1601}{6.5\times 10^{10}}\,\si{Pa^{-1}} = 1.785\times 10^{-11}\,\si{Pa^{-1}}.
\]
\begin{equation}
\boxed{\ \beta_{\mathrm{bcc}} = \SI{1.78e-11}{Pa^{-1}}.\ }
\label{eq:betabox}
\end{equation}

\subsection*{(e) Mass of sodium that fits in the pressure cell}

Cylindrical volume with radius $50\,\si{\micro m}$ and height $1\,\si{mm}$:
\[
V_{\mathrm{cyl}} = \pi(50\times 10^{-6})^2\times 10^{-3}\,\si{m^3}.
\]
Evaluate:
\[
V_{\mathrm{cyl}} = \pi\times 2.5\times 10^{-9}\times 10^{-3}\,\si{m^3} = 7.854\times 10^{-12}\,\si{m^3}.
\]
With $35\%$ void, Na occupies $0.65\,V_{\mathrm{cyl}}$:
\[
V_{\mathrm{Na}} = 0.65\times 7.854\times 10^{-12}\,\si{m^3} = 5.105\times 10^{-12}\,\si{m^3}.
\]
Ambient bcc density uses $a_0^3=(4.27\times 10^{-10})^3 = \SI{77.854e-30}{m^3}$:
\[
\rho_0 = \frac{2\times 23\times 10^{-3}}{6.022\times 10^{23}\times 77.854\times 10^{-30}}\,\si{kg\,m^{-3}}.
\]
Reduce:
\[
\rho_0 = \frac{46\times 10^{-3}}{4.689\times 10^{-5}}\,\si{kg\,m^{-3}} = \SI{981}{kg\,m^{-3}}.
\]
Mass:
\[
m_{\mathrm{Na}} = \rho_0\,V_{\mathrm{Na}} = 981\times 5.105\times 10^{-12}\,\si{kg}.
\]
Reduce:
\[
m_{\mathrm{Na}} = 5.008\times 10^{-9}\,\si{kg}.
\]
\begin{equation}
\boxed{\ m_{\mathrm{Na}} \approx \SI{5.0e-9}{kg} \approx \SI{5}{ng}.\ }
\label{eq:massbox}
\end{equation}

\section*{Question 2 --- Heat capacity of YbMg$_2$Bi$_2$ in 3D and 2D}

\textbf{Problem.} 3D molar heat capacity is $C=z\NA\kB[(\pi^2/2)(T/\TF)+(12\pi^4/5)(T/\TD)^3]$ at low $T$, saturating at high $T$. (a) Identify $z$ from the saturation. (b) Justify each term, extract $\TF,\TD$ from $C/T$ vs $T^2$, find cross-over $T^\ast$. (c) For a 2D film with two in-plane polarisations, find $g_{\mathrm{ph}}^{(2)}(\omega)$, the high-$T$ limit, and the low-$T$ form $C=AT+BT^\delta$ with $A(g(\EF))$, $B(\omega_D)$. Use $\int_0^\infty x^3/(e^x-1)\,\de x=\pi^4/15$, $\int_0^\infty x^2/(e^x-1)\,\de x=2\zeta(3)\simeq 2.4$.

\subsection*{(a) Origin of the high-$T$ saturation and Dulong--Petit limit}

Equipartition: each of $3N$ classical modes contributes $\kB$:
\[
C_{\mathrm{lat}}^{(\infty)} = 3N\kB.
\]
Per mole of formula units use $N=z\NA$:
\[
C^{(\infty)} = 3z\NA\kB = 3zR.
\]
Read $C^{(\infty)}\simeq \SI{120}{J\,mol^{-1}\,K^{-1}}$ from the figure:
\[
z = \frac{120}{3R} = \frac{120}{3\times 8.314}.
\]
Reduce:
\[
z \approx 4.81.
\]
\begin{equation}
\boxed{\ z=5\text{ atoms per formula unit.}\ }
\label{eq:zbox}
\end{equation}

\subsection*{(b)(i) Justification of the two terms}

\textbf{Linear electronic term.} Sommerfeld expansion to leading order in $T/\TF$:
\begin{equation}
U_{\mathrm{el}}(T) = U_{\mathrm{el}}(0) + \tfrac{\pi^2}{6}g(\EF)(\kB T)^2.
\label{eq:Uel}
\end{equation}
Differentiate:
\begin{equation}
C_{\mathrm{el}} = \tfrac{\pi^2}{3}g(\EF)\kB^2 T \equiv \gamma T.
\label{eq:CelGamma}
\end{equation}
Free-electron form $g(\EF) = 3N_e/(2\kB\TF)$. Substitute:
\[
C_{\mathrm{el}} = \frac{\pi^2}{3}\cdot\frac{3N_e}{2\kB\TF}\,\kB^2 T.
\]
Cancel and tidy:
\[
C_{\mathrm{el}} = \frac{\pi^2}{2}\,N_e\kB\,\frac{T}{\TF}.
\]
Identify $N_e=z\NA$ to recover the stated prefactor.

\textbf{Cubic Debye term.} 3D phonon DOS (3 acoustic branches):
\begin{equation}
g_{\mathrm{ph}}(\omega) = \frac{3V\omega^2}{2\pi^2 v_s^3}.
\label{eq:gph3D}
\end{equation}
Internal energy is a Bose integral over the DOS:
\[
U_{\mathrm{ph}}(T) = \int_0^{\omega_D}\frac{\hbar\omega}{e^{\hbar\omega/\kB T}-1}g_{\mathrm{ph}}(\omega)\,\de\omega.
\]
Substitute $x=\hbar\omega/\kB T$:
\[
U_{\mathrm{ph}}(T) = \frac{3V}{2\pi^2 v_s^3 \hbar^3}(\kB T)^4 \int_0^{x_D}\frac{x^3}{e^x-1}\,\de x.
\]
Low $T$, extend the upper limit to $\infty$:
\[
\int_0^{\infty}\frac{x^3}{e^x-1}\,\de x = \frac{\pi^4}{15}.
\]
Insert this value:
\[
U_{\mathrm{ph}}(T) \approx \frac{3V}{2\pi^2 v_s^3 \hbar^3}(\kB T)^4 \cdot \frac{\pi^4}{15}.
\]
Eliminate $V/v_s^3$ via the Debye condition $\omega_D^3 = 6\pi^2 N v_s^3/V$:
\[
\frac{V}{v_s^3} = \frac{6\pi^2 N}{\omega_D^3}.
\]
Use $\hbar\omega_D = \kB\TD$:
\[
\frac{V}{v_s^3} = \frac{6\pi^2 N\hbar^3}{(\kB\TD)^3}.
\]
Substitute into $U_{\mathrm{ph}}$:
\[
U_{\mathrm{ph}}(T) = \frac{3(\kB T)^4}{2\pi^2 \hbar^3}\cdot\frac{6\pi^2 N\hbar^3}{(\kB\TD)^3}\cdot\frac{\pi^4}{15}.
\]
Cancel $\hbar^3$ and $2\pi^2$:
\[
U_{\mathrm{ph}}(T) = \frac{3\pi^4}{5}\,N\kB\,\frac{T^4}{\TD^3}.
\]
Differentiate with respect to $T$:
\[
C_{\mathrm{ph}}(T) = \frac{12\pi^4}{5}\,N\kB\,\frac{T^3}{\TD^3}.
\]
Per mole use $N=z\NA$:
\begin{equation}
\boxed{\ C_{\mathrm{ph}}(T) = \frac{12\pi^4}{5}\,z\NA\kB\,\frac{T^3}{\TD^3}.\ }
\label{eq:CphlowT}
\end{equation}

\subsection*{(b)(ii) Reading $\TF$ and $\TD$ from the $C/T$ vs $T^2$ plot}

Model:
\begin{equation}
\frac{C}{T} = \gamma + \beta T^2,\quad \gamma=\frac{\pi^2 z R}{2\TF},\ \beta=\frac{12\pi^4 zR}{5\TD^3}.
\label{eq:CTline}
\end{equation}
Read intercept $\gamma\simeq\SI{0.30}{J\,mol^{-1}\,K^{-2}}$, slope $\beta\simeq \SI{1.6e-3}{J\,mol^{-1}\,K^{-4}}$.

Solve for $\TF$:
\[
\TF = \frac{\pi^2 zR}{2\gamma}.
\]
Plug in numbers:
\[
\TF = \frac{\pi^2\times 5\times 8.314}{2\times 0.30}\,\si{K}.
\]
Reduce:
\[
\TF = \frac{410.3}{0.60}\,\si{K} \approx \SI{684}{K}.
\]
\begin{equation}
\boxed{\ \TF \approx \SI{7e2}{K}.\ }
\label{eq:TFbox}
\end{equation}

Solve for $\TD^3$:
\[
\TD^3 = \frac{12\pi^4\,zR}{5\beta}.
\]
Plug in numbers:
\[
\TD^3 = \frac{12\pi^4\times 5\times 8.314}{5\times 1.6\times 10^{-3}}\,\si{K^3}.
\]
Reduce numerator $12\pi^4 \times 5 \times 8.314 = 4.86\times 10^4$:
\[
\TD^3 = \frac{4.86\times 10^4}{8.0\times 10^{-3}}\,\si{K^3} = 6.07\times 10^6\,\si{K^3}.
\]
Cube root:
\[
\TD \approx \SI{183}{K}.
\]
\begin{equation}
\boxed{\ \TD \approx \SI{2e2}{K}.\ }
\label{eq:TDbox}
\end{equation}

\subsection*{(b)(iii) Cross-over temperature $T^\ast$}

Equate the two contributions:
\[
\frac{\pi^2}{2}\,\frac{T^*}{\TF} = \frac{12\pi^4}{5}\!\left(\frac{T^*}{\TD}\right)^3.
\]
Cancel one factor of $T^*$ and rearrange:
\[
1 = \frac{24\pi^2 (T^*)^2 \TF}{5\TD^3}.
\]
Solve for $(T^*)^2$:
\[
(T^*)^2 = \frac{5\TD^3}{24\pi^2 \TF}.
\]
Plug in $\TD=\SI{183}{K}$, $\TF=\SI{684}{K}$:
\[
(T^*)^2 = \frac{5\times (183)^3}{24\pi^2\times 684}\,\si{K^2}.
\]
Reduce numerator $5\times 183^3 = 3.064\times 10^7$:
\[
(T^*)^2 = \frac{3.064\times 10^7}{1.620\times 10^5}\,\si{K^2} = 189\,\si{K^2}.
\]
Take the square root:
\[
T^* \approx \SI{13.7}{K}.
\]
\begin{equation}
\boxed{\ T^* \approx \SI{14}{K}.\ }
\label{eq:Tstarbox}
\end{equation}

\subsection*{(c)(i) 2D phonon density of states}

Two in-plane polarisations, density $A/(2\pi)^2$ per polarisation in $\mathbf q$-space:
\[
g_{\mathrm{ph}}^{(2)}(\omega)\,\de\omega = 2\cdot\frac{A}{(2\pi)^2}\cdot 2\pi q\,\de q.
\]
Use linear dispersion $\omega = v_s q$ so $q = \omega/v_s$ and $\de q = \de\omega/v_s$:
\[
g_{\mathrm{ph}}^{(2)}(\omega)\,\de\omega = \frac{A}{\pi}\,\frac{\omega}{v_s}\,\frac{\de\omega}{v_s}.
\]
Combine the $v_s$ factors:
\[
g_{\mathrm{ph}}^{(2)}(\omega)\,\de\omega = \frac{A\omega}{\pi v_s^2}\,\de\omega.
\]
\begin{equation}
\boxed{\ g_{\mathrm{ph}}^{(2)}(\omega) = \frac{A\omega}{\pi v_s^2}.\ }
\label{eq:g2Dbox}
\end{equation}

\subsection*{(c)(ii) High-$T$ Dulong--Petit in 2D}

Total in-plane modes $=2N$, equipartition gives $\kB$ each:
\[
C^{(\infty)}_{2D} = 2N\kB.
\]
Per mole use $N=z\NA$ and $z=5$:
\[
C^{(\infty)}_{\mathrm{molar}} = 2zR = 10R.
\]
Plug in $R=8.314$:
\[
C^{(\infty)}_{\mathrm{molar}} = 10\times 8.314 \approx \SI{83}{J\,mol^{-1}\,K^{-1}}.
\]
\begin{equation}
\boxed{\ C^{(\infty)}_{\mathrm{molar}} = 2zR = 10R \approx \SI{83}{J\,mol^{-1}\,K^{-1}}.\ }
\label{eq:DP2Dbox}
\end{equation}

\subsection*{(c)(iii) Low-$T$ form $C=AT+BT^\delta$ in 2D}

Phonon energy with 2D DOS:
\[
U_{\mathrm{ph}}^{(2)}(T) = \int_0^{\omega_D}\frac{\hbar\omega}{e^{\hbar\omega/\kB T}-1}\,\frac{A\omega}{\pi v_s^2}\,\de\omega.
\]
Substitute $x=\hbar\omega/\kB T$, so $\omega = \kB T\,x/\hbar$ and $\de\omega = \kB T\,\de x/\hbar$:
\[
U_{\mathrm{ph}}^{(2)}(T) = \frac{A}{\pi v_s^2 \hbar^2}(\kB T)^3\int_0^{x_D}\frac{x^2}{e^x-1}\,\de x.
\]
Low $T$, extend the upper limit to $\infty$:
\[
\int_0^{\infty}\frac{x^2}{e^x-1}\,\de x = 2\zeta(3).
\]
Insert this value:
\[
U_{\mathrm{ph}}^{(2)}(T) \approx \frac{2\zeta(3)\,A}{\pi v_s^2 \hbar^2}(\kB T)^3.
\]
Differentiate with respect to $T$:
\[
C^{(2)}_{\mathrm{ph}}(T) = \frac{6\zeta(3)\,A\kB^3}{\pi v_s^2 \hbar^2}\,T^2.
\]
\begin{equation}
\boxed{\ C^{(2)}_{\mathrm{ph}}(T) = \frac{6\zeta(3)A\kB^3}{\pi v_s^2 \hbar^2}\,T^2 \equiv B T^\delta,\quad \delta=2.\ }
\label{eq:CphLowT2D}
\end{equation}

\subsection*{(c)(iv) Identification of $A$ in terms of $g(\EF)$, and $B$ in terms of $\omega_D$}

Sommerfeld result \eqref{eq:CelGamma} is dimension-independent:
\begin{equation}
\boxed{\ A = \tfrac{\pi^2}{3}\kB^2\,g(\EF).\ }
\label{eq:Aresult}
\end{equation}

Debye condition: total in-plane modes equals $2N$:
\[
2N = \int_0^{\omega_D}\frac{A\omega}{\pi v_s^2}\,\de\omega.
\]
Integrate:
\[
2N = \frac{A\omega_D^2}{2\pi v_s^2}.
\]
Rearrange:
\[
\frac{A}{\pi v_s^2} = \frac{4N}{\omega_D^2}.
\]
Substitute into the result for $C^{(2)}_{\mathrm{ph}}$:
\[
C^{(2)}_{\mathrm{ph}} = \frac{6\zeta(3)\kB^3}{\hbar^2}\cdot\frac{4N}{\omega_D^2}\,T^2.
\]
Combine factors:
\[
C^{(2)}_{\mathrm{ph}} = \frac{24\zeta(3)\,N\kB^3}{\hbar^2\omega_D^2}\,T^2.
\]
Per mole use $N=z\NA$:
\begin{equation}
\boxed{\ B = \frac{24\zeta(3)\,z\NA\kB^3}{\hbar^2\omega_D^2}.\ }
\label{eq:Bbox}
\end{equation}

\section*{Question 3 --- Tight-binding band on a 2D square lattice}

\textbf{Problem.} 2D square lattice (parameter $a$), tight-binding $\epsilon(k_x,k_y)=\epsilon_0+2t\cos k_xa+2t\cos k_ya$, $t>0$. Find dispersion along $\Gamma\to X\to M$, bandwidth, contours, effective masses; the linear $\eF=\eF'+\alpha N/A$ near the band minimum and $g(\epsilon)=\beta A$; for $A_{\mathrm{FS}}/A_{\mathrm{BZ}}=1/100$ at $a=\SI{3.5}{\angstrom}$, find $n_{2D}$ and Hall voltage.

\subsection*{(a)(i) Band dispersion along $\Gamma\to X\to M$}

Evaluate at $\Gamma=(0,0)$:
\[
\epsilon(\Gamma) = \epsilon_0 + 2t\cos 0 + 2t\cos 0 = \epsilon_0 + 4t.
\]
Evaluate at $X=(\pi/a,0)$:
\[
\epsilon(X) = \epsilon_0 + 2t\cos\pi + 2t\cos 0 = \epsilon_0.
\]
Evaluate at $M=(\pi/a,\pi/a)$:
\[
\epsilon(M) = \epsilon_0 + 2t\cos\pi + 2t\cos\pi = \epsilon_0 - 4t.
\]

\end{center}

\subsection*{(a)(ii) Bandwidth}

\begin{equation}
\boxed{\ W = \epsilon_{\max}-\epsilon_{\min} = 8t.\ }
\label{eq:Wbox}
\end{equation}

\subsection*{(a)(iii) Constant-energy contours}

Expand around $M=(\pi/a,\pi/a)$, $\mathbf k=\mathbf K_M+\mathbf q$, using $\cos(\pi+qa)\simeq -1+(qa)^2/2$:
\[
\epsilon(\mathbf k) \simeq \epsilon_0 + 2t[-1+(q_xa)^2/2] + 2t[-1+(q_ya)^2/2].
\]
Collect terms:
\begin{equation}
\epsilon(\mathbf k) \simeq \epsilon_0-4t + ta^2(q_x^2+q_y^2).
\label{eq:expM}
\end{equation}
Expand around $\Gamma$, using $\cos(qa)\simeq 1-(qa)^2/2$:
\[
\epsilon(\mathbf k) \simeq \epsilon_0 + 2t[1-(q_xa)^2/2] + 2t[1-(q_ya)^2/2].
\]
Collect terms:
\begin{equation}
\epsilon(\mathbf k) \simeq \epsilon_0+4t - ta^2(q_x^2+q_y^2).
\label{eq:expGamma}
\end{equation}
Both contours are circular near the extrema. Half-filling: $\epsilon=\epsilon_0$ gives $\cos k_xa = -\cos k_ya$:
\[
k_xa = \pm(\pi - k_ya).
\]
The contour is a square rotated $45^\circ$ joining the four $X$-points.

\subsection*{(a)(iv) Effective masses near band edges}

Use $1/m^* = \hbar^{-2}\,\partial^2\epsilon/\partial q^2$. From \eqref{eq:expM}:
\[
\frac{\partial^2 \epsilon}{\partial q_x^2} = 2ta^2.
\]
Invert:
\[
m^*_{\mathrm{min}} = +\frac{\hbar^2}{2ta^2} > 0\quad(\text{electron-like}).
\]
From \eqref{eq:expGamma}:
\[
\frac{\partial^2 \epsilon}{\partial q_x^2} = -2ta^2.
\]
Invert:
\[
m^*_{\mathrm{max}} = -\frac{\hbar^2}{2ta^2} < 0\quad(\text{hole-like}).
\]
\begin{equation}
\boxed{\ m^*_{\mathrm{min}} = +\frac{\hbar^2}{2ta^2},\qquad m^*_{\mathrm{max}} = -\frac{\hbar^2}{2ta^2}.\ }
\label{eq:meffbox}
\end{equation}

\subsection*{(b)(i)--(ii) $\eF(N/A)$ near the band minimum}

Parabolic band near $M$ with $m^*=\hbar^2/(2ta^2)$. Count states inside the Fermi disc, factor 2 for spin:
\[
N(\eF) = 2\cdot\frac{A}{(2\pi)^2}\cdot \pi q_F^2.
\]
Simplify:
\[
N(\eF) = \frac{A q_F^2}{2\pi}.
\]
From \eqref{eq:expM} relate $q_F$ to $\eF$:
\[
ta^2 q_F^2 = \eF - (\epsilon_0 - 4t).
\]
Solve for $q_F^2$:
\[
q_F^2 = \frac{\eF - \epsilon_0 + 4t}{ta^2}.
\]
Substitute into $N/A$:
\[
\frac{N}{A} = \frac{\eF - \epsilon_0 + 4t}{2\pi t a^2}.
\]
Solve for $\eF$:
\[
\eF = (\epsilon_0 - 4t) + 2\pi t a^2\,\frac{N}{A}.
\]
\begin{equation}
\boxed{\ \eF = \eF' + \alpha\,\frac{N}{A},\quad \eF'=\epsilon_0-4t,\ \alpha = 2\pi t a^2.\ }
\label{eq:eFlinear}
\end{equation}

\subsection*{(b)(iii) Density of states near the band minimum}

Differentiate $N$ with respect to $\eF$:
\[
g(\epsilon) = \pd{N}{\epsilon}.
\]
Apply to $N = A(\eF-\epsilon_0+4t)/(2\pi ta^2)$:
\[
g(\epsilon) = \frac{A}{2\pi t a^2}.
\]
Identify $\beta$ via $g=\beta A$:
\begin{equation}
\boxed{\ \beta = \frac{1}{2\pi t a^2}.\ }
\label{eq:gbox}
\end{equation}

\subsection*{(c)(i) Carrier density for $A_{\mathrm{FS}}/A_{\mathrm{BZ}}=1/100$}

BZ area $A_{\mathrm{BZ}}=(2\pi/a)^2$. Spin factor 2 in counting:
\[
\frac{N}{A} = 2\cdot\frac{A_{\mathrm{FS}}}{(2\pi)^2}.
\]
Use $A_{\mathrm{FS}} = A_{\mathrm{BZ}}/100 = (2\pi/a)^2/100$:
\[
\frac{N}{A} = \frac{2}{100}\cdot\frac{(2\pi/a)^2}{(2\pi)^2}.
\]
Cancel $(2\pi)^2$:
\[
\frac{N}{A} = \frac{1}{50\,a^2}.
\]
Plug in $a=\SI{3.5e-10}{m}$:
\[
n_{2D} = \frac{1}{50\times(3.5\times 10^{-10})^2}\,\si{m^{-2}}.
\]
Reduce $50\times(3.5)^2 = 612.5$:
\[
n_{2D} = \frac{1}{6.125\times 10^{-18}}\,\si{m^{-2}}.
\]
\begin{equation}
\boxed{\ n_{2D} = \SI{1.63e17}{m^{-2}}.\ }
\label{eq:nbox}
\end{equation}

\subsection*{(c)(ii) Hall voltage}

2D Hall coefficient (single carrier, magnitude):
\[
R_H^{(2D)} = \frac{1}{n_{2D}\,e}.
\]
Hall voltage across the strip equals $E_y\times w$ but the 2D current density is per unit width:
\[
V_H = \frac{j_x B_z}{n_{2D}\,e}.
\]
Plug in $j_x = \SI{10}{\micro A\,m^{-1}} = \SI{1.0e-5}{A\,m^{-1}}$ and $B_z=\SI{1}{T}$:
\[
V_H = \frac{1.0\times 10^{-5}\times 1}{1.63\times 10^{17}\times 1.602\times 10^{-19}}\,\si{V}.
\]
Evaluate the denominator:
\[
1.63\times 10^{17}\times 1.602\times 10^{-19} = 2.612\times 10^{-2}.
\]
Reduce:
\[
V_H = \frac{1.0\times 10^{-5}}{2.612\times 10^{-2}}\,\si{V} = 3.83\times 10^{-4}\,\si{V}.
\]
\begin{equation}
\boxed{\ V_H \approx \SI{3.8e-4}{V} \approx \SI{0.38}{mV}.\ }
\label{eq:VHbox}
\end{equation}

\section*{Question 4 --- Magnetism: Hund's rules, mean-field BCC ferromagnet}

\textbf{Problem.} (a) Hund's rules for Mn$^{2+}$ ($3d^5$), Tb$^{3+}$ ($4f^8$); $T=0$ moments. (b) $S=1/2$ Heisenberg + Zeeman on bcc (edge $a$): mean-field $B_{\mathrm{mf}}$, self-consistent $M$, Curie--Weiss $\chi$, $\Theta=T_C$, find $J_1$ for $T_C=\SI{30}{K}$. (c) Same on simple cubic with $T_C'=\SI{20}{K}$; find $J_1'$; sketch $\chi^{-1}(T)$.

\subsection*{(a)(i) Hund's rules for Mn$^{2+}$ ($3d^5$) and Tb$^{3+}$ ($4f^8$)}

\textbf{Mn$^{2+}$, $3d^5$.} Five $d$ electrons, one in each $m_\ell\in\{-2,-1,0,1,2\}$, all spin up:
\[
S = 5\times\tfrac12 = \tfrac52.
\]
Sum the $m_\ell$ values:
\[
L = -2-1+0+1+2 = 0.
\]
Half-filled shell with $L=0$ gives $J=S$:
\[
J = \tfrac52.
\]
Apply the Land\'e formula with $S(S+1)=35/4$, $L(L+1)=0$, $J(J+1)=35/4$:
\[
g_J = \tfrac32 + \frac{35/4 - 0}{2\times 35/4}.
\]
Reduce:
\[
g_J = \tfrac32 + \tfrac12 = 2.
\]
\begin{equation}
\boxed{\ \mathrm{Mn^{2+}}:\ S=\tfrac52,\ L=0,\ J=\tfrac52,\ g_J=2.\ }
\label{eq:Mnbox}
\end{equation}

\textbf{Tb$^{3+}$, $4f^8$.} Place 7 electrons one in each $m_\ell\in\{-3,\dots,3\}$ then the 8th pairs into $m_\ell=+3$:
\[
S = \tfrac{7}{2} - \tfrac12 = 3.
\]
Sum the $m_\ell$ values: the 7 aligned electrons give $L=0$, the extra spin-down at $m_\ell = +3$ adds:
\[
L = 0 + 3 = 3.
\]
More than half-full shell gives $J=L+S$:
\[
J = 3 + 3 = 6.
\]
Apply Land\'e with $S(S+1)=12$, $L(L+1)=12$, $J(J+1)=42$:
\[
g_J = \tfrac32 + \frac{12 - 12}{2\times 42}.
\]
Reduce:
\[
g_J = \tfrac32 + 0 = \tfrac32.
\]
\begin{equation}
\boxed{\ \mathrm{Tb^{3+}}:\ S=3,\ L=3,\ J=6,\ g_J=\tfrac32.\ }
\label{eq:Tbbox}
\end{equation}

\subsection*{(a)(ii) Saturation magnetic moment at $T=0$}

At $T=0$ the moment is $m=g_J J\,\muB$. Mn$^{2+}$:
\[
m_{\mathrm{Mn^{2+}}} = 2\times\tfrac52\,\muB = 5\,\muB.
\]
Tb$^{3+}$:
\[
m_{\mathrm{Tb^{3+}}} = \tfrac32\times 6\,\muB = 9\,\muB.
\]
\begin{equation}
\boxed{\ m_{\mathrm{Mn^{2+}}} = 5\,\muB,\qquad m_{\mathrm{Tb^{3+}}} = 9\,\muB.\ }
\label{eq:msatbox}
\end{equation}

\subsection*{(b)(i) Mean-field effective field on a bcc site}

bcc coordination $z=8$. Replace $\mathbf S_j\to\langle\mathbf S\rangle$ in the Heisenberg sum acting on site $i$:
\begin{equation}
\mathcal H_i^{\mathrm{MF}} = -J_1 z\langle\mathbf S\rangle\cdot\mathbf S_i + g\muB\mathbf B\cdot\mathbf S_i.
\label{eq:HMF}
\end{equation}
Define $\mathbf B_{\mathrm{mf}}$ by $\mathcal H_i^{\mathrm{MF}} \equiv g\muB\mathbf B_{\mathrm{mf}}\cdot\mathbf S_i$:
\[
\mathbf B_{\mathrm{mf}} = \mathbf B - \frac{J_1 z}{g\muB}\langle\mathbf S\rangle.
\]
Magnetisation in terms of $\langle\mathbf S\rangle$ uses $\bm\mu = -g\muB\mathbf S$:
\[
\mathbf M = -n g\muB\langle\mathbf S\rangle.
\]
Solve for $\langle\mathbf S\rangle$:
\[
\langle\mathbf S\rangle = -\frac{\mathbf M}{ng\muB}.
\]
Substitute into $\mathbf B_{\mathrm{mf}}$:
\[
\mathbf B_{\mathrm{mf}} = \mathbf B + \frac{J_1 z}{(g\muB)^2 n}\mathbf M.
\]
Identify $\lambda$ via $\mathbf B_{\mathrm{mf}} = \mathbf B + \lambda\mathbf M$:
\[
\lambda = \frac{J_1 z}{(g\muB)^2 n}.
\]
Insert $z=8$ and $n=2/a^3$:
\[
\lambda = \frac{8 J_1}{(g\muB)^2\cdot 2/a^3}.
\]
Simplify:
\begin{equation}
\boxed{\ \lambda = \frac{4 J_1 a^3}{(g\muB)^2}.\ }
\label{eq:lambdabox}
\end{equation}

\subsection*{(b)(ii) Self-consistent magnetisation}

Single $S=1/2$ in $B_{\mathrm{mf}}$ has partition function:
\[
Z = 2\cosh\!\left(\frac{g\muB B_{\mathrm{mf}}}{2\kB T}\right).
\]
Spin expectation along the field:
\[
\langle S_z\rangle = -\tfrac12\tanh\!\left(\frac{g\muB B_{\mathrm{mf}}}{2\kB T}\right).
\]
Magnetisation $M = -ng\muB\langle S_z\rangle$:
\[
M = \frac{ng\muB}{2}\tanh\!\left(\frac{g\muB B_{\mathrm{mf}}}{2\kB T}\right).
\]
Identify $M_0 = ng\muB/2$:
\[
M = M_0\tanh\!\left(\frac{g\muB B_{\mathrm{mf}}}{2\kB T}\right).
\]
Insert $n=2/a^3$:
\[
M_0 = \frac{2}{a^3}\cdot\frac{g\muB}{2}.
\]
Simplify:
\begin{equation}
\boxed{\ M_0 = \frac{g\muB}{a^3}.\ }
\label{eq:M0box}
\end{equation}

\subsection*{(b)(iii) Curie--Weiss susceptibility above $T_C$}

Expand $\tanh x\approx x$ for small argument, with $B_{\mathrm{mf}}=B+\lambda M$:
\[
M \simeq M_0\cdot\frac{g\muB(B+\lambda M)}{2\kB T}.
\]
Collect terms linear in $M$:
\[
M\!\left(1 - \frac{M_0 g\muB \lambda}{2\kB T}\right) = \frac{M_0 g\muB}{2\kB T}\,B.
\]
Divide and identify $\chi=M/B$:
\[
\chi = \frac{M_0 g\muB/(2\kB T)}{1 - M_0 g\muB \lambda/(2\kB T)}.
\]
Multiply top and bottom by $T$:
\[
\chi = \frac{M_0 g\muB/(2\kB)}{T - M_0 g\muB \lambda/(2\kB)} \equiv \frac{C}{T-\Theta}.
\]
Compute $\Theta$ with $M_0=g\muB/a^3$ and $\lambda=4J_1 a^3/(g\muB)^2$:
\[
\Theta = \frac{g\muB}{a^3}\cdot g\muB\cdot\frac{4J_1 a^3}{(g\muB)^2}\cdot\frac{1}{2\kB}.
\]
Cancel $(g\muB)^2$ and $a^3$:
\[
\Theta = \frac{4J_1}{2\kB} = \frac{2J_1}{\kB}.
\]
Compute $C$:
\[
C = \frac{M_0 g\muB}{2\kB} = \frac{g\muB}{a^3}\cdot\frac{g\muB}{2\kB}.
\]
Simplify:
\[
C = \frac{(g\muB)^2}{2\kB a^3}.
\]
\begin{equation}
\boxed{\ \Theta = \frac{2J_1}{\kB},\qquad C = \frac{(g\muB)^2}{2\kB a^3}.\ }
\label{eq:CWbox}
\end{equation}

\subsection*{(b)(iv) Relation $\Theta = T_C$}

At $B=0$ the self-consistent equation reads:
\[
M = M_0\tanh\!\left(\frac{g\muB\lambda M}{2\kB T}\right).
\]
A non-zero $M$ first appears when the slope at $M=0$ equals 1:
\[
\frac{M_0 g\muB \lambda}{2\kB T_C} = 1.
\]
Solve for $T_C$:
\[
T_C = \frac{M_0 g\muB \lambda}{2\kB} = \Theta.
\]
\begin{equation}
\boxed{\ T_C = \Theta = \frac{2J_1}{\kB}.\ }
\label{eq:TCbox}
\end{equation}

\subsection*{(b)(v) Numerical $J_1$ for $T_C=\SI{30}{K}$}

Solve $T_C = 2J_1/\kB$ for $J_1$:
\[
J_1 = \frac{\kB T_C}{2}.
\]
Plug in $T_C=\SI{30}{K}$:
\[
J_1 = \frac{1.381\times 10^{-23}\times 30}{2}\,\si{J}.
\]
Reduce:
\[
J_1 = 2.072\times 10^{-22}\,\si{J}.
\]
Convert to meV via $1\,\mathrm{eV} = 1.602\times 10^{-19}\,\si{J}$:
\[
J_1 = \frac{2.072\times 10^{-22}}{1.602\times 10^{-19}}\,\si{eV} = \SI{1.29}{meV}.
\]
\begin{equation}
\boxed{\ J_1 \approx \SI{2.07e-22}{J} \approx \SI{1.3}{meV},\quad J_1/\kB \approx \SI{15}{K}.\ }
\label{eq:J1box}
\end{equation}

\subsection*{(c)(i) $J_1'$ for the simple-cubic lattice with $T_C'=\SI{20}{K}$}

Simple cubic: $z'=6$, $n'=1/a^3$. Repeat (b)(i):
\[
\lambda' = \frac{J_1' z'}{(g\muB)^2 n'} = \frac{6 J_1'}{(g\muB)^2\cdot 1/a^3}.
\]
Simplify:
\[
\lambda' = \frac{6J_1' a^3}{(g\muB)^2}.
\]
Repeat (b)(ii) with $n'=1/a^3$:
\[
M_0' = \frac{n' g\muB}{2} = \frac{g\muB}{2a^3}.
\]
Compute $\Theta'$:
\[
\Theta' = \frac{M_0' g\muB \lambda'}{2\kB}.
\]
Substitute:
\[
\Theta' = \frac{g\muB}{2a^3}\cdot g\muB\cdot\frac{6J_1' a^3}{(g\muB)^2}\cdot\frac{1}{2\kB}.
\]
Cancel $(g\muB)^2$ and $a^3$:
\[
\Theta' = \frac{6J_1'}{4\kB} = \frac{3J_1'}{2\kB}.
\]
Set $T_C'=\Theta'$ and solve:
\[
J_1' = \frac{2\kB T_C'}{3}.
\]
Plug in $T_C'=\SI{20}{K}$:
\[
J_1' = \frac{2\times 1.381\times 10^{-23}\times 20}{3}\,\si{J}.
\]
Reduce:
\[
J_1' = \frac{5.524\times 10^{-22}}{3}\,\si{J} = 1.841\times 10^{-22}\,\si{J}.
\]
Convert to meV:
\[
J_1' = \frac{1.841\times 10^{-22}}{1.602\times 10^{-19}}\,\si{eV} = \SI{1.15}{meV}.
\]
\begin{equation}
\boxed{\ J_1' \approx \SI{1.84e-22}{J} \approx \SI{1.15}{meV},\quad J_1'/\kB \approx \SI{13.3}{K}.\ }
\label{eq:J1pbox}
\end{equation}

\subsection*{(c)(ii) Sketch of $\chi^{-1}(T)$ for both materials}

$\chi^{-1}=(T-\Theta)/C$, slope $1/C$, $T$-intercept $\Theta$. With $C_{\mathrm{bcc}}=(g\muB)^2/(2\kB a^3)$ and $C_{\mathrm{sc}}=(g\muB)^2/(4\kB a^3)$, the sc slope is twice the bcc slope.

\begin{center}
\begin{tikzpicture}[xscale=0.10, yscale=0.85]
