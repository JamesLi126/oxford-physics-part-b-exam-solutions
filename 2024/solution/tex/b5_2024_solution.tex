\section*{Question 1 --- Perfect-fluid conservation, modified Einstein equations}

\textbf{Problem.} (a) Find a current $j^\alpha$ obeying $j^\alpha{}_{;\alpha}=U^\alpha P_{,\alpha}$ from $T^{\alpha\beta}{}_{;\alpha}=0$ for the perfect-fluid stress-energy. (b) Show that pressureless flow is geodesic. (c) The set $R_{\alpha\beta}-(1/n)Rg_{\alpha\beta}=0$: count equations, find the special $n$, show $R$ is constant. (d) Show that a trace-modified Einstein equation reduces to standard Einstein with cosmological constant. (e) Find $\Lambda$ for $R_{\alpha\beta\mu\nu}=(K/12)(g_{\alpha\mu}g_{\beta\nu}-g_{\alpha\nu}g_{\beta\mu})$ in vacuum.

\subsection*{(a) Current from $T^{\alpha\beta}{}_{;\alpha}=0$}

Take the divergence of $T^{\alpha\beta}=Pg^{\alpha\beta}+(\rho+P/c^2)U^\alpha U^\beta$, using $g^{\alpha\beta}{}_{;\alpha}=0$:
\begin{equation}
T^{\alpha\beta}{}_{;\alpha}=g^{\alpha\beta}P_{,\alpha}+\bigl[(\rho c^2+P)U^\alpha U^\beta/c^2\bigr]_{;\alpha}=0.
\label{eq:Tdiv}
\end{equation}
Define the candidate current:
\[
j^\alpha\equiv (\rho c^2+P)U^\alpha.
\]
Insert into \eqref{eq:Tdiv}:
\[
g^{\alpha\beta}P_{,\alpha}+\frac{1}{c^2}\bigl(j^\alpha U^\beta\bigr)_{;\alpha}=0.
\]
Expand the product:
\[
\frac{1}{c^2}j^\alpha{}_{;\alpha}U^\beta+\frac{1}{c^2}j^\alpha U^\beta{}_{;\alpha}=-P^{,\beta}.
\]
Contract with $U_\beta$, using $U_\beta U^\beta=-c^2$:
\[
-j^\alpha{}_{;\alpha}+\frac{1}{c^2}j^\alpha U_\beta U^\beta{}_{;\alpha}=-U_\beta P^{,\beta}.
\]
Apply $U_\beta U^\beta{}_{;\alpha}=\tfrac12(U_\beta U^\beta)_{;\alpha}=0$:
\[
j^\alpha{}_{;\alpha}=U^\beta P_{,\beta}.
\]
\begin{equation}
\boxed{\ j^\alpha=(\rho c^2+P)U^\alpha,\qquad j^\alpha{}_{;\alpha}=U^\alpha P_{,\alpha}.\ }
\label{eq:jcurrent}
\end{equation}

\subsection*{(b) Pressureless flow is geodesic}

Set $P=0$ in $T^{\alpha\beta}=\rho U^\alpha U^\beta$. Take divergence:
\[
(\rho U^\alpha)_{;\alpha}U^\beta+\rho U^\alpha U^\beta{}_{;\alpha}=0.
\]
Contract with $U_\beta$ ($U_\beta U^\beta=-c^2$, $U_\beta U^\beta{}_{;\alpha}=0$):
\[
-c^2(\rho U^\alpha)_{;\alpha}=0.
\]
Hence rest-mass conservation:
\[
(\rho U^\alpha)_{;\alpha}=0.
\]
Substitute back:
\[
\rho U^\alpha U^\beta{}_{;\alpha}=0.
\]
\begin{equation}
\boxed{\ U^\alpha U^\beta{}_{;\alpha}=0.\ }
\label{eq:geodesic}
\end{equation}

\subsection*{(c) The set $R_{\alpha\beta}-(1/n)Rg_{\alpha\beta}=0$}

In 4D the symmetric tensor $R_{\alpha\beta}$ has $4(4+1)/2=10$ independent components, so the equations are 10. Take the trace $g^{\alpha\beta}(\cdot)$:
\[
R-\tfrac{4}{n}R=0\ \Longrightarrow\ \Bigl(1-\tfrac{4}{n}\Bigr)R=0.
\]
For $n=4$ the trace is automatically satisfied and is therefore not an independent equation:
\begin{equation}
\boxed{\ n=4\ \Rightarrow\ 9\text{ independent equations.}\ }
\label{eq:nfour}
\end{equation}
Take the divergence of $R^\beta_{\ \alpha}=(R/n)\delta^\beta_{\ \alpha}$:
\[
R^\beta_{\ \alpha;\beta}=\tfrac{1}{n}R_{,\alpha}.
\]
Contracted Bianchi $R_{,\alpha}=2R^\beta_{\ \alpha;\beta}$:
\[
R_{,\alpha}=\tfrac{2}{n}R_{,\alpha}\ \Longrightarrow\ \Bigl(1-\tfrac{2}{n}\Bigr)R_{,\alpha}=0.
\]
With $n=4$:
\[
\tfrac12 R_{,\alpha}=0\ \Longrightarrow\ \boxed{\ R=\text{const.}\ }
\]

\subsection*{(d) Trace-free modified Einstein $\Rightarrow$ standard Einstein $+\Lambda$}

Modified equation:
\begin{equation}
R_{\alpha\beta}-\tfrac{1}{4}Rg_{\alpha\beta}=-\tfrac{8\pi G}{c^4}\bigl(T_{\alpha\beta}-\tfrac{1}{4}Tg_{\alpha\beta}\bigr).
\label{eq:tracefreemod}
\end{equation}
Take divergence; use Bianchi $\nabla^\beta R_{\alpha\beta}=\tfrac12 R_{,\alpha}$ and $T^{\alpha\beta}{}_{;\alpha}=0$:
\[
\tfrac12 R_{,\alpha}-\tfrac14 R_{,\alpha}=-\tfrac{8\pi G}{c^4}\Bigl(0-\tfrac14 T_{,\alpha}\Bigr).
\]
Simplify both sides:
\[
\tfrac14 R_{,\alpha}=\tfrac{2\pi G}{c^4}T_{,\alpha}.
\]
Integrate (constant of integration $\equiv 4\Lambda$):
\begin{equation}
R=\tfrac{8\pi G}{c^4}T+4\Lambda.
\label{eq:Rint}
\end{equation}
Substitute \eqref{eq:Rint} into \eqref{eq:tracefreemod}:
\[
R_{\alpha\beta}-\tfrac{1}{4}\bigl(\tfrac{8\pi G}{c^4}T+4\Lambda\bigr)g_{\alpha\beta}=-\tfrac{8\pi G}{c^4}T_{\alpha\beta}+\tfrac{2\pi G}{c^4}Tg_{\alpha\beta}.
\]
Collect $T$ terms:
\[
R_{\alpha\beta}=\Lambda g_{\alpha\beta}+\tfrac{4\pi G}{c^4}Tg_{\alpha\beta}-\tfrac{8\pi G}{c^4}T_{\alpha\beta}.
\]
Use \eqref{eq:Rint} to write $\tfrac12 Rg_{\alpha\beta}=\tfrac{4\pi G}{c^4}Tg_{\alpha\beta}+2\Lambda g_{\alpha\beta}$:
\[
R_{\alpha\beta}-\tfrac12 Rg_{\alpha\beta}=\Lambda g_{\alpha\beta}-2\Lambda g_{\alpha\beta}-\tfrac{8\pi G}{c^4}T_{\alpha\beta}.
\]
Rearrange:
\begin{equation}
\boxed{\ R_{\alpha\beta}-\tfrac12 Rg_{\alpha\beta}-\Lambda g_{\alpha\beta}=-\tfrac{8\pi G}{c^4}T_{\alpha\beta}.\ }
\label{eq:Einsteinplus}
\end{equation}
This is the standard Einstein equation with cosmological constant $\Lambda$.

\subsection*{(e) Maximally symmetric vacuum: $\Lambda$ from $K$}

Use the paper's convention $R_{\alpha\beta}\equiv -R^\gamma_{\ \alpha\beta\gamma}$. Raise the first index of the given Riemann tensor:
\[
R^\gamma_{\ \alpha\beta\delta}=\tfrac{K}{12}\bigl(\delta^\gamma_{\ \beta}g_{\alpha\delta}-\delta^\gamma_{\ \delta}g_{\alpha\beta}\bigr).
\]
Contract on $\gamma=\delta$:
\[
-R^\gamma_{\ \alpha\beta\gamma}=-\tfrac{K}{12}\bigl(g_{\alpha\beta}-4g_{\alpha\beta}\bigr).
\]
Simplify:
\[
R_{\alpha\beta}=\tfrac{K}{4}g_{\alpha\beta}.
\]
Trace:
\[
R=K.
\]
Substitute into \eqref{eq:Einsteinplus} with $T_{\alpha\beta}=0$:
\[
\tfrac{K}{4}g_{\alpha\beta}-\tfrac{K}{2}g_{\alpha\beta}-\Lambda g_{\alpha\beta}=0.
\]
Solve:
\begin{equation}
\boxed{\ \Lambda=-\tfrac{K}{4}.\ }
\label{eq:Lambdabox}
\end{equation}

\section*{Question 2 --- Reissner--Nordstr\"om geodesics}

\textbf{Problem.} Charged BH metric, $A(r)=1-2GM/(c^2r)+GQ^2/(c^4r^2)$. (a) Conserved $k,h$. (b) Effective-potential equation. (c) Particle from rest at infinity: turning point $r_*$. (d) Free fall from rest at $r_+$ when $r_Q>r_G$: $\dot r^2$ form, motion. (e) Proper time from $r_+$ to $r_-$.

\subsection*{(a) Conserved quantities along the geodesic}

The metric is independent of $t$ and $\phi$. Lagrangian $2\mathcal L=g_{\mu\nu}\dot x^\mu\dot x^\nu$ with $\theta=\pi/2$, $\dot\theta=0$. Euler--Lagrange in $t$:
\[
\frac{\de}{\de\tau}\bigl(g_{tt}\dot t\bigr)=0\ \Longrightarrow\ -Ac^2\dot t=\text{const}.
\]
Define $k$ by $A\dot t=k$:
\begin{equation}
\boxed{\ A\dot t=k.\ }
\label{eq:k}
\end{equation}
Euler--Lagrange in $\phi$:
\[
\frac{\de}{\de\tau}\bigl(r^2\sin^2\theta\,\dot\phi\bigr)=0\ \Longrightarrow\ r^2\dot\phi=h.
\]
\begin{equation}
\boxed{\ r^2\dot\phi=h.\ }
\label{eq:h}
\end{equation}

\subsection*{(b) Effective-potential equation}

Normalisation $g_{\mu\nu}\dot x^\mu\dot x^\nu=-c^2$ on the equator:
\[
-Ac^2\dot t^{\,2}+\frac{\dot r^{\,2}}{A}+r^2\dot\phi^{\,2}=-c^2.
\]
Substitute \eqref{eq:k},\eqref{eq:h}:
\[
-\frac{c^2k^2}{A}+\frac{\dot r^{\,2}}{A}+\frac{h^2}{r^2}=-c^2.
\]
Multiply by $A$:
\[
\dot r^{\,2}=c^2k^2-c^2A-\frac{Ah^2}{r^2}.
\]
Halve and rearrange:
\[
\tfrac12 \dot r^{\,2}+\tfrac12 c^2(A-1)+\frac{Ah^2}{2r^2}=\tfrac12 c^2(k^2-1).
\]
Identify $V_{\rm eff}=\tfrac12 c^2(A-1)+Ah^2/(2r^2)$. Insert $A=1-\rG/r+\rQ^2/r^2$, expand $A-1$:
\[
\tfrac12 c^2(A-1)=-\frac{c^2\rG}{2r}+\frac{c^2\rQ^2}{2r^2}.
\]
Expand $Ah^2/(2r^2)$:
\[
\frac{Ah^2}{2r^2}=\frac{h^2}{2r^2}-\frac{h^2\rG}{2r^3}+\frac{h^2\rQ^2}{2r^4}.
\]
Add:
\begin{equation}
\boxed{\ V_{\rm eff}(r)=-\frac{c^2\rG}{2r}+\frac{c^2\rQ^2+h^2}{2r^2}-\frac{h^2\rG}{2r^3}+\frac{(h\rQ)^2}{2r^4}.\ }
\label{eq:Veff}
\end{equation}

\subsection*{(c) Stop point for a particle starting from rest at infinity}

At rest at infinity: $h=0$, $\dot t\to1$, so $k=A(\infty)^{1/2}\dot t=1$ and $k^2-1=0$:
\[
\tfrac12 \dot r^{\,2}=-V_{\rm eff}(r)\Big|_{h=0}=\frac{c^2\rG}{2r}-\frac{c^2\rQ^2}{2r^2}.
\]
Set $\dot r=0$:
\[
\frac{\rG}{r}-\frac{\rQ^2}{r^2}=0.
\]
Solve for $r$:
\begin{equation}
\boxed{\ r_*=\rQ^2/\rG.\ }
\label{eq:rstar}
\end{equation}
At $r=r_*$ the radial acceleration $-V'_{\rm eff}(r_*)$ is positive (charge term repulsive at small $r$): the particle bounces and accelerates back to infinity.

\subsection*{(d) Free fall from $r_+$ when $\rQ>\rG$}

At rest at $r_+$ ($\dot r=0$, $h=0$): $k^2=A(r_+)$. Plug into the radial equation with $h=0$:
\[
\dot r^{\,2}=c^2\bigl(k^2-A(r)\bigr)=c^2\bigl(A(r_+)-A(r)\bigr).
\]
Compute the difference:
\[
A(r_+)-A(r)=\rG\Bigl(\frac1r-\frac1{r_+}\Bigr)-\rQ^2\Bigl(\frac{1}{r^2}-\frac{1}{r_+^{\,2}}\Bigr).
\]
Reduce each fraction over the common denominator $r^2 r_+^{\,2}$:
\[
A(r_+)-A(r)=\frac{(r_+-r)\bigl[\rG r r_+ -\rQ^2(r_++r)\bigr]}{r^2 r_+^{\,2}}.
\]
Factor the bracket as linear in $r$:
\[
\rG r r_+-\rQ^2 r_+-\rQ^2 r=(\rG r_+-\rQ^2)\bigl(r-r_-\bigr),\qquad r_-\equiv\frac{\rQ^2 r_+}{\rG r_+-\rQ^2}.
\]
Use $\rQ^2=\rG r_*$ to rewrite the prefactor and the inner root:
\[
\rG r_+-\rQ^2=\rG(r_+-r_*),\qquad r_-=\frac{r_+ r_*}{r_+-r_*}.
\]
Substitute:
\begin{equation}
\boxed{\ \dot r^{\,2}=\frac{c^2\rG(r_+-r_*)}{r_+^{\,2}\,r^{\,2}}\,(r_+-r)(r-r_-).\ }
\label{eq:rdotsq}
\end{equation}
Both turning points are real with $r_-<r_+$. The particle oscillates between $r_+$ and $r_-$, never reaching $r=0$: the charge term provides an inner centrifugal-like barrier. In Schwarzschild ($Q=0$) one has $r_*=0$, $r_-=0$, so the inner barrier disappears and the particle reaches the singularity.

\subsection*{(e) Proper time $r_+\to r_-$}

Take the negative root of \eqref{eq:rdotsq} (infall):
\[
\de\tau=-\frac{r_+ r}{c\sqrt{\rG(r_+-r_*)}}\,\frac{\de r}{\sqrt{(r_+-r)(r-r_-)}}.
\]
Integrate from $r_+$ to $r_-$:
\[
\tau=\frac{r_+}{c\sqrt{\rG(r_+-r_*)}}\int_{r_-}^{r_+}\frac{r\,\de r}{\sqrt{(r_+-r)(r-r_-)}}.
\]
Substitute $r=\tfrac{r_++r_-}{2}+\tfrac{r_+-r_-}{2}u$, $\de r=\tfrac{r_+-r_-}{2}\de u$:
\[
(r_+-r)(r-r_-)=\tfrac{(r_+-r_-)^2}{4}(1-u^2).
\]
Reduce the integrand:
\[
\frac{r\,\de r}{\sqrt{(r_+-r)(r-r_-)}}=\frac{r}{\sqrt{1-u^2}}\,\de u.
\]
Split $r$ and use parity (the $u$-piece integrates to zero):
\[
\int_{-1}^{1}\frac{r}{\sqrt{1-u^2}}\,\de u=\frac{r_++r_-}{2}\int_{-1}^{1}\frac{\de u}{\sqrt{1-u^2}}=\frac{\pi(r_++r_-)}{2}.
\]
Substitute:
\[
\tau=\frac{\pi r_+(r_++r_-)}{2c\sqrt{\rG(r_+-r_*)}}.
\]
Use the algebraic identity $r_+^{\,2}=(r_+-r_*)(r_++r_-)$ (from $r_-=r_+r_*/(r_+-r_*)$):
\[
\frac{r_+}{\sqrt{\rG(r_+-r_*)}}=\sqrt{\frac{r_++r_-}{\rG}}.
\]
Insert:
\begin{equation}
\boxed{\ \tau=\frac{\pi\rG}{2c}\Bigl(\frac{r_++r_-}{\rG}\Bigr)^{3/2}.\ }
\label{eq:taubox}
\end{equation}

\section*{Question 3 --- Anisotropic cosmology}

\textbf{Problem.} Bianchi-I metric $\de s^2=-c^2\de t^2+e^{2\alpha}\de x^2+e^{2\beta}(\de y^2+\de z^2)$ with anisotropic perfect fluid (pressures $w_A\rho c^2$ along $x$, $w_B\rho c^2$ along $y,z$). (a) Christoffels. (b) $T^{\mu\nu}{}_{;\mu}=0$ gives $\rho\propto e^{-(1+w_A)\alpha-2(1+w_B)\beta}$. (c) Einstein equations. (d) $w_A=w_B=-1$: isotropic stable de Sitter. (e) Volume growth. (f) Two observational tests of cosmic anisotropy.

\subsection*{(a) Christoffel symbols}

Use $x^0=ct$, so $g_{00}=-1$ and $\partial_0=(1/c)\partial_t$. Diagonal metric formula $\Gamma^\lambda_{\mu\nu}=\tfrac12 g^{\lambda\sigma}(g_{\sigma\mu,\nu}+g_{\sigma\nu,\mu}-g_{\mu\nu,\sigma})$ with $\sigma=\lambda$.

Compute $\Gamma^0_{xx}$ ($\mu=\nu=x$, $\lambda=0$):
\[
\Gamma^0_{xx}=\tfrac12 g^{00}(-g_{xx,0})=\tfrac12 (-1)(-\partial_0 e^{2\alpha})=\frac{\dot\alpha\,e^{2\alpha}}{c}.
\]
Same calculation for $\Gamma^0_{yy}$, $\Gamma^0_{zz}$ with $g_{yy}=g_{zz}=e^{2\beta}$:
\[
\Gamma^0_{yy}=\Gamma^0_{zz}=\frac{\dot\beta\,e^{2\beta}}{c}.
\]
Compute $\Gamma^x_{0x}$ ($\mu=0,\nu=x,\lambda=x$):
\[
\Gamma^x_{0x}=\tfrac12 g^{xx}g_{xx,0}=\tfrac12 e^{-2\alpha}\partial_0 e^{2\alpha}=\frac{\dot\alpha}{c}.
\]
Same for $y,z$:
\[
\Gamma^y_{0y}=\Gamma^z_{0z}=\frac{\dot\beta}{c}.
\]
\begin{equation}
\boxed{\ \Gamma^0_{xx}=\frac{\dot\alpha}{c}e^{2\alpha},\quad \Gamma^0_{yy}=\Gamma^0_{zz}=\frac{\dot\beta}{c}e^{2\beta},\quad \Gamma^x_{0x}=\frac{\dot\alpha}{c},\quad \Gamma^y_{0y}=\Gamma^z_{0z}=\frac{\dot\beta}{c}.\ }
\label{eq:gammas}
\end{equation}

\subsection*{(b) Continuity for $\rho$}

The matrix in the question is $T^\mu{}_\nu=\mathrm{diag}(-\rho c^2,w_A\rho c^2,w_B\rho c^2,w_B\rho c^2)$. Then $T^{00}=g^{00}T^0{}_0=\rho c^2$. Conservation $\nabla_\mu T^{\mu 0}=0$ with $T^{i0}=0$:
\[
\partial_0 T^{00}+\Gamma^\mu_{\mu 0}T^{00}+\Gamma^0_{\mu\lambda}T^{\mu\lambda}=0.
\]
Insert $\Gamma^\mu_{\mu 0}=(\dot\alpha+2\dot\beta)/c$:
\[
c\dot\rho+\frac{\dot\alpha+2\dot\beta}{c}\rho c^2+\Gamma^0_{\mu\lambda}T^{\mu\lambda}=0.
\]
Compute $\Gamma^0_{\mu\lambda}T^{\mu\lambda}=\Gamma^0_{xx}T^{xx}+2\Gamma^0_{yy}T^{yy}$, with $T^{xx}=w_A\rho c^2 e^{-2\alpha}$, $T^{yy}=w_B\rho c^2 e^{-2\beta}$:
\[
\Gamma^0_{\mu\lambda}T^{\mu\lambda}=\frac{\dot\alpha w_A+2\dot\beta w_B}{c}\rho c^2.
\]
Combine and divide by $c\rho$:
\[
\frac{\dot\rho}{\rho}=-(1+w_A)\dot\alpha-2(1+w_B)\dot\beta.
\]
Integrate:
\begin{equation}
\boxed{\ \rho(t)\propto e^{-(1+w_A)\alpha(t)-2(1+w_B)\beta(t)}.\ }
\label{eq:rhotbox}
\end{equation}

\subsection*{(c) Einstein field equations}

With $\ln|g|=2\alpha+4\beta$, the given Ricci formula becomes:
\[
R_{\mu\nu}=\tfrac12 \partial_\mu\partial_\nu\ln|g|-\partial_\gamma\Gamma^\gamma_{\mu\nu}+\Gamma^\delta_{\mu\gamma}\Gamma^\gamma_{\nu\delta}-\tfrac12 \Gamma^\gamma_{\mu\nu}\partial_\gamma\ln|g|.
\]

\textbf{$R_{00}$.} Only the first and third terms survive ($\Gamma^\gamma_{00}=0$):
\[
R_{00}=\tfrac{1}{2c^2}(2\ddot\alpha+4\ddot\beta)+\tfrac{1}{c^2}(\dot\alpha^2+2\dot\beta^2).
\]
Simplify:
\[
R_{00}=\tfrac{1}{c^2}\bigl(\ddot\alpha+2\ddot\beta+\dot\alpha^2+2\dot\beta^2\bigr).
\]

\textbf{$R_{xx}$.} Only $\Gamma^0_{xx}$ contributes:
\[
\partial_0\Gamma^0_{xx}=\frac{e^{2\alpha}}{c^2}(\ddot\alpha+2\dot\alpha^2).
\]
Quadratic term $\Gamma^\delta_{x\gamma}\Gamma^\gamma_{x\delta}$: the pairs $(\delta,\gamma)=(0,x),(x,0)$ contribute $(e^{2\alpha}\dot\alpha/c)(\dot\alpha/c)$ each:
\[
\Gamma^\delta_{x\gamma}\Gamma^\gamma_{x\delta}=\frac{2e^{2\alpha}\dot\alpha^2}{c^2}.
\]
Last term:
\[
\tfrac12\Gamma^0_{xx}\partial_0\ln|g|=\tfrac12 \frac{e^{2\alpha}\dot\alpha}{c}\cdot\frac{2\dot\alpha+4\dot\beta}{c}=\frac{e^{2\alpha}\dot\alpha(\dot\alpha+2\dot\beta)}{c^2}.
\]
Combine:
\[
R_{xx}=-\frac{e^{2\alpha}}{c^2}\bigl(\ddot\alpha+\dot\alpha^2+2\dot\alpha\dot\beta\bigr).
\]

\textbf{$R_{yy}$ ($=R_{zz}$ by symmetry).} Same procedure with $\beta\leftrightarrow\alpha$ in the relevant slots:
\[
R_{yy}=-\frac{e^{2\beta}}{c^2}\bigl(\ddot\beta+\dot\alpha\dot\beta+2\dot\beta^2\bigr).
\]

\textbf{Ricci scalar.}
\[
R=g^{00}R_{00}+e^{-2\alpha}R_{xx}+2e^{-2\beta}R_{yy}.
\]
Substitute:
\[
R=-\frac{2}{c^2}\bigl(\ddot\alpha+2\ddot\beta+\dot\alpha^2+3\dot\beta^2+2\dot\alpha\dot\beta\bigr).
\]

\textbf{Einstein equations} $G_{\mu\nu}=R_{\mu\nu}-\tfrac12 Rg_{\mu\nu}=-\tfrac{8\pi G}{c^4}T_{\mu\nu}$.

$tt$-component, $g_{00}=-1$, $T_{00}=\rho c^2$:
\[
G_{00}=R_{00}+\tfrac12 R=-\frac{1}{c^2}\bigl(\dot\beta^2+2\dot\alpha\dot\beta\bigr).
\]
Set $G_{00}=-\tfrac{8\pi G}{c^4}\rho c^2$ and cancel $-1/c^2$:
\begin{equation}
\boxed{\ \dot\beta^2+2\dot\alpha\dot\beta=8\pi G\rho.\ }
\label{eq:Friedmann1}
\end{equation}

$xx$-component, $g_{xx}=e^{2\alpha}$, $T_{xx}=g_{xx}T^x{}_x=e^{2\alpha}w_A\rho c^2$:
\[
G_{xx}=R_{xx}-\tfrac12 R\,e^{2\alpha}=\frac{e^{2\alpha}}{c^2}\bigl(2\ddot\beta+3\dot\beta^2\bigr).
\]
Set $G_{xx}=-\tfrac{8\pi G}{c^4}e^{2\alpha}w_A\rho c^2$ and cancel $e^{2\alpha}/c^2$:
\begin{equation}
\boxed{\ 2\ddot\beta+3\dot\beta^2=-8\pi G w_A\rho.\ }
\label{eq:Friedmann2}
\end{equation}

$yy$-component, $T_{yy}=e^{2\beta}w_B\rho c^2$:
\[
G_{yy}=\frac{e^{2\beta}}{c^2}\bigl(\ddot\alpha+\ddot\beta+\dot\alpha^2+\dot\beta^2+\dot\alpha\dot\beta\bigr).
\]
Cancel $e^{2\beta}/c^2$:
\begin{equation}
\boxed{\ \ddot\alpha+\ddot\beta+\dot\alpha^2+\dot\beta^2+\dot\alpha\dot\beta=-8\pi G w_B\rho.\ }
\label{eq:Friedmann3}
\end{equation}

\subsection*{(d) Cosmological-constant case $w_A=w_B=-1$}

From \eqref{eq:rhotbox} the exponent is $0$:
\[
\rho=\rho_0=\text{const}.
\]
Search for fixed point $\dot\alpha=\dot\beta=H$, $\ddot\alpha=\ddot\beta=0$. Equation \eqref{eq:Friedmann1}:
\[
3H^2=8\pi G\rho_0\ \Longrightarrow\ H=\sqrt{\tfrac{8\pi G\rho_0}{3}}\equiv q.
\]
Check \eqref{eq:Friedmann2}: $0+3H^2=8\pi G\rho_0$. \eqref{eq:Friedmann3}: $0+0+3H^2=8\pi G\rho_0$. Both consistent. Hence isotropic exponential solution $\alpha=\beta=qt$.

\textbf{Linear stability.} Set $\alpha=qt+\delta\alpha$, $\beta=qt+\delta\beta$, both small. Linearise \eqref{eq:Friedmann1}:
\[
2q\,\dot{\delta\beta}+2q(\dot{\delta\alpha}+\dot{\delta\beta})=0.
\]
Simplify:
\[
\dot{\delta\alpha}+2\dot{\delta\beta}=0.
\]
Linearise \eqref{eq:Friedmann2}:
\[
2\,\ddot{\delta\beta}+6q\,\dot{\delta\beta}=0.
\]
Solve:
\[
\dot{\delta\beta}\propto e^{-3qt}\xrightarrow{t\to\infty}0.
\]
And $\dot{\delta\alpha}=-2\dot{\delta\beta}\to 0$. Both shears decay; the isotropic de Sitter solution is stable.
\[
\boxed{\ \rho=\text{const},\quad \alpha=\beta=qt,\quad \text{shear}\propto e^{-3qt}.\ }
\]

\subsection*{(e) Physical volume of a comoving region}

Volume element of a constant-$t$ slice:
\[
\de V=\sqrt{|g_3|}\,\de x\,\de y\,\de z=e^{\alpha+2\beta}\de x\,\de y\,\de z.
\]
Insert $\alpha=\beta=qt$ from (d):
\[
V(t)=V_0\,e^{3qt}.
\]
\begin{equation}
\boxed{\ V(t)\propto e^{3qt}.\ }
\label{eq:Vbox}
\end{equation}

\subsection*{(f) Detecting cosmic anisotropy}

Two probes are:

\textbf{(i) CMB anisotropy pattern.} Bianchi-I expansion imprints a direction-dependent quadrupole / octopole in the CMB temperature map (different recession rates in $x$ vs $y,z$ produce a preferred-axis signature). Stack with foreground-cleaned all-sky maps (Planck) and constrain shear $\Sigma\equiv(\dot\alpha-\dot\beta)/H$ from the absence of an excess large-angle quadrupole.

\textbf{(ii) Direction-dependent Hubble diagram.} Compile Type Ia supernovae (or BAO standard rulers, or strong-lensing time delays) across the sky and fit $H_0(\hat n)$ as a function of direction. A non-zero dipole / quadrupole in $H_0(\hat n)$ above the peculiar-velocity floor implies anisotropic expansion.

\section*{Question 4 --- Gravitational waves from a radial binary}

\textbf{Problem.} Two equal masses $\mu$ on the $z$-axis, separation $\Delta z(t)=L+a\cos\omega t$. (a) Identify configuration; compute $I^{ij}=\int T^{00}r^ir^j\de^3r$. (b) Linearised gravity, find $\bar h^{TT}_{ij}$ propagating along $\hat x$. (c) Motion of a small ring of test masses centred on, perpendicular to, $\hat x$. (d) Total GW luminosity. (e) Energy radiated per period; discuss $L/a$ dependence.

\subsection*{(a) Physical configuration and quadrupole moment}

A pair of equal masses on a spring along $\hat z$ (or two stars in a radial-only oscillation about a fixed mean separation $L$, amplitude $a$). Place the centre of mass at the origin:
\[
z_{1,2}(t)=\mp\tfrac12\bigl(L+a\cos\omega t\bigr),\qquad x_{1,2}=y_{1,2}=0.
\]
For point particles, $T^{00}(t,\vec r)=\mu c^2\bigl[\delta^3(\vec r-\vec r_1)+\delta^3(\vec r-\vec r_2)\bigr]$. Insert into the integral:
\[
I^{ij}=\mu c^2\bigl(r_1^i r_1^j+r_2^i r_2^j\bigr).
\]
Only $i=j=z$ is non-zero, with $z_1^{\,2}=z_2^{\,2}=(L+a\cos\omega t)^2/4$:
\[
I^{zz}=2\mu c^2\cdot\frac{(L+a\cos\omega t)^2}{4}=\frac{\mu c^2}{2}\bigl(L+a\cos\omega t\bigr)^2.
\]
\begin{equation}
\boxed{\ I^{ij}=\delta^i_z\delta^j_z\,\frac{\mu c^2}{2}\bigl(L+a\cos\omega t\bigr)^2.\ }
\label{eq:Ibox}
\end{equation}

\subsection*{(b) Transverse-traceless metric perturbation along $\hat x$}

Linearised quadrupole formula (Lorenz gauge, far zone):
\begin{equation}
\bar h^{ij}(t,\vec r)=\frac{2G}{c^4 r}\,\ddot I^{\,ij}\!\bigl(t-r/c\bigr).
\label{eq:quad}
\end{equation}
Differentiate $I^{zz}$ once:
\[
\dot I^{zz}=\mu c^2(L+a\cos\omega t)(-a\omega\sin\omega t).
\]
Differentiate again:
\[
\ddot I^{zz}=\mu c^2\bigl[a^2\omega^2\sin^2\omega t-(L+a\cos\omega t)a\omega^2\cos\omega t\bigr].
\]
Use $\sin^2\omega t=\tfrac12(1-\cos2\omega t)$ and $\cos^2\omega t=\tfrac12(1+\cos2\omega t)$:
\[
\ddot I^{zz}=-\mu c^2\omega^2\bigl[La\cos\omega t+a^2\cos2\omega t\bigr].
\]
Project to TT along $\hat n=\hat x$. Then $P_{ij}=\mathrm{diag}(0,1,1)$ in $(x,y,z)$, and for a tensor with only $A_{zz}$ nonzero:
\[
A^{TT}_{ij}=P_{iz}P_{jz}A_{zz}-\tfrac12 P_{ij}A_{zz}.
\]
Read off the non-vanishing components:
\[
A^{TT}_{zz}=\bigl(1-\tfrac12\bigr)A_{zz}=\tfrac12 A_{zz},\qquad A^{TT}_{yy}=-\tfrac12 A_{zz},\qquad A^{TT}_{yz}=0.
\]
Apply to $\bar h^{ij}$ via \eqref{eq:quad} with $u\equiv t-r/c$:
\begin{equation}
\boxed{\ \bar h^{TT}_{zz}=-\bar h^{TT}_{yy}=-\frac{G\mu\omega^2}{c^2 r}\bigl[La\cos\omega u+a^2\cos2\omega u\bigr].\ }
\label{eq:hTTbox}
\end{equation}

\subsection*{(c) Motion of the test-mass ring}

The wave is $+$-polarised along $(y,z)$. Geodesic-deviation gives $\ddot{\xi}^i=\tfrac12 \ddot{\bar h}^{TT}_{ij}\xi^j$, integrated:
\[
\xi^i(t)=\xi^i_0+\tfrac12 \bar h^{TT}_{ij}(t)\xi^j_0.
\]
Initial ring of radius $\delta R$ in the $(y,z)$ plane:
\[
\delta y(t)=\delta R\cos\varphi\bigl[1-\tfrac12 \bar h^{TT}_{zz}(t)\bigr],\qquad \delta z(t)=\delta R\sin\varphi\bigl[1+\tfrac12 \bar h^{TT}_{zz}(t)\bigr].
\]
Hence the circle deforms into an ellipse whose principal axes coincide with $\hat y,\hat z$ and whose strain alternates sign at the two harmonics $\omega$ (dominant when $L\gg a$) and $2\omega$. The ring breathes between a $z$-stretched / $y$-squeezed ellipse and a $y$-stretched / $z$-squeezed ellipse, with constant area to linear order in $\bar h$.

  \draw[->] (0,-2.6)--(0,2.6) node[above] {$z$};

\subsection*{(d) Total gravitational-wave luminosity}

Use the standard Einstein quadrupole formula in terms of the trace-free reduced quadrupole $\bar Q_{ij}=Q_{ij}-\tfrac13 Q^k_{\ k}\delta_{ij}$ with $Q_{ij}\equiv I_{ij}/c^2$:
\begin{equation}
\mathcal L=\frac{G}{5c^5}\bigl\langle\dddot{\bar Q}_{ij}\dddot{\bar Q}^{ij}\bigr\rangle.
\label{eq:Lstd}
\end{equation}
Read $Q_{zz}=\mu(L+a\cos\omega t)^2/2$, all other $Q_{ij}=0$. Trace $Q^k_{\ k}=Q_{zz}$:
\[
\bar Q_{zz}=\tfrac23 Q_{zz},\qquad \bar Q_{xx}=\bar Q_{yy}=-\tfrac13 Q_{zz}.
\]
Differentiate $Q_{zz}$ three times. Expand:
\[
Q_{zz}=\tfrac{\mu}{2}\bigl[L^2+2La\cos\omega t+\tfrac{a^2}{2}+\tfrac{a^2}{2}\cos2\omega t\bigr].
\]
Triple time derivative:
\[
\dddot Q_{zz}=\mu\bigl[La\omega^3\sin\omega t+2a^2\omega^3\sin2\omega t\bigr]\equiv \mu S(t).
\]
Square and time-average ($\langle\sin^2\omega t\rangle=\langle\sin^2 2\omega t\rangle=\tfrac12$, cross term averages to zero):
\[
\langle S^2\rangle=\tfrac{1}{2}L^2 a^2\omega^6+2a^4\omega^6=\tfrac12 a^2\omega^6\bigl(L^2+4a^2\bigr).
\]
Sum the trace-free components:
\[
\dddot{\bar Q}_{ij}\dddot{\bar Q}^{ij}=\bigl(\tfrac{4}{9}+2\cdot\tfrac{1}{9}\bigr)\mu^2 S^2=\tfrac23\mu^2 S^2.
\]
Average:
\[
\langle\dddot{\bar Q}_{ij}\dddot{\bar Q}^{ij}\rangle=\tfrac{\mu^2 a^2\omega^6(L^2+4a^2)}{3}.
\]
Insert into \eqref{eq:Lstd}:
\begin{equation}
\boxed{\ \mathcal L=\frac{G\mu^2\omega^6 a^2\bigl(L^2+4a^2\bigr)}{15\,c^5}.\ }
\label{eq:Lboxed}
\end{equation}

\subsection*{(e) Energy radiated per period}

Period $T=2\pi/\omega$:
\[
\Delta E=\mathcal L\,T=\frac{2\pi G\mu^2\omega^5 a^2(L^2+4a^2)}{15 c^5}.
\]
Maximum speed of each mass: $\dot z_{1,2}=\pm \tfrac12 a\omega\sin\omega t$, so $v_{\max}=a\omega/2$, giving $a^2\omega^2=4v_{\max}^2$. Define a translational characteristic velocity $v_L\equiv L\omega/2$ (max speed of a hypothetical mass moving with relative-equilibrium phase $L\omega/2$), so $L^2\omega^2=4v_L^2$. Substitute:
\[
\Delta E=\frac{2\pi G\mu^2\omega^3}{15c^5}\bigl(L^2+4a^2\bigr)\cdot 4v_{\max}^2=\frac{8\pi G\mu^2\omega^3}{15c^5}\bigl(L^2+4a^2\bigr)v_{\max}^2.
\]
Or, eliminating $L,a$ entirely:
\begin{equation}
\boxed{\ \Delta E=\frac{8\pi G\mu^2\omega}{15c^5}\bigl(4v_L^2 v_{\max}^2+4v_{\max}^4\bigr)=\frac{32\pi G\mu^2\omega\,v_{\max}^2}{15c^5}\bigl(v_L^2+v_{\max}^2\bigr).\ }
\label{eq:DeltaEbox}
\end{equation}

\textbf{Discussion of $L/a$.}
\begin{itemize}
\item $L/a\gg 1$ (small radial oscillation about a large separation): the $\omega$-harmonic dominates, $\Delta E\approx\frac{2\pi G\mu^2\omega^3 L^2 a^2}{15c^5}\propto (L/a)^2 a^4\omega^5$. Energy loss grows as $(L/a)^2$, so the radiation is sourced by the mass-monopole modulation and scales linearly with the equilibrium length squared.
\item $L/a\to 0$ (pure breathing through the centre): only the $2\omega$ harmonic survives, $\Delta E\to\frac{8\pi G\mu^2\omega^3 a^4}{15c^5}=\frac{32\pi G\mu^2\omega v_{\max}^4}{15c^5}$. Independent of $L$; this is the lowest-order quadrupole emission of a symmetric oscillator.
\item Crossover at $L/a\sim 2$: the two contributions are comparable, and the GW signal contains a clean two-tone spectrum at $\omega$ and $2\omega$.
\end{itemize}
